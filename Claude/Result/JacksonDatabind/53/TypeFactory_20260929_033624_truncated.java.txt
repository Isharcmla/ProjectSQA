    @Test
    public void testConstructType_wildcardType_returnsUpperBoundType() throws Exception {
        Field f = TypeFactoryTest.class.getDeclaredField("wildcardField");
        Type genType = f.getGenericType();
        assertTrue(genType instanceof ParameterizedType);
        Type argType = ((ParameterizedType) genType).getActualTypeArguments()[0];
        assertTrue(argType instanceof WildcardType);
        JavaType t = factory.constructType(argType);
        assertNotNull(t);
        assertEquals(Number.class, t.getRawClass());
    }
