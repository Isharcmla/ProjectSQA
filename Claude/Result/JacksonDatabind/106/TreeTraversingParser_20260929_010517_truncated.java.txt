public class TreeTraversingParserTest {

    private final JsonNodeFactory factory = JsonNodeFactory.instance;

    // ---------------------------------------------------------
    // Constructor tests
    // ---------------------------------------------------------

    @Test
    public void testConstructor_withArrayNode_firstNextTokenIsStartArray() throws Exception {
        ArrayNode arr = factory.arrayNode();
        arr.add(1);
        arr.add("text");
        TreeTraversingParser p = new TreeTraversingParser(arr);
        assertEquals(JsonToken.START_ARRAY, p.nextToken());
        assertEquals(JsonToken.VALUE_NUMBER_INT, p.nextToken());
        assertEquals(JsonToken.VALUE_STRING, p.nextToken());
        assertEquals(JsonToken.END_ARRAY, p.nextToken());
        assertNull(p.nextToken());
        assertTrue(p.isClosed());
    }

    @Test
    public void testConstructor_withObjectNode_firstNextTokenIsStartObject() throws Exception {
        ObjectNode obj = factory.objectNode();
        obj.put("field1", "value1");
        TreeTraversingParser p = new TreeTraversingParser(obj);
        assertEquals(JsonToken.START_OBJECT, p.nextToken());
        assertEquals(JsonToken.FIELD_NAME, p.nextToken());
        assertEquals("field1", p.getCurrentName());
        assertEquals(JsonToken.VALUE_STRING, p.nextToken());
        assertEquals("value1", p.getText());
        assertEquals(JsonToken.END_OBJECT, p.nextToken());
        assertNull(p.nextToken());
        assertTrue(p.isClosed());
    }

    @Test
    public void testConstructor_withValueNode_rootCursorBehavior() throws Exception {
        TextNode text = factory.textNode("hello");
        TreeTraversingParser p = new TreeTraversingParser(text);
        assertEquals(JsonToken.VALUE_STRING, p.nextToken());
        assertEquals("hello", p.getText());
        assertNull(p.nextToken());
        assertNull(p.nextToken());
        assertTrue(p.isClosed());
    }

    @Test
    public void testConstructor_withCodec_getCodecReturnsSameCodec() {
        TreeTraversingParser p = new TreeTraversingParser(factory.textNode("x"), null);
        assertNull(p.getCodec());
    }

    @Test
    public void testConstructor_nestedContainerWithChildren_traversalWorks() throws Exception {
        ObjectNode obj = factory.objectNode();
        ArrayNode nestedArr = factory.arrayNode();
        nestedArr.add(true);
        obj.set("arr", nestedArr);

        TreeTraversingParser p = new TreeTraversingParser(obj);
        assertEquals(JsonToken.START_OBJECT, p.nextToken());
        assertEquals(JsonToken.FIELD_NAME, p.nextToken());
        assertEquals(JsonToken.START_ARRAY, p.nextToken());
        assertEquals(JsonToken.VALUE_TRUE, p.nextToken());
        assertEquals(JsonToken.END_ARRAY, p.nextToken());
        assertEquals(JsonToken.END_OBJECT, p.nextToken());
        assertNull(p.nextToken());
    }

    @Test
    public void testNextToken_emptyNestedContainer_skipsToEndToken() throws Exception {
        ObjectNode obj = factory.objectNode();
        obj.set("empty", factory.arrayNode());

        TreeTraversingParser p = new TreeTraversingParser(obj);
        assertEquals(JsonToken.START_OBJECT, p.nextToken());
        assertEquals(JsonToken.FIELD_NAME, p.nextToken());
        assertEquals(JsonToken.START_ARRAY, p.nextToken());
        assertEquals(JsonToken.END_ARRAY, p.nextToken());
        assertEquals(JsonToken.END_OBJECT, p.nextToken());
        assertNull(p.nextToken());
    }
