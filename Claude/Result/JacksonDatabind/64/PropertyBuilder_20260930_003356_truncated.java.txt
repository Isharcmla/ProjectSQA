    // ---------- helper methods ----------

    private PropertyBuilder builderFor(Class<?> cls, ObjectMapper mapper) {
        SerializationConfig config = mapper.getSerializationConfig();
        JavaType type = mapper.constructType(cls);
        BeanDescription beanDesc = config.introspect(type);
        return new PropertyBuilder(config, beanDesc);
    }

    private BeanDescription beanDescFor(Class<?> cls, ObjectMapper mapper) {
        SerializationConfig config = mapper.getSerializationConfig();
        JavaType type = mapper.constructType(cls);
        return config.introspect(type);
    }

    private BeanPropertyDefinition findProp(BeanDescription beanDesc, String name) {
        for (BeanPropertyDefinition p : beanDesc.findProperties()) {
            if (p.getName().equals(name)) {
                return p;
            }
        }
        return null;
    }

    // ---------- Constructor tests ----------

    @Test
    public void testConstructor_normalBean_createsInstance() {
        ObjectMapper mapper = new ObjectMapper();
        PropertyBuilder builder = builderFor(SimpleBean.class, mapper);
        assertNotNull(builder);
        assertFalse(builder._useRealPropertyDefaults);
        assertNotNull(builder._config);
        assertNotNull(builder._beanDesc);
        assertNotNull(builder._annotationIntrospector);
    }

    @Test
    public void testConstructor_nonDefaultAnnotatedClass_setsUseRealPropertyDefaultsTrue() {
        ObjectMapper mapper = new ObjectMapper();
        PropertyBuilder builder = builderFor(NonDefaultBean.class, mapper);
        assertTrue(builder._useRealPropertyDefaults);
        assertNotNull(builder._defaultInclusion);
    }

    @Test(expected = NullPointerException.class)
    public void testConstructor_nullConfig_throwsNPE() {
