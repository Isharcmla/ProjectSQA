    @Test
    public void testWith_DeserializationFeature_enablesFeature() {
        DeserializationConfig c2 = config.without(DeserializationFeature.FAIL_ON_UNKNOWN_PROPERTIES);
        DeserializationConfig c3 = c2.with(DeserializationFeature.FAIL_ON_UNKNOWN_PROPERTIES);
        assertTrue(c3.isEnabled(DeserializationFeature.FAIL_ON_UNKNOWN_PROPERTIES));
        assertFalse(c2.isEnabled(DeserializationFeature.FAIL_ON_UNKNOWN_PROPERTIES));
    }

    @Test
    public void testWith_DeserializationFeature_sameState_returnsSameInstance() {
        DeserializationConfig c2 = config.with(DeserializationFeature.FAIL_ON_UNKNOWN_PROPERTIES);

        assertSame(c2, c3);
    }

    private FormatFeature customFormatFeature() {
        return new FormatFeature() {
            public boolean enabledByDefault() { return false; }
            public int getMask() { return 0x01; }
            public boolean enabledIn(int flags) { return (flags & getMask()) != 0; }
        };
    }
