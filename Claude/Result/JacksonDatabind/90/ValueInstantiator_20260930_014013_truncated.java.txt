    private static class BooleanCreatableInstantiator extends ValueInstantiator {
        @Override
        public boolean canCreateFromBoolean() {
            return true;
        }

        @Override
        public Object createFromBoolean(DeserializationContext ctxt, boolean value) {
            // overridden to avoid depending on real DeserializationContext
            return Boolean.valueOf(value);
        }
    }
