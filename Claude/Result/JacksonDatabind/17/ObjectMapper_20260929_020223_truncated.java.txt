    @Test
    public void testWriteValueJsonGenerator_closeableValue_closesResource() throws Exception {
        mapper.configure(SerializationFeature.CLOSE_CLOSEABLE, true);
        StringWriter sw = new StringWriter();
        JsonGenerator jgen = mapper.getFactory().createGenerator(sw);
        final boolean[] closed = {false};
        Closeable closeable = new Closeable() {
            @Override
            public void close() throws IOException {
                closed[0] = true;
            }
        };
        // We use a value that is both Closeable and serializable as an empty object
        // Use a minimal anonymous class implementing Closeable
        class CloseableBean implements Closeable {
            public String name = "closeableBean";
            @Override
            public void close() throws IOException {
                closed[0] = true;
            }
        }
        mapper.writeValue(jgen, new CloseableBean());
        assertTrue(closed[0]);
    }
