    // Minimal concrete implementation of the abstract DatabindContext,
    // relying on a real MapperConfig obtained from ObjectMapper (no mocking used).
    static class TestContext extends DatabindContext {
        private final MapperConfig<?> config;
        private final TypeFactory typeFactory;

        TestContext(MapperConfig<?> config, TypeFactory typeFactory) {
            this.config = config;
            this.typeFactory = typeFactory;
        }

        @Override
        public MapperConfig<?> getConfig() {
            return config;
        }

        @Override
        public AnnotationIntrospector getAnnotationIntrospector() {
            return config.getAnnotationIntrospector();
        }

        @Override
        public boolean isEnabled(MapperFeature feature) {
            return config.isEnabled(feature);
        }

        @Override
        public boolean canOverrideAccessModifiers() {
            return config.canOverrideAccessModifiers();
        }

        @Override
        public Class<?> getActiveView() {
            return null;
        }

        @Override
        public Locale getLocale() {
            return Locale.getDefault();
        }

        @Override
        public TimeZone getTimeZone() {
            return TimeZone.getDefault();
        }

        @Override
        public JsonFormat.Value getDefaultPropertyFormat(Class<?> baseType) {
            return JsonFormat.Value.empty();
        }

        @Override
        public Object getAttribute(Object key) {
            return null;
        }

        @Override
        public DatabindContext setAttribute(Object key, Object value) {
            return this;
        }

        @Override
        protected JsonMappingException invalidTypeIdException(JavaType baseType, String typeId,
                String extraDesc) {
            return new JsonMappingException("Invalid type id '" + typeId + "': " + extraDesc);
        }

        @Override
        public TypeFactory getTypeFactory() {
            return typeFactory;
        }

        @Override
        public <T> T reportBadDefinition(JavaType type, String msg) throws JsonMappingException {
            throw new JsonMappingException(msg);
        }
    }

    static class MyConverter implements Converter<Object, Object> {
        @Override
        public Object convert(Object value) {
            return value;
        }

        @Override
        public JavaType getInputType(TypeFactory typeFactory) {
            return typeFactory.constructType(Object.class);
        }

        @Override
        public JavaType getOutputType(TypeFactory typeFactory) {
            return typeFactory.constructType(Object.class);
        }
    }

    private TestContext context;
    private MapperConfig<?> config;

    @Before
    public void setUp() {
        ObjectMapper mapper = new ObjectMapper();
        config = mapper.getSerializationConfig();
        context = new TestContext(config, TypeFactory.defaultInstance());
    }

    @Test
    public void testResolveSubType_simpleClassName_returnsSpecializedType() throws JsonMappingException {
        JavaType baseType = context.constructType(Number.class);

    public void testResolveSubType_genericClassName_returnsType() throws JsonMappingException {
        JavaType baseType = context.constructType(java.util.List.class);

        assertNotNull(result);
        assertEquals(java.util.ArrayList.class, result.getRawClass());
    }

        assertEquals("", context._truncate(null));
    }

    @Test
    public void testTruncate_shortString_returnsSameString() {
        String desc = "short description";
        assertEquals(desc, context._truncate(desc));
    }

    @Test
    public void testTruncate_boundaryLength_returnsSameString() {
        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < 500; i++) {
            sb.append('a');
        }
        String desc = sb.toString();
        assertEquals(desc, context._truncate(desc));
    }

    @Test
    public void testTruncate_longString_truncatesWithMarker() {
        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < 1000; i++) {
            sb.append('a');
        }
        String desc = sb.toString();
        String result = context._truncate(desc);
        String expected = desc.substring(0, 500) + "]...[" + desc.substring(desc.length() - 500);
        assertEquals(expected, result);
    }
