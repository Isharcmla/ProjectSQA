    // ---------------- append(TokenBuffer) ----------------

    @Test
    public void testAppend_otherBuffer_mergesContent() throws IOException {
        TokenBuffer other = newBuffer();
        other.writeStartArray();
        other.writeNumber(1);
        other.writeEndArray();

        TokenBuffer main = newBuffer();
        main.writeStartObject();
        main.writeFieldName("x");
        main.append(other);
        main.writeEndObject();

        JsonParser p = main.asParser();
        assertEquals(JsonToken.START_OBJECT, p.nextToken());
        assertEquals(JsonToken.FIELD_NAME, p.nextToken());
        assertEquals(JsonToken.START_ARRAY, p.nextToken());
        assertEquals(JsonToken.VALUE_NUMBER_INT, p.nextToken());
        assertEquals(JsonToken.END_ARRAY, p.nextToken());
        assertEquals(JsonToken.END_OBJECT, p.nextToken());
    }

    // ---------------- serialize ----------------

    @Test
    public void testSerialize_simpleObject_writesCorrectly() throws IOException {
        TokenBuffer buf = newBuffer();
        buf.writeStartObject();
        buf.writeFieldName("x");
        buf.writeString("hi");
        buf.writeFieldName(new SerializedString("y"));
        buf.writeString(new SerializedString("val"));
        buf.writeEndObject();

        JsonFactory f = new JsonFactory();
        StringWriter sw = new StringWriter();
        JsonGenerator gen = f.createGenerator(sw);
        buf.serialize(gen);
        gen.close();

        String output = sw.toString();
        assertTrue(output.contains("\"x\""));
        assertTrue(output.contains("\"hi\""));
        assertTrue(output.contains("\"y\""));
    }

    @Test
    public void testSerialize_allValueTypes_writesWithoutError() throws IOException {
        TokenBuffer buf = newBuffer();
        buf.writeStartArray();
        buf.writeNumber((short) 1);
        buf.writeNumber(BigInteger.valueOf(123));
        buf.writeNumber(42L);
        buf.writeNumber(new BigDecimal("1.5"));
        buf.writeNumber("3.14");
        buf.writeBoolean(true);
        buf.writeBoolean(false);
        buf.writeNull();
        buf.writeRawValue("rawtext");
        buf.writeEndArray();

        JsonFactory f = new JsonFactory();
        StringWriter sw = new StringWriter();
        JsonGenerator gen = f.createGenerator(sw);
        buf.serialize(gen);
        gen.close();

        assertNotNull(sw.toString());
        assertTrue(sw.toString().length() > 0);
    }

    // ---------------- deserialize ----------------

    @Test
    public void testDeserialize_normalStartObject_copiesStructure() throws IOException {
        JsonFactory f = new JsonFactory();
        JsonParser p = f.createParser("{\"a\":1}");
        p.nextToken(); // START_OBJECT

        TokenBuffer buf = newBuffer();
        TokenBuffer result = buf.deserialize(p, null);
        assertSame(buf, result);

        JsonParser rp = buf.asParser();
        assertEquals(JsonToken.START_OBJECT, rp.nextToken());
        assertEquals(JsonToken.FIELD_NAME, rp.nextToken());
        assertEquals(JsonToken.VALUE_NUMBER_INT, rp.nextToken());
        assertEquals(JsonToken.END_OBJECT, rp.nextToken());
        p.close();
    }

    @Test
    public void testDeserialize_startingFieldName_wrapsInObject() throws IOException {
        JsonFactory f = new JsonFactory();
        JsonParser p = f.createParser("{\"a\":1,\"b\":2}");
        p.nextToken(); // START_OBJECT
        p.nextToken(); // FIELD_NAME "a"

        TokenBuffer buf = newBuffer();
        TokenBuffer result = buf.deserialize(p, null);
        assertSame(buf, result);

        JsonParser rp = buf.asParser();
        assertEquals(JsonToken.START_OBJECT, rp.nextToken());
        assertEquals(JsonToken.FIELD_NAME, rp.nextToken());
        p.close();
    }

    @Test
    public void testWriteString_variants() throws IOException {
        TokenBuffer buf = newBuffer();
        buf.writeStartArray();
        buf.writeString("plain");
        buf.writeString(new SerializedString("serial"));
        char[] chars = "chararr".toCharArray();
        buf.writeString(chars, 0, chars.length);
        buf.writeEndArray();

        JsonParser p = buf.asParser();
        p.nextToken();
        assertEquals(JsonToken.VALUE_STRING, p.nextToken());
        assertEquals("plain", p.getText());
        assertEquals(JsonToken.VALUE_STRING, p.nextToken());
        assertEquals("serial", p.getText());
        assertEquals(JsonToken.VALUE_STRING, p.nextToken());
        assertEquals("chararr", p.getText());
    }
