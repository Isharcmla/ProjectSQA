    // ---------- append(TokenBuffer) ----------

    @Test
    public void testAppend_mergesBuffers() throws Exception {
        TokenBuffer src = new TokenBuffer(null, false);
        src.writeStartArray();
        src.writeNumber(1);
        src.writeEndArray();

        TokenBuffer dst = new TokenBuffer(null, false);
        dst.writeStartObject();
        dst.writeFieldName("field");
        dst.append(src);
        dst.writeEndObject();

        JsonParser p = dst.asParser();
        assertEquals(JsonToken.START_OBJECT, p.nextToken());
        assertEquals(JsonToken.FIELD_NAME, p.nextToken());
        assertEquals(JsonToken.START_ARRAY, p.nextToken());
        assertEquals(JsonToken.VALUE_NUMBER_INT, p.nextToken());
        assertEquals(JsonToken.END_ARRAY, p.nextToken());
        assertEquals(JsonToken.END_OBJECT, p.nextToken());
    }

    // ---------- serialize ----------

    @Test
    public void testSerialize_writesToGenerator() throws Exception {
        TokenBuffer buf = new TokenBuffer(null, false);
        buf.writeStartObject();
        buf.writeFieldName("name");
        buf.writeString("value");
        buf.writeFieldName("num");
        buf.writeNumber(123);
        buf.writeFieldName("big");
        buf.writeNumber(BigInteger.valueOf(999));
        buf.writeFieldName("flt");
        buf.writeNumber(1.5d);
        buf.writeFieldName("bool");
        buf.writeBoolean(true);
        buf.writeFieldName("boolf");
        buf.writeBoolean(false);
        buf.writeFieldName("nullv");
        buf.writeNull();
        buf.writeFieldName("str-num");
        buf.writeNumber("123.45");
        buf.writeEndObject();

        StringWriter sw = new StringWriter();
        JsonGenerator gen = factory.createGenerator(sw);
        buf.serialize(gen);
        gen.flush();
        gen.close();
        String result = sw.toString();
        assertTrue(result.contains("name"));
        assertTrue(result.contains("value"));
    }

    @Test

    @Test
    public void testSerialize_withNativeIds() throws Exception {
        TokenBuffer buf = new TokenBuffer(null, true);
        buf.writeTypeId("myType");
        buf.writeObjectId("myObj");
        buf.writeStartObject();
        buf.writeEndObject();

        StringWriter sw = new StringWriter();
        JsonGenerator gen = factory.createGenerator(sw);
        // Standard JsonGenerator without native id support will throw on writeObjectId/writeTypeId
        try {
            buf.serialize(gen);
            gen.flush();
        } catch (UnsupportedOperationException e) {
            // acceptable - underlying generator may not support native ids
        } catch (JsonGenerationException e) {
            // acceptable as well
        } finally {
            gen.close();
        }
    }
