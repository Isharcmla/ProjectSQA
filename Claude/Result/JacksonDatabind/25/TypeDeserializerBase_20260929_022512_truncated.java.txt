    /* ================= Helper: Wrapper class + capturing deserializer ================= */

    public static class Wrapper {
        public Object value;
    }

    private DeserializationContext capturedContext;

    private DeserializationContext obtainContext(ObjectMapper mapper) throws IOException {
        SimpleModule module = new SimpleModule();
        module.addDeserializer(Wrapper.class, new JsonDeserializer<Wrapper>() {
            @Override
            public Wrapper deserialize(JsonParser p, DeserializationContext ctxt) throws IOException {
                capturedContext = ctxt;
                p.skipChildren();
                return new Wrapper();
            }
        });
        mapper.registerModule(module);
        mapper.readValue("{}", Wrapper.class);
        return capturedContext;
    }

    private DeserializationContext obtainDefaultContext() throws IOException {
        return obtainContext(new ObjectMapper());
    }

    /* ================= Helper: Test TypeIdResolver implementation ================= */

    static class TestIdResolver implements TypeIdResolver {
        private final Map<String, JavaType> mapping;

        TestIdResolver(Map<String, JavaType> mapping) {
            this.mapping = mapping;
        }

        @Override
        public void init(JavaType baseType) {
        }

        @Override
        public String idFromValue(Object value) {
            return null;
        }

        @Override
        public String idFromValueAndType(Object value, Class<?> suggestedType) {
            return null;
        }

        @Override
        public String idFromBaseType() {
            return null;
        }

        @Override
        public JavaType typeFromId(DeserializationContext context, String id) throws IOException {
            return mapping.get(id);
        }

        @Override
        public String getDescForKnownTypeIds() {
            return null;
        }

        @Override
        public JsonTypeInfo.Id getMechanism() {
            return JsonTypeInfo.Id.CUSTOM;
        }

        @Override
        public String toString() {
            return "TestIdResolver";
        }
    }

    /* ================= Helper: Concrete subclass of TypeDeserializerBase ================= */

    static class ConcreteTypeDeserializer extends TypeDeserializerBase {

        ConcreteTypeDeserializer(JavaType baseType, TypeIdResolver idRes,
                String typePropertyName, boolean typeIdVisible, Class<?> defaultImpl) {
            super(baseType, idRes, typePropertyName, typeIdVisible, defaultImpl);
        }

        ConcreteTypeDeserializer(TypeDeserializerBase src, BeanProperty property) {
            super(src, property);
        }

        @Override
        public TypeDeserializer forProperty(BeanProperty prop) {
            return new ConcreteTypeDeserializer(this, prop);
        }

        @Override
        public JsonTypeInfo.As getTypeInclusion() {
            return JsonTypeInfo.As.PROPERTY;
        }

        @Override
        public Object deserializeTypedFromObject(JsonParser jp, DeserializationContext ctxt) throws IOException {
            return null;
        }

        @Override
        public Object deserializeTypedFromArray(JsonParser jp, DeserializationContext ctxt) throws IOException {
            return null;
        }

        @Override
        public Object deserializeTypedFromScalar(JsonParser jp, DeserializationContext ctxt) throws IOException {
            return null;
        }

        @Override
        public Object deserializeTypedFromAny(JsonParser jp, DeserializationContext ctxt) throws IOException {
            return null;
        }
    }

    /* ================= Tests: Accessors / basic construction ================= */

    @Test
    public void testConstructor_withDefaultImpl_accessorsReturnExpectedValues() throws IOException {
        JavaType baseType = TypeFactory.defaultInstance().constructType(Object.class);
        TestIdResolver resolver = new TestIdResolver(new HashMap<String, JavaType>());
        ConcreteTypeDeserializer deser = new ConcreteTypeDeserializer(baseType, resolver, "@type", true, String.class);

        assertEquals("java.lang.Object", deser.baseTypeName());
        assertEquals("@type", deser.getPropertyName());
        assertSame(resolver, deser.getTypeIdResolver());
        assertEquals(String.class, deser.getDefaultImpl());
        assertEquals(JsonTypeInfo.As.PROPERTY, deser.getTypeInclusion());
    }

    @Test
    public void testConstructor_withoutDefaultImpl_getDefaultImplReturnsNull() {
        JavaType baseType = TypeFactory.defaultInstance().constructType(Object.class);
        TestIdResolver resolver = new TestIdResolver(new HashMap<String, JavaType>());
        ConcreteTypeDeserializer deser = new ConcreteTypeDeserializer(baseType, resolver, "@type", false, null);

        assertNull(deser.getDefaultImpl());
    }

    public void testBaseTypeName_withNullBaseType_throwsNullPointerException() {

            // expected
        }
    }

    @Test
    public void testToString_containsClassNameAndBaseType() {
        JavaType baseType = TypeFactory.defaultInstance().constructType(String.class);
        TestIdResolver resolver = new TestIdResolver(new HashMap<String, JavaType>());
        ConcreteTypeDeserializer deser = new ConcreteTypeDeserializer(baseType, resolver, "@type", false, null);

        String str = deser.toString();
        assertTrue(str.contains(ConcreteTypeDeserializer.class.getName()));
        assertTrue(str.contains("base-type"));
        assertTrue(str.contains("id-resolver"));
    }

    @Test
    public void testForProperty_returnsNewInstance() {
        JavaType baseType = TypeFactory.defaultInstance().constructType(Object.class);
        TestIdResolver resolver = new TestIdResolver(new HashMap<String, JavaType>());
        ConcreteTypeDeserializer deser = new ConcreteTypeDeserializer(baseType, resolver, "@type", false, null);

        TypeDeserializer copy = deser.forProperty(null);
        assertNotNull(copy);
        assertTrue(copy instanceof ConcreteTypeDeserializer);
        assertNotSame(deser, copy);
        assertEquals(deser.getPropertyName(), copy.getPropertyName());
        assertSame(deser.getTypeIdResolver(), copy.getTypeIdResolver());
    }

    /* ================= Tests: _findDeserializer ================= */

    @Test
    public void testFindDeserializer_knownTypeId_returnsContextualDeserializer() throws IOException {
        JavaType baseType = TypeFactory.defaultInstance().constructType(Object.class);
        Map<String, JavaType> mapping = new HashMap<String, JavaType>();
        mapping.put("str", TypeFactory.defaultInstance().constructType(String.class));
        TestIdResolver resolver = new TestIdResolver(mapping);
        ConcreteTypeDeserializer deser = new ConcreteTypeDeserializer(baseType, resolver, "@type", false, null);

        DeserializationContext ctxt = obtainDefaultContext();

        JsonDeserializer<Object> found = deser._findDeserializer(ctxt, "str");
        assertNotNull(found);

        // second call should hit cache but still return non-null
        JsonDeserializer<Object> foundAgain = deser._findDeserializer(ctxt, "str");
        assertNotNull(foundAgain);
    }

    @Test
    public void testFindDeserializer_unknownTypeId_withNoDefaultImpl_failOnInvalidSubtypeEnabled_throwsException() throws IOException {
        JavaType baseType = TypeFactory.defaultInstance().constructType(Object.class);
        TestIdResolver resolver = new TestIdResolver(new HashMap<String, JavaType>());
        ConcreteTypeDeserializer deser = new ConcreteTypeDeserializer(baseType, resolver, "@type", false, null);

        ObjectMapper mapper = new ObjectMapper();
        mapper.enable(DeserializationFeature.FAIL_ON_INVALID_SUBTYPE);
        DeserializationContext ctxt = obtainContext(mapper);

        try {
            deser._findDeserializer(ctxt, "unknown");
            fail("Expected exception for unknown type id");
        } catch (IOException expected) {
            // expected (JsonMappingException subtype)
        }
    }

    @Test
    public void testFindDeserializer_unknownTypeId_withNoDefaultImpl_failOnInvalidSubtypeDisabled_returnsNullifying() throws IOException {
        JavaType baseType = TypeFactory.defaultInstance().constructType(Object.class);
