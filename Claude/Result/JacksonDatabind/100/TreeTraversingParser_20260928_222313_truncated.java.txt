    // ---------- nextToken traversal ----------

    @Test
    public void testNextToken_flatArrayTraversal_returnsCorrectSequence() throws IOException {
        ArrayNode arr = factory.arrayNode();
        arr.add(1);
        arr.add("hello");
        arr.add(true);
        arr.addNull();

        TreeTraversingParser p = new TreeTraversingParser(arr);

        assertEquals(JsonToken.START_ARRAY, p.nextToken());
        assertEquals(JsonToken.VALUE_NUMBER_INT, p.nextToken());
        assertEquals(1, p.getIntValue());
        assertEquals(JsonToken.VALUE_STRING, p.nextToken());
        assertEquals("hello", p.getText());
        assertEquals(JsonToken.VALUE_TRUE, p.nextToken());
        assertEquals(JsonToken.VALUE_NULL, p.nextToken());
        assertEquals(JsonToken.END_ARRAY, p.nextToken());
        assertNull(p.nextToken());
        assertTrue(p.isClosed());
    }

    @Test

    @Test
    public void testNextToken_nestedContainers_coversAllBranches() throws IOException {
        ObjectNode root = factory.objectNode();
        root.putArray("arr1"); // empty nested array -> triggers skip-empty branch
        ArrayNode arr2 = root.putArray("arr2");
        arr2.add(1);
        arr2.add(2);

        TreeTraversingParser p = new TreeTraversingParser(root);

        assertEquals(JsonToken.START_OBJECT, p.nextToken());
        assertEquals(JsonToken.FIELD_NAME, p.nextToken());
        assertEquals("arr1", p.getCurrentName());
        assertEquals(JsonToken.START_ARRAY, p.nextToken());
        assertEquals(JsonToken.END_ARRAY, p.nextToken()); // empty-skip branch

        assertEquals(JsonToken.FIELD_NAME, p.nextToken());
        assertEquals("arr2", p.getCurrentName());
        assertEquals(JsonToken.START_ARRAY, p.nextToken());
        assertEquals(JsonToken.VALUE_NUMBER_INT, p.nextToken()); // descend branch
        assertEquals(1, p.getIntValue());
        assertEquals(JsonToken.VALUE_NUMBER_INT, p.nextToken());
        assertEquals(2, p.getIntValue());
        assertEquals(JsonToken.END_ARRAY, p.nextToken());
        assertEquals(JsonToken.END_OBJECT, p.nextToken());
        assertNull(p.nextToken());
        assertTrue(p.isClosed());
    }
