    @Test
    public void testAppend_mergesTokensAndNativeIdFlags() throws Exception
    {
        TokenBuffer source = new TokenBuffer(null, true);
        source.writeStartArray();
        source.writeNumber(42);
        source.writeEndArray();

        TokenBuffer dest = new TokenBuffer(null, false);
        dest.append(source);

        assertTrue(dest.canWriteTypeId());
        assertTrue(dest.canWriteObjectId());

        JsonParser p = dest.asParser();
        assertEquals(JsonToken.START_ARRAY, p.nextToken());
        assertEquals(JsonToken.VALUE_NUMBER_INT, p.nextToken());
        assertEquals(42, p.getIntValue());
        assertEquals(JsonToken.END_ARRAY, p.nextToken());
        assertNull(p.nextToken());
    }

    @Test
    public void testSerialize_writesFullStructure() throws Exception
    {
        TokenBuffer buf = new TokenBuffer(null, false);
        buf.writeStartObject();
        buf.writeFieldName("field");
        buf.writeString("value");
        buf.writeEndObject();

        JsonFactory factory = new JsonFactory();
        StringWriter writer = new StringWriter();
        JsonGenerator gen = factory.createGenerator(writer);
        buf.serialize(gen);
        gen.flush();
        assertEquals("{\"field\":\"value\"}", writer.toString());
        gen.close();
    }

    @Test
    public void testDeserialize_copiesCurrentStructureFromParser() throws Exception
    {
        JsonFactory factory = new JsonFactory();
        JsonParser jp = factory.createParser("{\"a\":1}");
        jp.nextToken();
        TokenBuffer buf = new TokenBuffer(null, false);
        DeserializationContext ctxt = null;
        TokenBuffer result = buf.deserialize(jp, ctxt);
        assertSame(buf, result);

        JsonParser out = buf.asParser();
        assertEquals(JsonToken.START_OBJECT, out.nextToken());
        assertEquals(JsonToken.FIELD_NAME, out.nextToken());
        assertEquals("a", out.getCurrentName());
        assertEquals(JsonToken.VALUE_NUMBER_INT, out.nextToken());
        assertEquals(1, out.getIntValue());
        assertEquals(JsonToken.END_OBJECT, out.nextToken());
        jp.close();
    }

    @Test
    public void testWriteNumber_allPrimitiveAndObjectVariants() throws Exception
    {
        TokenBuffer buf = new TokenBuffer(null, false);
        buf.writeNumber((short) 5);
        buf.writeNumber(10);
        buf.writeNumber(100L);
        buf.writeNumber(3.14d);
        buf.writeNumber(2.5f);
        buf.writeNumber(BigDecimal.valueOf(1.23));
        buf.writeNumber(BigInteger.valueOf(999));
        buf.writeNumber("123.456");

        JsonParser p = buf.asParser();

        p.nextToken();
        assertEquals(JsonToken.VALUE_NUMBER_INT, p.getCurrentToken());
        assertEquals(5, p.getIntValue());

        p.nextToken();
        assertEquals(10, p.getIntValue());

        p.nextToken();
        assertEquals(100L, p.getLongValue());

        p.nextToken();
        assertEquals(3.14d, p.getDoubleValue(), 0.0001);

        p.nextToken();
        assertEquals(2.5f
