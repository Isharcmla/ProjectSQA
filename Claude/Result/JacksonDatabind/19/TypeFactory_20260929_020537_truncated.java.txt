    // ---------------------------------------------------------
    // defaultInstance / clearCache / withModifier
    // ---------------------------------------------------------

    @Test
    public void testDefaultInstance_returnsSameSingleton() {
        TypeFactory a = TypeFactory.defaultInstance();
        TypeFactory b = TypeFactory.defaultInstance();
        assertNotNull(a);
        assertSame(a, b);
    }

    @Test
    public void testClearCache_doesNotThrow() {
        factory.constructType(StringList.class);
        factory.clearCache();
        JavaType t = factory.constructType(StringList.class);
        assertNotNull(t);
    }

    @Test
    public void testWithModifier_null_returnsNewFactoryInstance() {
        // NOTE: TypeModifier API details are not provided in source_code,
        // so only the null-branch (which does not require instantiating TypeModifier) is tested.
        TypeFactory newFactory = factory.withModifier(null);
        assertNotNull(newFactory);
        assertNotSame(factory, newFactory);
    }

    // ---------------------------------------------------------
    // unknownType / rawClass (static)
    // ---------------------------------------------------------

    @Test
    public void testUnknownType_returnsObjectType() {
        JavaType t = TypeFactory.unknownType();
        assertNotNull(t);
        assertEquals(Object.class, t.getRawClass());
    }

    @Test
    public void testRawClass_classInput_returnsSameClass() {
        Class<?> c = TypeFactory.rawClass(String.class);
        assertEquals(String.class, c);
    }

    @Test
    public void testRawClass_parameterizedType_returnsRawClass() throws Exception {
        Field f = Holder.class.getDeclaredField("listField");
        Type genType = f.getGenericType();
        Class<?> c = TypeFactory.rawClass(genType);
        assertEquals(List.class, c);
    }

    // ---------------------------------------------------------
    // constructSpecializedType
    // ---------------------------------------------------------

    @Test
    public void testConstructSpecializedType_sameRawClass_returnsSameInstance() {
        JavaType base = factory.constructType(String.class);
        JavaType result = factory.constructSpecializedType(base, String.class);
        assertSame(base, result);
    }

    @Test
    public void testConstructSpecializedType_simpleTypeCollectionSubclass_specializes() {
        JavaType base = factory.uncheckedSimpleType(Collection.class);
        JavaType result = factory.constructSpecializedType(base, ArrayList.class);
        assertNotNull(result);
        assertEquals(ArrayList.class, result.getRawClass());
    }

    @Test(expected = IllegalArgumentException.class)
    public void testConstructSpecializedType_incompatibleSubclass_throwsException() {
        JavaType base = factory.uncheckedSimpleType(String.class);
        factory.constructSpecializedType(base, ArrayList.class);
    }

    @Test
    public void testConstructSpecializedType_narrowByRegularClass_works() {
        JavaType base = factory.constructType(Number.class);
        JavaType result = factory.constructSpecializedType(base, Integer.class);
        assertNotNull(result);
        assertEquals(Integer.class, result.getRawClass());
    }

    @Test
    public void testFindTypeParameters_javaType_returnsCorrectParams() {
        JavaType type = factory.constructType(StringList.class);
        JavaType[] params = factory.findTypeParameters(type, Collection.class);
        assertNotNull(params);
        assertEquals(1, params.length);
        assertEquals(String.class, params[0].getRawClass());
    }

    @Test
    public void testFindTypeParameters_nonGenericSuperType_returnsNull() {

    public void testFindTypeParameters_mapSubtype_returnsTwoParams() {

    @Test
    public void testConstructType_genericArrayType_returnsArrayType() throws Exception {
        Field f = ArrayHolder.class.getDeclaredField("arrField");
        Type genType = f.getGenericType();
        assertTrue(genType instanceof GenericArrayType);
        JavaType t = factory.constructType(genType);
        assertTrue(t.isArrayType());
    }

    @Test
    public void testConstructType_typeVariable_resolvedViaBounds() throws Exception {
        Field f = NumHolder.class.getDeclaredField("value");
        Type genType = f.getGenericType();
        assertTrue(genType instanceof TypeVariable<?>);
        JavaType t = factory.constructType(genType);
        assertEquals(Number.class, t.getRawClass());
    }
