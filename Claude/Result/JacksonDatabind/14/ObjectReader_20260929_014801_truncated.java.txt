    @Test
    public void testReadValues_jsonParser_class_iteratesValues() throws IOException {
        JsonParser p = mapper.getFactory().createParser("[{\"name\":\"a\",\"age\":1},{\"name\":\"b\",\"age\":2}]");

        p.close();
