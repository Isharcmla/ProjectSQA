    @Test
    public void testFormat_date_normal() {
        FastDateFormat fdf = FastDateFormat.getInstance("yyyy-MM-dd", TimeZone.getTimeZone("GMT"), Locale.US);
        Calendar cal = new GregorianCalendar(TimeZone.getTimeZone("GMT"));
        cal.set(2020, Calendar.JANUARY, 15, 10, 30, 0);
        cal.set(Calendar.MILLISECOND, 0);
        Date date = cal.getTime();
        String result = fdf.format(date);
        assertEquals("2020-01-15", result);
    }

    public void testFormat_calendar_normal() {

    @Test
    public void testFormat_dateWithBuffer() {
        FastDateFormat fdf = FastDateFormat.getInstance("yyyy", TimeZone.getTimeZone("GMT"), Locale.US);
        Calendar cal = new GregorianCalendar(TimeZone.getTimeZone("GMT"));
        cal.set(2019, Calendar.JANUARY, 1, 0, 0, 0);
        cal.set(Calendar.MILLISECOND, 0);
        StringBuffer buf = new StringBuffer();
        StringBuffer result = fdf.format(cal.getTime(), buf);
        assertEquals("2019", result.toString());
        assertSame(buf, result);
    }

    @Test
    public void testFormat_calendarWithBuffer_notForced() {
        FastDateFormat fdf = FastDateFormat.getInstance("yyyy", null, Locale.US);

        cal.set(2018, Calendar.JANUARY, 1, 0, 0, 0);
        cal.set(Calendar.MILLISECOND, 0);
        StringBuffer buf = new StringBuffer();
        StringBuffer result = fdf.format(cal, buf);
        assertEquals("2018", result.toString());
    }

    @Test
    public void testFormat_calendarWithBuffer_forcedTimeZone() {
        TimeZone forcedTz = TimeZone.getTimeZone("GMT");
        FastDateFormat fdf = FastDateFormat.getInstance("yyyy-MM-dd", forcedTz, Locale.US);
        Calendar cal = new GregorianCalendar(TimeZone.getTimeZone("America/New_York"));
        cal.set(2020, Calendar.JULY, 4, 12, 0, 0);
        cal.set(Calendar.MILLISECOND, 0);
        StringBuffer buf = new StringBuffer();
        StringBuffer result = fdf.format(cal, buf);
        assertNotNull(result);
        // The original calendar's time zone must not be mutated.
        assertEquals("America/New_York", cal.getTimeZone().getID());
    }
