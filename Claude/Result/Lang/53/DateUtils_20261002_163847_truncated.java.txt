public class DateUtilsTest {

    private Date buildDate(int year, int month, int day, int hour, int minute, int second, int millis) {
        Calendar cal = Calendar.getInstance();
        cal.clear();
        cal.set(year, month - 1, day, hour, minute, second);
        cal.set(Calendar.MILLISECOND, millis);
        return cal.getTime();
    }

    private Calendar toCalendar(Date date) {
        Calendar cal = Calendar.getInstance();
        cal.setTime(date);
        return cal;
    }

    // ---------------------- isSameDay(Date, Date) ----------------------

    @Test
    public void testIsSameDayDate_sameDay_returnsTrue() {
        Date d1 = buildDate(2002, 3, 28, 13, 45, 0, 0);
        Date d2 = buildDate(2002, 3, 28, 6, 1, 0, 0);
        assertTrue(DateUtils.isSameDay(d1, d2));
    }

    @Test
    public void testIsSameDayDate_differentDay_returnsFalse() {
        Date d1 = buildDate(2002, 3, 28, 13, 45, 0, 0);
        Date d2 = buildDate(2002, 3, 12, 13, 45, 0, 0);
        assertFalse(DateUtils.isSameDay(d1, d2));
    }

    @Test
    public void testIsSameLocalTime_differentClass_returnsFalse() {
        Calendar c1 = new GregorianCalendar(2002, 2, 28, 13, 45, 1);
        c1.set(Calendar.MILLISECOND, 100);
        Calendar c2 = new java.util.Calendar() {
            public void add(int field, int amount) {}
            public void roll(int field, boolean up) {}
            protected void computeTime() {}
            protected void computeFields() {}
            public int getMinimum(int field) { return 0; }
            public int getMaximum(int field) { return 0; }
            public int getGreatestMinimum(int field) { return 0; }
            public int getLeastMaximum(int field) { return 0; }
        };
        c2.setTimeInMillis(c1.getTimeInMillis());
        assertFalse(DateUtils.isSameLocalTime(c1, c2));
    }
