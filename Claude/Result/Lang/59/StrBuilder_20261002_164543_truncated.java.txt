    @Test
    public void testConstructor_withZeroOrNegativeCapacity_defaultsTo32() {
        StrBuilder b1 = new StrBuilder(0);
        assertEquals(32, b1.capacity());
        StrBuilder b2 = new StrBuilder(-5);
        assertEquals(32, b2.capacity());
    }

    @Test
    public void testConstructor_withString_copiesContent() {
        StrBuilder b = new StrBuilder("hello");
        assertEquals("hello", b.toString());

    }

    @Test
    public void testConstructor_withNullString_empty() {
        StrBuilder b = new StrBuilder((String) null);
