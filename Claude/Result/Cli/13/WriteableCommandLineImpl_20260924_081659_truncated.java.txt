/**
 * JUnit4 test suite for {@link org.apache.commons.cli2.commandline.WriteableCommandLineImpl}.

 * NOTE: Since Option/Argument are interfaces coming from the commons-cli2
 * dependency (no mocking framework allowed), simple concrete test
 * implementations are provided below that implement only the methods
 * required by the Option / Argument contracts as used by

    /**
     * Minimal concrete implementation of the Option interface, sufficient to
     * exercise WriteableCommandLineImpl's public API.
     */
    private static class TestOption implements Option {
        private final String preferredName;
        private final List triggers;
        private final Set prefixes;

        TestOption(String preferredName) {
            this(preferredName, new ArrayList());
        }

        TestOption(String preferredName, List triggers) {
            this.preferredName = preferredName;
            this.triggers = triggers;
            Set p = new HashSet();
            p.add("-");
            p.add("--");
            this.prefixes = p;
        }

        public int getId() {
            return 0;
        }

        public boolean canProcess(WriteableCommandLine commandLine, String arg) {
            return false;
        }

        public Set getPrefixes() {
            return prefixes;
        }

        public List getTriggers() {
            return triggers;
        }

        public void process(WriteableCommandLine commandLine, ListIterator arguments) {
            // no-op for testing purposes
        }

        public void validate(WriteableCommandLine commandLine) {
            // no-op for testing purposes
        }

        public void appendUsage(StringBuffer buffer, Set helpSettings, Comparator comp) {
            // no-op for testing purposes
        }

        public void defaults(WriteableCommandLine commandLine) {
            // no-op for testing purposes
        }

        public String getPreferredName() {
            return preferredName;
        }

        public String getDescription() {
            return "test option";
        }

        public List helpLines(int depth, Set helpSettings, Comparator comp) {
            return Collections.EMPTY_LIST;
        }

        public boolean isRequired() {
            return false;
        }
    }

    /**
     * Minimal concrete implementation of the Argument interface (which
     * extends Option) sufficient to exercise the "instanceof Argument"
     * branch inside addValue().
     */
    private static class TestArgument extends TestOption implements Argument {
        TestArgument(String preferredName) {
            super(preferredName);
        }

        TestArgument(String preferredName, List triggers) {
            super(preferredName, triggers);
        }

        public void validate(WriteableCommandLine commandLine, Option option) {
            // no-op for testing purposes
        }

        public String getInitialSeparator() {
            return "=";
        }

        public String getSubsequentSeparator() {
            return ",";
        }
    }

    private Option rootOption;
    private List normalisedArgs;

    @Before
    public void setUp() {
        rootOption = new TestOption("root");
        normalisedArgs = new ArrayList();
        normalisedArgs.add("arg1");
        normalisedArgs.add("arg2");
    }

    private WriteableCommandLineImpl createCommandLine() {
        return new WriteableCommandLineImpl(rootOption, normalisedArgs);
    }

    @Test
    public void testAddOption_normal_addsOptionAndTriggers() {
        WriteableCommandLineImpl cmdLine = createCommandLine();
        List triggers = new ArrayList();
        triggers.add("t");
        triggers.add("test");
        Option option = new TestOption("test", triggers);

        cmdLine.addOption(option);

        assertTrue(cmdLine.hasOption(option));
        assertSame(option, cmdLine.getOption("t"));
        assertSame(option, cmdLine.getOption("test"));
    }

    @Test
    public void testAddValue_withNonArgumentOption_doesNotAutoAddOption() {
        WriteableCommandLineImpl cmdLine = createCommandLine();
        Option option = new TestOption("plain");

        cmdLine.addValue(option, "value1");

        // Non-Argument options are not auto-added via addValue
        assertFalse(cmdLine.hasOption(option));
        List values = cmdLine.getValues(option, null);
        assertEquals(1, values.size());
        assertEquals("value1", values.get(0));
    }

    @Test
    public void testAddValue_withArgumentOption_autoAddsOption() {
        WriteableCommandLineImpl cmdLine = createCommandLine();
        Argument argument = new TestArgument("arg");

        cmdLine.addValue(argument, "value1");

        assertTrue(cmdLine.hasOption(argument));
    }
