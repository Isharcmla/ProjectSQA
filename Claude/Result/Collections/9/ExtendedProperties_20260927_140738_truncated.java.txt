public class ExtendedPropertiesTest {

    private File tempDir;

    @Before
    public void setUp() throws IOException {
        File tmp = File.createTempFile("edp", "");
        tmp.delete();
        tmp.mkdir();
        tempDir = tmp;
    }

    @After
    public void tearDown() {
        if (tempDir != null && tempDir.exists()) {
            File[] files = tempDir.listFiles();
            if (files != null) {
                for (File f : files) {
                    f.delete();
                }
            }
            tempDir.delete();
        }
    }

    private File writeFile(String name, String content) throws IOException {
        File f = new File(tempDir, name);
        FileWriter fw = new FileWriter(f);
        try {
            fw.write(content);
        } finally {
            fw.close();
        }
        return f;
    }

    @Test
    public void testDefaultConstructor_createsEmptyProperties() {
        ExtendedProperties props = new ExtendedProperties();

    @Test
    public void testConstructorWithFile_loadsProperties() throws IOException {
        File f = writeFile("main.properties", "key1=value1\n");
        ExtendedProperties props = new ExtendedProperties(f.getAbsolutePath());
        assertEquals("value1", props.getString("key1"));
        assertTrue(props.isInitialized());

        ExtendedProperties props = new ExtendedProperties(mainFile.getAbsolutePath(), defFile.getAbsolutePath());
        assertEquals("mainValue", props.getString("mainKey"));
        assertEquals("defValue", props.getString("defKey"));
    }

    @Test
    public void testLoad_withInputStream_parsesProperties() throws IOException {
        ExtendedProperties props = new ExtendedProperties();
        String content = "# comment\n\nkey1=value1\nkey2=value2\n";
        InputStream is = new ByteArrayInputStream(content.getBytes());
        props.load(is);
        assertEquals("value1", props.getString("key1"));
        assertEquals("value2", props.getString("key2"));
    }

    @Test
    public void testLoad_withEncoding_parsesProperties() throws IOException {
        ExtendedProperties props = new ExtendedProperties();
        String content = "keyEnc=valueEnc\n";
        InputStream is = new ByteArrayInputStream(content.getBytes());
        props.load(is, "UTF-8");
        assertEquals("valueEnc", props.getString("keyEnc"));
    }

    @Test
    public void testLoad_withInvalidEncoding_fallsBackToDefault() throws IOException {
        ExtendedProperties props = new ExtendedProperties();
        String content = "keyBad=valueBad\n";
        InputStream is = new ByteArrayInputStream(content.getBytes());
        props.load(is, "NOT_A_REAL_ENCODING");

    @Test
    public void testLoad_lineWithoutEquals_isIgnored() throws IOException {
        ExtendedProperties props = new ExtendedProperties();

        props.load(is);
        assertEquals("realValue", props.getString("realKey"));

    @Test

        props.load(is);

    @Test
    public void testLoad_multiLineContinuation_concatenatesValue() throws IOException {
        ExtendedProperties props = new ExtendedProperties();
        String content = "longvalue = aaa\\\nbbb\n";
        InputStream is = new ByteArrayInputStream(content.getBytes());
        props.load(is);
        String val = props.getString("longvalue");
        assertEquals("aaabbb", val);
    }

    @Test
    public void testLoad_tokensOnALine_createsMultipleValues() throws IOException {
        ExtendedProperties props = new ExtendedProperties();
        String content = "tokens = first token, second token\n";
        InputStream is = new ByteArrayInputStream(content.getBytes());
        props.load(is);
        String[] arr = props.getStringArray("tokens");
        assertEquals(2, arr.length);
        assertEquals("first token", arr[0]);
        assertEquals("second token", arr[1]);
    }

    @Test
    public void testLoad_escapedComma_isUnescaped() throws IOException {
        ExtendedProperties props = new ExtendedProperties();
        String content = "commas.escaped = Hi\\, what'up?\n";
        InputStream is = new ByteArrayInputStream(content.getBytes());
        props.load(is);
        assertEquals("Hi, what'up?", props.getString("commas.escaped"));
    }

    }

    @Test
    public void testSave_withOutputStream_writesProperties() throws IOException {
        ExtendedProperties props = new ExtendedProperties();
        props.addProperty("k1", "v1");
        props.addProperty("k2", "v2a");
        props.addProperty("k2", "v2b");
        ByteArrayOutputStream baos = new ByteArrayOutputStream();
        props.save(baos, "header comment");
        String output = baos.toString();
        assertTrue(output.contains("header comment"));
        assertTrue(output.contains("k1=v1"));
        assertTrue(output.contains("k2=v2a"));
        assertTrue(output.contains("k2=v2b"));
    }

    @Test
    public void testGetKeysWithPrefix_returnsMatchingKeys() {
        ExtendedProperties props = new ExtendedProperties();
        props.addProperty("prefix.a", "v1");
        props.addProperty("prefix.b", "v2");
        props.addProperty("other.c", "v3");
        Iterator it = props.getKeys("prefix");
        int count = 0;
        while (it.hasNext()) {
            String key = (String) it.next();
            assertTrue(key.startsWith("prefix"));
            count++;
        }
        assertEquals(2, count);
    }
