public class Flat3MapTest {

    //-----------------------------------------------------------------------
    // Constructors
    //-----------------------------------------------------------------------

    @Test
    public void testDefaultConstructor_isEmpty() {
        Flat3Map map = new Flat3Map();
        assertTrue(map.isEmpty());
        assertEquals(0, map.size());
    }

    @Test
    public void testConstructorWithMap_copiesElements() {
        Map source = new HashMap();
        source.put("a", "1");
        source.put("b", "2");
        Flat3Map map = new Flat3Map(source);
        assertEquals(2, map.size());
        assertEquals("1", map.get("a"));
        assertEquals("2", map.get("b"));
    }

    @Test(expected = NullPointerException.class)
    public void testConstructorWithMap_null_throwsNPE() {
        new Flat3Map(null);
    }

    //-----------------------------------------------------------------------
    // put
    //-----------------------------------------------------------------------

    @Test
    public void testPut_newKey_returnsNull() {
        Flat3Map map = new Flat3Map();
        assertNull(map.put("a", "1"));
        assertEquals(1, map.size());
    }

    @Test
    public void testPut_existingKey_returnsOldValue() {
        Flat3Map map = new Flat3Map();
        map.put("a", "1");
        Object old = map.put("a", "2");
        assertEquals("1", old);
        assertEquals("2", map.get("a"));
        assertEquals(1, map.size());
    }

    @Test
    public void testPut_nullKey() {
        Flat3Map map = new Flat3Map();
        map.put(null, "v1");
        assertEquals("v1", map.get(null));
        Object old = map.put(null, "v2");
        assertEquals("v1", old);
        assertEquals("v2", map.get(null));
    }

    @Test
    public void testPut_threeEntries_flatMode() {
        Flat3Map map = new Flat3Map();
        map.put("a", "1");
        map.put("b", "2");
        map.put("c", "3");
        assertEquals(3, map.size());
        assertEquals("1", map.get("a"));
        assertEquals("2", map.get("b"));
        assertEquals("3", map.get("c"));
    }

    @Test
    public void testPut_growsToDelegateMode() {
        Flat3Map map = new Flat3Map();
        map.put("a", "1");
        map.put("b", "2");
        map.put("c", "3");
        map.put("d", "4");
        assertEquals(4, map.size());
        assertEquals("1", map.get("a"));
        assertEquals("4", map.get("d"));
    }

    @Test
    public void testPut_delegateMode_updateExisting() {
        Flat3Map map = new Flat3Map();
        map.put("a", "1");
        map.put("b", "2");
        map.put("c", "3");
        map.put("d", "4");
        Object old = map.put("a", "100");
        assertEquals("1", old);
        assertEquals("100", map.get("a"));
    }

        Flat3Map map = new Flat3Map();

        assertEquals(0, map.size());
