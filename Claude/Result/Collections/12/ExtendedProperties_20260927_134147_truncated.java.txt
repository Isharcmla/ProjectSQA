package org.apache.commons.collections;

import static org.junit.Assert.*;

import java.io.ByteArrayInputStream;

@SuppressWarnings({ "unchecked", "rawtypes" })
public class ExtendedPropertiesTest {

    private ExtendedProperties props;

    @Before
    public void setUp() {
        props = new ExtendedProperties();
    }

    private File createTempPropertiesFile(String content) throws IOException {
        File f = File.createTempFile("extprop", ".properties");
        f.deleteOnExit();
        PrintWriter pw = new PrintWriter(new FileOutputStream(f));
        pw.print(content);
        pw.close();
        return f;
    }

    // ---------------- Constructors ----------------

    @Test
    public void testDefaultConstructor_createsEmptyProperties() {
        ExtendedProperties p = new ExtendedProperties();
        assertFalse(p.isInitialized());
        assertNull(p.getString("anyKey"));
    }

    @Test
    public void testFileConstructor_loadsPropertiesFromFile() throws IOException {
        File f = createTempPropertiesFile("key=value\n");
        ExtendedProperties p = new ExtendedProperties(f.getAbsolutePath());
        assertEquals("value", p.getString("key"));
        assertTrue(p.isInitialized());
    }

    @Test
    public void testFileConstructor_withDefaultFile_loadsDefaultsAndMain() throws IOException {
        File defaultFile = createTempPropertiesFile("defaultKey=defaultValue\n");
        File mainFile = createTempPropertiesFile("mainKey=mainValue\n");
        ExtendedProperties p = new ExtendedProperties(mainFile.getAbsolutePath(), defaultFile.getAbsolutePath());
        assertEquals("mainValue", p.getString("mainKey"));
        assertEquals("defaultValue", p.getString("defaultKey"));
    }

    @Test(expected = IOException.class)
    public void testFileConstructor_fileNotFound_throwsIOException() throws IOException {
        new ExtendedProperties("/this/path/does/not/exist/file.properties");
    }

    @Test
    public void testFileConstructor_withInclude_loadsIncludedFile() throws IOException {
        File includedFile = createTempPropertiesFile("includedKey=includedValue\n");
        String includedFileName = includedFile.getName();
        File mainFile = createTempPropertiesFile("include=" + includedFileName + "\nmainKey=mainValue\n");
        ExtendedProperties p = new ExtendedProperties(mainFile.getAbsolutePath());
        assertEquals("includedValue", p.getString("includedKey"));
        assertEquals("mainValue", p.getString("mainKey"));
    }

    // ---------------- isInitialized ----------------

    @Test
    public void testIsInitialized_beforeAnyAction_returnsFalse() {
        assertFalse(props.isInitialized());
    }

    @Test
    public void testIsInitialized_afterAddProperty_returnsTrue() {
        props.addProperty("key", "value");
        assertTrue(props.isInitialized());
    }

    // ---------------- getInclude / setInclude ----------------

    @Test
    public void testGetInclude_default_returnsInclude() {
        assertEquals("include", props.getInclude());
    }

    @Test
    public void testSetInclude_customName_getIncludeReturnsCustomName() {
        props.setInclude("myInclude");
        assertEquals("myInclude", props.getInclude());
    }

    @Test
    public void testSetInclude_emptyString_getIncludeReturnsNull() {
        props.setInclude("");
        assertNull(props.getInclude());
    }

    @Test
    public void testSetInclude_null_getIncludeReturnsNull() {
        props.setInclude(null);
        assertNull(props.getInclude());
    }

    // ---------------- load ----------------

    @Test
    public void testLoad_viaInputStream_loadsProperties() throws IOException {
        String content = "key1=value1\nkey2=value2\n";
        props.load(new ByteArrayInputStream(content.getBytes()));
        assertEquals("value1", props.getString("key1"));
        assertEquals("value2", props.getString("key2"));
        assertTrue(props.isInitialized());
    }

    @Test
    public void testLoad_withCommentsAndBlankLines_skipsThem() throws IOException {
        String content = "# a comment\n\nkey=value\n";
        props.load(new ByteArrayInputStream(content.getBytes()));
        assertEquals("value", props.getString("key"));
    }

    @Test
    public void testLoad_withLineContinuation_concatenatesLines() throws IOException {
        String content = "longvalue = aaa\\\nbbb\n";
        props.load(new ByteArrayInputStream(content.getBytes()));
        assertEquals("aaabbb", props.getString("longvalue"));
    }

    @Test
    public void testLoad_withValidEncoding_loadsProperties() throws IOException {
        String content = "key=value\n";
        props.load(new ByteArrayInputStream(content.getBytes("UTF-8")), "UTF-8");
        assertEquals("value", props.getString("key"));
    }

    @Test
    public void testLoad_withUnsupportedEncoding_fallsBackAndLoads() throws IOException {
        String content = "key=value\n";
        props.load(new ByteArrayInputStream(content.getBytes()), "UnsupportedEncodingXYZ");
        assertEquals("value", props.getString("key"));
    }

    // ---------------- getProperty ----------------

    @Test
    public void testGetProperty_existingKey_returnsValue() {
        props.addProperty("key", "value");
        assertEquals("value", props.getProperty("key"));
    }

    @Test
    public void testGetProperty_nonExistingKeyNoDefaults_returnsNull() {
        assertNull(props.getProperty("missing"));
    }

    // ---------------- addProperty ----------------

    @Test
    public void testAddProperty_singleValue_addsCorrectly() {
        props.addProperty("key", "value");
        assertEquals("value", props.getString("key"));
    }

    @Test
    public void testAddProperty_multipleValuesSameKey_createsVector() {
        props.addProperty("key", "value1");
        props.addProperty("key", "value2");
        Vector v = props.getVector("key");
        assertEquals(2, v.size());
        assertTrue(v.contains("value1"));
        assertTrue(v.contains("value2"));
    }

    @Test
    public void testAddProperty_commaSeparatedValue_splitsTokens() {
        props.addProperty("key", "token1,token2");
        Vector v = props.getVector("key");
        assertEquals(2, v.size());
        assertEquals("token1", v.get(0));
        assertEquals("token2", v.get(1));
    }

    @Test
    public void testAddProperty_escapedComma_unescapesCorrectly() {
        props.addProperty("key", "Hi\\, what'up?");
        assertEquals("Hi, what'up?", props.getString("key"));
    }

    // ---------------- setProperty ----------------

    @Test
    public void testSetProperty_overwritesExistingValue() {
        props.addProperty("key", "value1");
        props.addProperty("key", "value2");
        props.setProperty("key", "newValue");
        assertEquals("newValue", props.getString("key"));

    // ---------------- save ----------------

    @Test
    public void testSave_withStringValue_writesCorrectFormat() throws IOException {
        props.setProperty("key1", "value1");
        ByteArrayOutputStream baos = new ByteArrayOutputStream();
        props.save(baos, "Header comment");
        String output = baos.toString();
        assertTrue(output.contains("Header comment"));
        assertTrue(output.contains("key1=value1"));
    }

    @Test
    public void testSave_withListValue_writesMultipleLines() throws IOException {
        props.addProperty("listKey", "a");
        props.addProperty("listKey", "b");
        ByteArrayOutputStream baos = new ByteArrayOutputStream();
        props.save(baos, null);
        String output = baos.toString();
        assertTrue(output.contains("listKey=a"));
        assertTrue(output.contains("listKey=b"));
    }

    @Test
    public void testSave_nullOutputStream_doesNothing() throws IOException {
        props.setProperty("key", "value");
        props.save(null, "header");
        // no exception expected
        assertTrue(true);
    }

    // ---------------- combine ----------------

    @Test
    public void testCombine_mergesProperties() {
        ExtendedProperties p1 = new ExtendedProperties();
        p1.setProperty("a", "1");
        ExtendedProperties p2 = new ExtendedProperties();
        p2.setProperty("b", "2");
        p1.combine(p2);
        assertEquals("2", p1.getString("b"));
    }

    // ---------------- clearProperty ----------------

    @Test
    public void testClearProperty_existingKey_removesKey() {
        props.setProperty("toRemove", "val");
        props.clearProperty("toRemove");
        assertNull(props.get("toRemove"));
    }

    @Test
    public void testClearProperty_nonExistingKey_doesNothing() {
        props.clearProperty("nonexistent");
        assertTrue(true);
    }

    // ---------------- getKeys ----------------

    @Test
    public void testGetKeys_returnsAllKeysInOrder() {
        props.setProperty("a", "1");
        props.setProperty("b", "2");
        Iterator it = props.getKeys();
        List keys = new ArrayList();
        while (it.hasNext()) {
            keys.add(it.next());
        }
        assertEquals(Arrays.asList("a", "b"), keys);
    }

    @Test
    public void testGetKeysWithPrefix_returnsMatchingKeys() {
        props.setProperty("prefix.a", "1");
        props.setProperty("prefix.b", "2");
        props.setProperty("other", "3");
        Iterator it = props.getKeys("prefix");
        List matched = new ArrayList();
        while (it.hasNext()) {
            matched.add(it.next());
        }
        assertTrue(matched.contains("prefix.a"));
        assertTrue(matched.contains("prefix.b"));
        assertFalse(matched.contains("other"));
    }

    // ---------------- subset ----------------

    @Test
    public void testSubset_withMatchingPrefix_returnsSubset() {
        props.setProperty("db.user", "admin");
        props.setProperty("db.pass", "secret");
        ExtendedProperties sub = props.subset("db");
        assertNotNull(sub);
        assertEquals("admin", sub.getString("user"));
        assertEquals("secret", sub.getString("pass"));
    }

    @Test
    public void testSubset_noMatchingPrefix_returnsNull() {
        props.setProperty("other", "value");
        assertNull(props.subset("nomatch"));
    }

    @Test
    public void testSubset_exactPrefixMatch_returnsKeyEqualToPrefix() {
        props.setProperty("db", "value");
        ExtendedProperties sub = props.subset("db");
        assertNotNull(sub);
        assertEquals("value", sub.getString("db"));
    }

    public void testGetString_missingKeyNoDefaultValue_returnsNull() {
