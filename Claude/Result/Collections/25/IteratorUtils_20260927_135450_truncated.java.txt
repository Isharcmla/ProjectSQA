    // Helper: simple Predicate implementation
    private static final Predicate<Object> NOT_NULL_PREDICATE = new Predicate<Object>() {
        @Override
        public boolean evaluate(Object object) {
            return object != null;
        }
    };

    private static final Predicate<Integer> EVEN_PREDICATE = new Predicate<Integer>() {
        @Override
        public boolean evaluate(Integer object) {
            return object % 2 == 0;
        }
    };

    private static final Transformer<Object, String> TO_STRING_TRANSFORMER = new Transformer<Object, String>() {
        @Override
        public String transform(Object input) {
            return String.valueOf(input);
        }
    };

    // Custom class with public iterator() method, not Iterable
    public static class CustomIterableLike {
        private final List<String> data;

        public CustomIterableLike(List<String> data) {
            this.data = data;
        }

        public Iterator<String> iterator() {
            return data.iterator();
        }
    }

    private Document createDocument() throws Exception {
        DocumentBuilderFactory factory = DocumentBuilderFactory.newInstance();
        DocumentBuilder builder = factory.newDocumentBuilder();
        Document doc = builder.newDocument();
        Element root = doc.createElement("root");
        doc.appendChild(root);
        Element child1 = doc.createElement("child1");
        Element child2 = doc.createElement("child2");
        root.appendChild(child1);
        root.appendChild(child2);
        return doc;
    }

    public void testObjectGraphIterator_simpleTransformer_flattensGraph() {
        List<Object> root = new ArrayList<Object>();
        root.add("a");
        root.add(java.util.Arrays.asList("b", "c").iterator());
        Transformer<Object, Object> transformer = new Transformer<Object, Object>() {
            @Override
            public Object transform(Object input) {
                if (input instanceof List) {
                    return ((List<?>) input).iterator();
                }
                return input;
            }
        };
        Iterator<Object> it = IteratorUtils.objectGraphIterator(root, transformer);
        List<Object> result = IteratorUtils.toList(it);
        Assert.assertTrue(result.contains("a"));
    }
