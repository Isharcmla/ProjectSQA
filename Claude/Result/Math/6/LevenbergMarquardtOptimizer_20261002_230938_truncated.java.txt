public class LevenbergMarquardtOptimizerTest {

    // Simple linear model: f(x0,x1) = { x0 + x1, x0 - x1 }
    private final MultivariateVectorFunction linearModel = new MultivariateVectorFunction() {
        public double[] value(double[] point) {
            return new double[] { point[0] + point[1], point[0] - point[1] };
        }
    };

    private final MultivariateMatrixFunction linearJacobian = new MultivariateMatrixFunction() {
        public double[][] value(double[] point) {
            return new double[][] { { 1, 1 }, { 1, -1 } };
        }
    };

    // Exponential model: f(a,b;x) = a * exp(b*x) evaluated at x = 0,1,2,3
    private static final double[] XS = { 0, 1, 2, 3 };

    private final MultivariateVectorFunction expModel = new MultivariateVectorFunction() {
        public double[] value(double[] p) {
            double a = p[0];
            double b = p[1];
            double[] res = new double[XS.length];
            for (int i = 0; i < XS.length; i++) {
                res[i] = a * Math.exp(b * XS[i]);
            }
            return res;
        }
    };

    private final MultivariateMatrixFunction expJacobian = new MultivariateMatrixFunction() {
        public double[][] value(double[] p) {
            double a = p[0];
            double b = p[1];
            double[][] jac = new double[XS.length][2];
            for (int i = 0; i < XS.length; i++) {
                jac[i][0] = Math.exp(b * XS[i]);
                jac[i][1] = a * XS[i] * Math.exp(b * XS[i]);
            }
            return jac;
        }
    };

    @Test
    public void testOptimize_linearSystem_returnsCorrectSolution() {
        LevenbergMarquardtOptimizer optimizer = new LevenbergMarquardtOptimizer();

        PointVectorValuePair result = optimizer.optimize(
            new MaxEval(1000),
            new Target(new double[] { 3, 1 }),
            new Weight(new double[] { 1, 1 }),
            new ModelFunction(linearModel),
            new ModelFunctionJacobian(linearJacobian),
            new InitialGuess(new double[] { 0, 0 }));

        double[] point = result.getPoint();
        Assert.assertEquals(2.0, point[0], 1e-6);
        Assert.assertEquals(1.0, point[1], 1e-6);
    }

    @Test
    public void testOptimize_overDeterminedSystem_returnsCorrectSolution() {
        // nR (3) > nC (1)
        MultivariateVectorFunction model = new MultivariateVectorFunction() {
            public double[] value(double[] point) {
                return new double[] { point[0], point[0], point[0] };
            }
        };
        MultivariateMatrixFunction jacobian = new MultivariateMatrixFunction() {
            public double[][] value(double[] point) {
                return new double[][] { { 1 }, { 1 }, { 1 } };
            }
        };

        LevenbergMarquardtOptimizer optimizer = new LevenbergMarquardtOptimizer();
        PointVectorValuePair result = optimizer.optimize(
            new MaxEval(1000),
            new Target(new double[] { 2, 2, 2 }),
            new Weight(new double[] { 1, 1, 1 }),
            new ModelFunction(model),
            new ModelFunctionJacobian(jacobian),
            new InitialGuess(new double[] { 0 }));

        Assert.assertEquals(2.0, result.getPoint()[0], 1e-6);
    }

    @Test
    public void testOptimize_underDeterminedSystem_returnsConsistentSolution() {

        MultivariateVectorFunction model = new MultivariateVectorFunction() {

            }
        };
        MultivariateMatrixFunction jacobian = new MultivariateMatrixFunction() {

    @Test
    public void testOptimize_nonlinearExponentialModel_convergesApproximately() {
        double[] target = new double[XS.length];
        for (int i = 0; i < XS.length; i++) {
            target[i] = Math.exp(XS[i]); // a=1, b=1
        }

        LevenbergMarquardtOptimizer optimizer = new LevenbergMarquardtOptimizer();
        PointVectorValuePair result = optimizer.optimize(
            new MaxEval(2000),
            new Target(target),
            new Weight(new double[] { 1, 1, 1, 1 }),
            new ModelFunction(expModel),
            new ModelFunctionJacobian(expJacobian),
            new InitialGuess(new double[] { 1.2, 0.8 }));

        Assert.assertEquals(1.0, result.getPoint()[0], 1e-3);
        Assert.assertEquals(1.0, result.getPoint()[1], 1e-3);
    }
