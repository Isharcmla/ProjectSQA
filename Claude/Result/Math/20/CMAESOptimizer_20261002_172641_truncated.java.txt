package org.apache.commons.math3.optimization.direct;

import static org.junit.Assert.*;

import java.util.List;

import org.apache.commons.math3.analysis.MultivariateFunction;
import org.apache.commons.math3.exception.DimensionMismatchException;
import org.apache.commons.math3.exception.MathUnsupportedOperationException;

public class CMAESOptimizerTest {

    /** Simple sphere function: f(x) = sum(x_i^2). Minimum at origin. */
    private static class SphereFunction implements MultivariateFunction {
        public double value(double[] point) {
            double sum = 0;
            for (double v : point) {
                sum += v * v;
            }
            return sum;
        }
    }

    /** Negative sphere function, used to test GoalType.MAXIMIZE. */
    private static class NegativeSphereFunction implements MultivariateFunction {
        public double value(double[] point) {
            double sum = 0;
            for (double v : point) {
                sum += v * v;
            }
            return -sum;
        }
    }

    // ---------------------- Normal optimize scenarios ----------------------

    @Test
    public void testOptimize_sphereFunctionNoBounds_findsMinimumNearZero() {
        double[] sigma = {0.5, 0.5};
        CMAESOptimizer optimizer = new CMAESOptimizer(10, sigma, 2000, 1e-10,
                true, 0, 0, new MersenneTwister(12345), false,
                new SimpleValueChecker(1e-12, 1e-12));
        PointValuePair result = optimizer.optimize(20000, new SphereFunction(),
                GoalType.MINIMIZE, new double[]{0.5, 0.5});
        assertNotNull(result);
        assertEquals(0.0, result.getValue(), 1.0);
    }

    @Test
    public void testOptimize_sphereFunctionWithBounds_findsMinimumNearZero() {
        double[] sigma = {1.0, 1.0};
        CMAESOptimizer optimizer = new CMAESOptimizer(10, sigma, 2000, 1e-10,
                true, 0, 0, new MersenneTwister(12345), false,
                new SimpleValueChecker(1e-12, 1e-12));
        PointValuePair result = optimizer.optimize(20000, new SphereFunction(),
                GoalType.MINIMIZE, new double[]{1.0, 1.0},
                new double[]{-10.0, -10.0}, new double[]{10.0, 10.0});
        assertNotNull(result);
        assertEquals(0.0, result.getValue(), 2.0);
    }

    @Test
    public void testOptimize_maximizeNegativeSphere_findsMaximumNearZero() {
        double[] sigma = {0.5, 0.5};
        CMAESOptimizer optimizer = new CMAESOptimizer(10, sigma, 2000, 0,
                true, 0, 0, new MersenneTwister(12345), false,
                new SimpleValueChecker(1e-12, 1e-12));
        PointValuePair result = optimizer.optimize(20000, new NegativeSphereFunction(),
                GoalType.MAXIMIZE, new double[]{0.5, 0.5});
        assertNotNull(result);
        assertEquals(0.0, result.getValue(), 1.0);
    }

    @Test
    public void testOptimize_generateStatisticsTrue_historiesPopulated() {
        double[] sigma = {0.5, 0.5};
        CMAESOptimizer optimizer = new CMAESOptimizer(10, sigma, 200, 0,
                true, 0, 0, new MersenneTwister(12345), true,
                new SimpleValueChecker(1e-12, 1e-12));
        optimizer.optimize(2000, new SphereFunction(), GoalType.MINIMIZE,
                new double[]{0.5, 0.5});
        assertFalse(optimizer.getStatisticsSigmaHistory().isEmpty());
        assertFalse(optimizer.getStatisticsMeanHistory().isEmpty());
        assertFalse(optimizer.getStatisticsFitnessHistory().isEmpty());
        assertFalse(optimizer.getStatisticsDHistory().isEmpty());
    }

    // ---------------------- Edge cases / exceptions ----------------------

    @Test(expected = NotPositiveException.class)
    public void testOptimize_negativeInputSigma_throwsNotPositiveException() {
        double[] sigma = {-1.0, 0.5};
        CMAESOptimizer optimizer = new CMAESOptimizer(5, sigma);
        optimizer.optimize(1000, new SphereFunction(), GoalType.MINIMIZE,
                new double[]{0.5, 0.5});
    }

    @Test(expected = DimensionMismatchException.class)
    public void testOptimize_inputSigmaDimensionMismatch_throwsDimensionMismatchException() {
        double[] sigma = {0.5, 0.5, 0.5};
        CMAESOptimizer optimizer = new CMAESOptimizer(5, sigma);
        optimizer.optimize(1000, new SphereFunction(), GoalType.MINIMIZE,
                new double[]{0.5, 0.5});
    }

    @Test(expected = MathUnsupportedOperationException.class)
    public void testOptimize_mixedFiniteInfiniteBounds_throwsMathUnsupportedOperationException() {
        double[] sigma = {0.5, 0.5};
        CMAESOptimizer optimizer = new CMAESOptimizer(5, sigma);
        optimizer.optimize(1000, new SphereFunction(), GoalType.MINIMIZE,
                new double[]{0.5, 0.5},
                new double[]{-1.0, Double.NEGATIVE_INFINITY},
                new double[]{1.0, Double.POSITIVE_INFINITY});
    }

    @Test(expected = OutOfRangeException.class)
    public void testOptimize_inputSigmaOutOfRange_throwsOutOfRangeException() {

        optimizer.optimize(1000, new SphereFunction(), GoalType.MINIMIZE,
                new double[]{0.5, 0.5},

    }

    @Test
    public void testOptimize_zeroStartPoint_optimizeSuccessfully
