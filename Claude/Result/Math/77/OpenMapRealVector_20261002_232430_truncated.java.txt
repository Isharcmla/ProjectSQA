    @Test
    public void testSparseIterator_normalInput_iteratesOverNonZeroEntries() {
        OpenMapRealVector v = new OpenMapRealVector(new double[]{0.0, 5.0, 0.0, 10.0});
        java.util.Iterator<RealVector.Entry> iter = v.sparseIterator();
        int count = 0;
        double sum = 0.0;
        while (iter.hasNext()) {
            RealVector.Entry entry = iter.next();
            sum += entry.getValue();
            count++;
            assertTrue(entry.getIndex() >= 0 && entry.getIndex() < 4);
        }
        assertEquals(2, count);
        assertEquals(15.0, sum, 0.0001);
    }
