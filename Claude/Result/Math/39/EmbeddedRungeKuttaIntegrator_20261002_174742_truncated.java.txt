package org.apache.commons.math.ode.nonstiff;

import org.apache.commons.math.exception.MathIllegalArgumentException;
import org.apache.commons.math.ode.ExpandableStatefulODE;
import org.apache.commons.math.ode.FirstOrderDifferentialEquations;
import org.junit.Test;

import static org.junit.Assert.*;

/**
 * Test suite for {@link EmbeddedRungeKuttaIntegrator}.
 *
 * Since EmbeddedRungeKuttaIntegrator is abstract, it is exercised through
 * concrete subclasses that already exist in the same package
 * (DormandPrince54Integrator, HighamHall54Integrator), which only differ
 * in the Butcher array / interpolator they use, but share all the public
 * API defined in the class under test (integrate, getOrder, getSafety,
 * setSafety, getMinReduction, setMinReduction, getMaxGrowth, setMaxGrowth).
 */
public class EmbeddedRungeKuttaIntegratorTest {

    /** dy/dt = y  ->  y(t) = y0 * exp(t) */
    private static class ExponentialODE implements FirstOrderDifferentialEquations {
        public int getDimension() {
            return 1;
        }
        public void computeDerivatives(double t, double[] y, double[] yDot) {
            yDot[0] = y[0];
        }
    }

    /** dy1/dt = 1, dy2/dt = 2 -> linear growth, easy to check analytically */
    private static class ConstantODE implements FirstOrderDifferentialEquations {
        public int getDimension() {
            return 2;
        }
        public void computeDerivatives(double t, double[] y, double[] yDot) {
            yDot[0] = 1.0;
            yDot[1] = 2.0;
        }
    }

    // ---------------------------------------------------------------
    // (ก) Normal / typical input
    // ---------------------------------------------------------------

    @Test
    public void testIntegrate_forwardFsalMethod_producesExpectedResult() {
        DormandPrince54Integrator integrator =
            new DormandPrince54Integrator(1.0e-8, 100.0, 1.0e-10, 1.0e-10);

        ExpandableStatefulODE equations = new ExpandableStatefulODE(new ExponentialODE());
        equations.setTime(0.0);
        equations.setPrimaryState(new double[] { 1.0 });

        integrator.integrate(equations, 1.0);

        double[] result = equations.getPrimaryState();
        assertEquals(Math.E, result[0], 1.0e-6);
        assertEquals(1.0, equations.getTime(), 1.0e-12);
    }

    @Test
    public void testIntegrate_nonFsalMethod_producesExpectedResult() {
        HighamHall54Integrator integrator =
            new HighamHall54Integrator(1.0e-8, 100.0, 1.0e-10, 1.0e-10);

        ExpandableStatefulODE equations = new ExpandableStatefulODE(new ConstantODE());
        equations.setTime(0.0);
        equations.setPrimaryState(new double[] { 0.0, 0.0 });

        integrator.integrate(equations, 2.0);

        double[] result = equations.getPrimaryState();
        assertEquals(2.0, result[0], 1.0e-6);
        assertEquals(4.0, result[1], 1.0e-6);
    }

    @Test
    public void testIntegrate_vectorTolerance_producesExpectedResult() {
        DormandPrince54Integrator integrator =
            new DormandPrince54Integrator(1.0e-8, 100.0,
                                           new double[] { 1.0e-10 },
                                           new double[] { 1.0e-10 });

        ExpandableStatefulODE equations = new ExpandableStatefulODE(new ExponentialODE());
        equations.setTime(0.0);
        equations.setPrimaryState(new double[] { 1.0 });

        integrator.integrate(equations, 1.0);

        double[] result = equations.getPrimaryState();
        assertEquals(Math.E, result[0], 1.0e-6);
    }
