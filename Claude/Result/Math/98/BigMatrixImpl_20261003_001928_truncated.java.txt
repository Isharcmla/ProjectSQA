    private BigDecimal[][] toBigDecimal(double[][] d) {
        BigDecimal[][] result = new BigDecimal[d.length][d[0].length];
        for (int i = 0; i < d.length; i++) {
            for (int j = 0; j < d[0].length; j++) {
                result[i][j] = new BigDecimal(d[i][j]);
            }
        }
        return result;
    }

    private BigMatrixImpl createMatrix(double[][] d) {
        return new BigMatrixImpl(toBigDecimal(d));
    }
