import org.junit.Test;
import org.junit.Before;
import static org.junit.Assert.*;

import java.io.ByteArrayInputStream;

import org.apache.commons.math.optimization.GoalType;

public class SimplexTableauTest {

    private static final double TOL = 1e-9;

    private SimplexTableau buildLeqTableau() {
        LinearObjectiveFunction f = new LinearObjectiveFunction(new double[] {1, 2}, 0);
        List<LinearConstraint> constraints = new ArrayList<LinearConstraint>();
        constraints.add(new LinearConstraint(new double[] {1, 0}, Relationship.LEQ, 4));
        constraints.add(new LinearConstraint(new double[] {0, 1}, Relationship.LEQ, 10));
        return new SimplexTableau(f, constraints, GoalType.MAXIMIZE, true, 1e-6);
    }

    private SimplexTableau buildGeqTableau() {
        LinearObjectiveFunction f = new LinearObjectiveFunction(new double[] {1, 1}, 0);
        List<LinearConstraint> constraints = new ArrayList<LinearConstraint>();
        constraints.add(new LinearConstraint(new double[] {1, 1}, Relationship.GEQ, 1));
        return new SimplexTableau(f, constraints, GoalType.MAXIMIZE, true, 1e-6);
    }

    @Test
    public void testConstructor_typicalLEQConstraints_createsTableau() {
        SimplexTableau tableau = buildLeqTableau();

        assertEquals(1, tableau.getNumObjectiveFunctions());
        assertEquals(2, tableau.getNumDecisionVariables());
        assertEquals(2, tableau.getOriginalNumDecisionVariables());
        assertEquals(2, tableau.getNumSlackVariables());
        assertEquals(0, tableau.getNumArtificialVariables());
        assertEquals(6, tableau.getWidth());
        assertEquals(3, tableau.getHeight());

        assertEquals(1.0, tableau.getEntry(0, 0), TOL);
        assertEquals(-1.0, tableau.getEntry(0, 1), TOL);
        assertEquals(-2.0, tableau.getEntry(0, 2), TOL);
        assertEquals(0.0, tableau.getEntry(0, 5), TOL);

        assertEquals(1.0, tableau.getEntry(1, 1), TOL);
        assertEquals(0.0, tableau.getEntry(1, 2), TOL);
        assertEquals(1.0, tableau.getEntry(1, 3), TOL);
        assertEquals(4.0, tableau.getEntry(1, 5), TOL);

        assertEquals(0.0, tableau.getEntry(2, 1), TOL);
        assertEquals(1.0, tableau.getEntry(2, 2), TOL);
        assertEquals(1.0, tableau.getEntry(2, 4), TOL);
        assertEquals(10.0, tableau.getEntry(2, 5), TOL);
    }

    @Test
    public void testConstructor_withMaxUlps_createsTableau() {
        LinearObjectiveFunction f = new LinearObjectiveFunction(new double[] {1, 2}, 0);
        List<LinearConstraint> constraints = new ArrayList<LinearConstraint>();
        constraints.add(new LinearConstraint(new double[] {1, 0}, Relationship.LEQ, 4));
        SimplexTableau tableau = new SimplexTableau(f, constraints, GoalType.MAXIMIZE, true, 1e-6, 5);
        assertEquals(2, tableau.getNumDecisionVariables());
    }

    @Test
    public void testConstructor_geqConstraintTriggersPhase1_createsTableauWithArtificialVar() {
        SimplexTableau tableau = buildGeqTableau();

        assertEquals(2, tableau.getNumObjectiveFunctions());
        assertEquals(2, tableau.getNumDecisionVariables());
        assertEquals(1, tableau.getNumSlackVariables());
        assertEquals(1, tableau.getNumArtificialVariables());
        assertEquals(7, tableau.getWidth());
        assertEquals(3, tableau.getHeight());

        assertEquals(-1.0, tableau.getEntry(0, 0), TOL);
        assertEquals(-1.0, tableau.getEntry(0, 2), TOL);
        assertEquals(-1.0, tableau.getEntry(0, 3), TOL);
        assertEquals(1.0, tableau.getEntry(0, 4), TOL);
        assertEquals(0.0, tableau.getEntry(0, 5), TOL);
        assertEquals(-1.0, tableau.getEntry(0, 6), TOL);

        assertEquals(1.0, tableau.getEntry(1, 1), TOL);
        assertEquals(-1.0, tableau.getEntry(1, 2), TOL);
        assertEquals(-1.0, tableau.getEntry(1, 3), TOL);

        assertEquals(1.0, tableau.getEntry(2, 2), TOL);
        assertEquals(1.0, tableau.getEntry(2, 3), TOL);
        assertEquals(-1.0, tableau.getEntry(2, 4), TOL);
        assertEquals(1.0, tableau.getEntry(2, 5), TOL);
        assertEquals(1.0, tableau.getEntry(2, 6), TOL);
    }

    @Test
    public void testConstructor_restrictToNonNegativeFalse_addsExtraDecisionVariable() {
        LinearObjectiveFunction f = new LinearObjectiveFunction(new double[] {1, 1}, 0);
        List<LinearConstraint> constraints = new ArrayList<LinearConstraint>();
        constraints.add(new LinearConstraint(new double[] {1, 1}, Relationship.LEQ, 10));
        SimplexTableau tableau = new SimplexTableau(f, constraints, GoalType.MAXIMIZE, false, 1e-6);

        assertEquals(3, tableau.getNumDecisionVariables());
        assertEquals(2, tableau.getOriginalNumDecisionVariables());
        assertEquals(6, tableau.getWidth());
        assertEquals(2, tableau.getHeight());

        assertEquals(1.0, tableau.getEntry(0, 0), TOL);
        assertEquals(-1.0, tableau.getEntry(0, 1), TOL);
        assertEquals(-1.0, tableau.getEntry(0, 2), TOL);
        assertEquals(2.0, tableau.getEntry(0, 3), TOL);
        assertEquals(0.0, tableau.getEntry(0, 5), TOL);

        assertEquals(1.0, tableau.getEntry(1, 1), TOL);
        assertEquals(1.0, tableau.getEntry(1, 2), TOL);
        assertEquals(-2.0, tableau.getEntry(1, 3), TOL);
        assertEquals(1.0, tableau.getEntry(1, 4), TOL);
        assertEquals(10.0, tableau.getEntry(1, 5), TOL);
    }

    @Test
    public void testIsOptimal_unsolvedTableau_returnsFalse() {
        SimplexTableau tableau = buildLeqTableau();
        assertFalse(tableau.isOptimal());
    }

    @Test
    public void testIsOptimal_optimalStateEntries_returnsTrue() {
        SimplexTableau tableau = buildLeqTableau();
        // make all objective-row entries (except obj col and RHS) non-negative
        tableau.setEntry(0, 1, 0.0);
        tableau.setEntry(0, 2, 0.0);
        assertTrue(tableau.isOptimal());
    }

    @Test
    public void testGetBasicRow_slackColumn_returnsCorrectRow() {
        SimplexTableau tableau = buildLeqTableau();
        assertEquals(Integer.valueOf(1), tableau.getBasicRow(3));
        assertEquals(Integer.valueOf(2), tableau.getBasicRow(4));
    }

    @Test
    public void testGetBasicRow_notBasicColumn_returnsNull() {

        assertNull(tableau.getBasicRow(1));
    }

    @Test
    public void testGetBasicRow_allZeroColumn_returnsNull() {
        SimplexTableau tableau = buildLeqTableau();

            tableau.setEntry(i, 1, 0.0);
        }
        assertNull(tableau.getBasicRow(1));
    }

    @Test
    public void testGetSolution_unsolvedTableau_returnsZeros() {
        SimplexTableau tableau = buildLeqTableau();
        RealPointValuePair solution = tableau.getSolution();
        assertArrayEquals(new double[] {0.0, 0.0}, solution.getPoint(), TOL);
        assertEquals(0.0, solution.getValue(), TOL);
    }

    @Test
    public void testGetSolution_simulatedSolvedState_returnsExpectedValues() {
        SimplexTableau tableau = buildLeqTableau();
        // make columns x0, x1 "basic" by zeroing their objective-row entries
        tableau.setEntry(0, 1, 0.0);
        tableau.setEntry(0, 2, 0.0);

        RealPointValuePair solution = tableau.getSolution();
        assertArrayEquals(new double[] {4.0, 10.0}, solution.getPoint(), TOL);
        assertEquals(24.0, solution.getValue(), TOL);
    }

    @Test
    public void testDivideRow_dividesEntriesCorrectly() {
        SimplexTableau tableau = buildLeqTableau();
        tableau.divideRow(1, 2.0);
        assertEquals(0.0, tableau.getEntry(1, 0), TOL);
        assertEquals(0.5, tableau.getEntry(1, 1), TOL);
        assertEquals(0.0, tableau.getEntry(1, 2), TOL);
        assertEquals(0.5, tableau.getEntry(1, 3), TOL);
        assertEquals(0.0, tableau.getEntry(1, 4), TOL);
        assertEquals(2.0, tableau.getEntry(1, 5), TOL);
    }

    @Test
    public void testSubtractRow_subtractsMultipleOfRow() {
        SimplexTableau tableau = buildLeqTableau();
        // row2 - 1*row1
        tableau.subtractRow(2, 1, 1.0);
        assertEquals(0.0, tableau.getEntry(2, 1), TOL);
        assertEquals(1.0, tableau.getEntry(2, 2), TOL);
        assertEquals(-1.0, tableau.getEntry(2, 3), TOL);
        assertEquals(1.0, tableau.getEntry(2, 4), TOL);
        assertEquals(6.0, tableau.getEntry(2, 5), TOL);
    }

    @Test
    public void testGetEntryAndSetEntry_workCorrectly() {
        SimplexTableau tableau = buildLeqTableau();
        tableau.setEntry(1, 1, 99.0);
        assertEquals(99.0, tableau.getEntry(1, 1), TOL);
    }

    @Test
    public void testGetData_matchesGetEntry() {
        SimplexTableau tableau = buildLeqTableau();
        double[][] data = tableau.getData();
        for (int i = 0; i < tableau.getHeight(); i++) {
            for (int j = 0; j < tableau.getWidth(); j++) {
                assertEquals(tableau.getEntry(i, j), data[i][j], TOL);
            }
        }
    }
