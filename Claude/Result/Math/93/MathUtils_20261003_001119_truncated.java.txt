import java.math.BigDecimal;
import org.apache.commons.math.util.MathUtils;

public class MathUtilsTest {

    // ---------- addAndCheck(int,int) ----------
    @Test
    public void testAddAndCheckInt_normal_returnsSum() {
        Assert.assertEquals(7, MathUtils.addAndCheck(3, 4));
    }

    @Test(expected = ArithmeticException.class)
    public void testAddAndCheckInt_overflow_throwsException() {
        MathUtils.addAndCheck(Integer.MAX_VALUE, 1);
    }

    // ---------- addAndCheck(long,long) ----------
    @Test
    public void testAddAndCheckLong_normal_returnsSum() {
        Assert.assertEquals(15L, MathUtils.addAndCheck(7L, 8L));
    }

    @Test(expected = ArithmeticException.class)
    public void testAddAndCheckLong_positiveOverflow_throwsException() {
        MathUtils.addAndCheck(Long.MAX_VALUE, 1L);
    }

    @Test(expected = ArithmeticException.class)
    public void testAddAndCheckLong_negativeOverflow_throwsException() {
        MathUtils.addAndCheck(Long.MIN_VALUE, -1L);
    }

    @Test
    public void testAddAndCheckLong_oppositeSign_returnsSum() {
        Assert.assertEquals(2L, MathUtils.addAndCheck(-3L, 5L));
    }

    // ---------- binomialCoefficient ----------

        Assert.assertEquals(1L, MathUtils.binomialCoefficient(5, 0));
    }

    @Test

        Assert.assertEquals(5L, MathUtils.binomialCoefficient(5, 1));

    @Test
    public void testBinomialCoefficient_kEqualsNMinusOne_returnsN() {

    public void testBinomialCoefficient_nLessThanK_throwsException() {

        MathUtils.binomialCoefficient(-1, 0);

        MathUtils.binomialCoefficientLog(-1, 0);

    public void testCosh_zero_returnsOne() {

    }

    // ---------- equals(double,double) ----------
    @Test
    public void testEqualsDouble_sameValues_returnsTrue() {
        Assert.assertTrue(MathUtils.equals(1.5, 1.5));
    }

    @Test
    public void testEqualsDouble_bothNaN_returnsTrue() {
        Assert.assertTrue(MathUtils.equals(Double.NaN, Double.NaN));
    }

    @Test
    public void testEqualsDouble_differentValues_returnsFalse() {
        Assert.assertFalse(MathUtils.equals(1.0, 2.0));
    }

    // ---------- equals(double[],double[]) ----------
