    // ---------- equals(double,double) ----------
    @Test
    public void testEqualsDouble_bothNaN_returnsTrue() {
        assertTrue(MathUtils.equals(Double.NaN, Double.NaN));
    }

    @Test
    public void testEqualsDouble_equalValues_returnsTrue() {
        assertTrue(MathUtils.equals(1.0, 1.0));
    }

    @Test
    public void testEqualsDouble_differentValues_returnsFalse() {
        assertFalse(MathUtils.equals(1.0, 2.0));
    }

    // ---------- equals(double,double,double) ----------
    @Test
    public void testEqualsDoubleEps_withinEps_returnsTrue() {
        assertTrue(MathUtils.equals(1.0, 1.0005, 0.001));
    }

    @Test
    public void testEqualsDoubleEps_outsideEps_returnsFalse() {
        assertFalse(MathUtils.equals(1.0, 2.0, 0.001));
    }

    // ---------- equals(double,double,int) ----------
    @Test
    public void testEqualsDoubleUlps_withinUlps_returnsTrue() {
        double x = 1.0;
        double y = MathUtils.nextAfter(x, Double.POSITIVE_INFINITY);
        assertTrue(MathUtils.equals(x, y, 1));
    }

    @Test
    public void testEqualsDoubleUlps_outsideUlps_returnsFalse() {
        assertFalse(MathUtils.equals(1.0, 2.0, 1));
    }

    @Test
    public void testEqualsDoubleUlps_negativeNumbers_returnsExpected() {
        assertTrue(MathUtils.equals(-1.0, -1.0, 1));
    }

    // ---------- equals(double[],double[]) ----------
    @Test
    public void testEqualsDoubleArray_bothNull_returnsTrue() {
        assertTrue(MathUtils.equals(null, null));
    }

    @Test
    public void testEqualsDoubleArray_oneNull_returnsFalse() {
        assertFalse(MathUtils.equals(new double[]{1.0}, null));
        assertFalse(MathUtils.equals(null, new double[]{1.0}));
    }

    @Test
    public void testEqualsDoubleArray_differentLength_returnsFalse() {
        assertFalse(MathUtils.equals(new double[]{1.0}, new double[]{1.0, 2.0}));
    }

    @Test
    public void testEqualsDoubleArray_equalArrays_returnsTrue() {
        assertTrue(MathUtils.equals(new double[]{1.0, 2.0}, new double[]{1.0, 2.0}));
    }

    @Test
    public void testEqualsDoubleArray_differentElements_returnsFalse() {
        assertFalse(MathUtils.equals(new double[]{1.0, 2.0}, new double[]{1.0, 3.0}));
    }
