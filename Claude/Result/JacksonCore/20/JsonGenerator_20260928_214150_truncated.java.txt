    // ---------------------------------------------------------------
    // Feature enum tests
    // ---------------------------------------------------------------

    @Test
    public void testFeature_collectDefaults_returnsExpectedBitmask() {
        int expected = JsonGenerator.Feature.AUTO_CLOSE_TARGET.getMask()
                | JsonGenerator.Feature.AUTO_CLOSE_JSON_CONTENT.getMask()
                | JsonGenerator.Feature.FLUSH_PASSED_TO_STREAM.getMask()
                | JsonGenerator.Feature.QUOTE_FIELD_NAMES.getMask()
                | JsonGenerator.Feature.QUOTE_NON_NUMERIC_NUMBERS.getMask();
        assertEquals(expected, JsonGenerator.Feature.collectDefaults());
    }

    @Test
    public void testFeature_enabledByDefault_matchesExpectedValues() {
        assertTrue(JsonGenerator.Feature.AUTO_CLOSE_TARGET.enabledByDefault());
        assertTrue(JsonGenerator.Feature.QUOTE_FIELD_NAMES.enabledByDefault());
        assertFalse(JsonGenerator.Feature.WRITE_NUMBERS_AS_STRINGS.enabledByDefault());
        assertFalse(JsonGenerator.Feature.STRICT_DUPLICATE_DETECTION.enabledByDefault());
        assertFalse(JsonGenerator.Feature.IGNORE_UNKNOWN.enabledByDefault());
    }

    @Test
    public void testFeature_getMask_isPowerOfTwoBasedOnOrdinal() {
        assertEquals(1 << JsonGenerator.Feature.AUTO_CLOSE_TARGET.ordinal(),
                JsonGenerator.Feature.AUTO_CLOSE_TARGET.getMask());
    }

    @Test
    public void testFeature_enabledIn_trueWhenBitSet() {
        int mask = JsonGenerator.Feature.QUOTE_FIELD_NAMES.getMask();
        assertTrue(JsonGenerator.Feature.QUOTE_FIELD_NAMES.enabledIn(mask));
        assertFalse(JsonGenerator.Feature.ESCAPE_NON_ASCII.enabledIn(mask));
    }

    @Test
    public void testConfigure_stateTrue_enablesFeature() {
        gen.configure(JsonGenerator.Feature.ESCAPE_NON_ASCII, true);
        assertTrue(gen.isEnabled(JsonGenerator.Feature.ESCAPE_NON_ASCII));

    @Test
    public void testConfigure_stateFalse_disablesFeature() {
        gen.enable(JsonGenerator.Feature.ESCAPE_NON_ASCII);
        gen.configure(JsonGenerator.Feature.ESCAPE_NON_ASCII, false);
        assertFalse(gen.isEnabled(JsonGenerator.Feature.ESCAPE_NON_ASCII));
    }
