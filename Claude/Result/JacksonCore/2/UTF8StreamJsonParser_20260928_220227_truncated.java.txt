public class UTF8StreamJsonParserTest {

    private JsonFactory factory;

    @Before
    public void setUp() {
        factory = new JsonFactory();
    }

    private JsonParser createParser(String json) throws IOException {
        return factory.createParser(json.getBytes("UTF-8"));
    }

    private JsonParser createParserFromStream(InputStream in) throws IOException {
        return factory.createParser(in);
    }

    // ---------------- Basic token traversal ----------------

    @Test
    public void testNextToken_simpleObject_returnsFieldNamesAndValues() throws Exception {
        JsonParser p = createParser("{\"a\":1,\"b\":\"text\"}");
        assertEquals(JsonToken.START_OBJECT, p.nextToken());
        assertEquals(JsonToken.FIELD_NAME, p.nextToken());
        assertEquals("a", p.getCurrentName());
        assertEquals(JsonToken.VALUE_NUMBER_INT, p.nextToken());
        assertEquals(1, p.getIntValue());
        assertEquals(JsonToken.FIELD_NAME, p.nextToken());
        assertEquals("b", p.getCurrentName());
        assertEquals(JsonToken.VALUE_STRING, p.nextToken());
        assertEquals("text", p.getText());
        assertEquals(JsonToken.END_OBJECT, p.nextToken());
        assertNull(p.nextToken());
        p.close();
    }

    @Test
    public void testNextToken_emptyObject_returnsStartAndEndObject() throws Exception {
        JsonParser p = createParser("{}");
        assertEquals(JsonToken.START_OBJECT, p.nextToken());
        assertEquals(JsonToken.END_OBJECT, p.nextToken());
        assertNull(p.nextToken());
        p.close();
    }

    @Test
    public void testNextToken_array_returnsStartAndEndArray() throws Exception {
        JsonParser p = createParser("[1,2,3]");
        assertEquals(JsonToken.START_ARRAY, p.nextToken());
        assertEquals(JsonToken.VALUE_NUMBER_INT, p.nextToken());
        assertEquals(1, p.getIntValue());
        assertEquals(JsonToken.VALUE_NUMBER_INT, p.nextToken());
        assertEquals(JsonToken.VALUE_NUMBER_INT, p.nextToken());
        assertEquals(JsonToken.END_ARRAY, p.nextToken());
        assertNull(p.nextToken());
        p.close();
    }

    @Test
    public void testNextToken_nestedStructure_parsesCorrectly() throws Exception {
        JsonParser p = createParser("{\"arr\":[{\"x\":true},{\"y\":false}]}");
        assertEquals(JsonToken.START_OBJECT, p.nextToken());
        assertEquals(JsonToken.FIELD_NAME, p.nextToken());
        assertEquals(JsonToken.START_ARRAY, p.nextToken());
        assertEquals(JsonToken.START_OBJECT, p.nextToken());
        assertEquals(JsonToken.FIELD_NAME, p.nextToken());
        assertEquals(JsonToken.VALUE_TRUE, p.nextToken());
        assertEquals(JsonToken.END_OBJECT, p.nextToken());
        assertEquals(JsonToken.START_OBJECT, p.nextToken());
        assertEquals(JsonToken.FIELD_NAME, p.nextToken());
        assertEquals(JsonToken.VALUE_FALSE, p.nextToken());
        assertEquals(JsonToken.END_OBJECT, p.nextToken());
        assertEquals(JsonToken.END_ARRAY, p.nextToken());
        assertEquals(JsonToken.END_OBJECT, p.nextToken());
        p.close();

    // ---------------- Binary / base64 ----------------

    @Test
    public void testGetBinaryValue_base64String_returnsDecodedBytes() throws Exception {
        byte[] original = "hello world".getBytes("UTF-8");
        String encoded = Base64Variants.getDefaultVariant().encode(original);
        JsonParser p = createParser("\"" + encoded + "\"");
        p.nextToken();
        byte[] decoded = p.getBinaryValue(Base64Variants.getDefaultVariant());
        assertArrayEquals(original, decoded);
        p.close();
    }

    @Test
    public void testReadBinaryValue_base64String_writesToOutputStream() throws Exception {
        byte[] original = "some binary data here".getBytes("UTF-8");
        String encoded = Base64Variants.getDefaultVariant().encode(original);
        JsonParser p = createParser("\"" + encoded + "\"");
        p.nextToken();
        ByteArrayOutputStream out = new ByteArrayOutputStream();
        int count = p.readBinaryValue(Base64Variants.getDefaultVariant(), out);
        assertEquals(original.length, count);
        assertArrayEquals(original, out.toByteArray());
        p.close();
    }

    @Test
    public void testGetBinaryValue_afterTextAlreadyRead_returnsDecodedBytes() throws Exception {
        byte[] original = "abc".getBytes("UTF-8");
        String encoded = Base64Variants.getDefaultVariant().encode(original);
        JsonParser p = createParser("\"" + encoded + "\"");
        p.nextToken();
        // force reading text first, making token no longer incomplete
        p.getText();
        byte[] decoded = p.getBinaryValue(Base64Variants.getDefaultVariant());
        assertArrayEquals(original, decoded);
        p.close();
    }

    @Test
    public void testNextFieldName_matchingName_returnsTrue() throws Exception {
        JsonParser p = createParser("{\"name\":\"value\"}");
        p.nextToken();

        assertTrue(matched);
