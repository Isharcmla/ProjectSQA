package com.fasterxml.jackson.core.json;

import com.fasterxml.jackson.core.*;
import org.junit.Before;
import org.junit.Test;

import java.io.ByteArrayOutputStream;
import java.io.IOException;
import java.io.StringReader;
import java.io.StringWriter;

import static org.junit.Assert.*;

public class ReaderBasedJsonParserTest {

    private JsonFactory factory;

    @Before
    public void setUp() {
        factory = new JsonFactory();
    }

    private ReaderBasedJsonParser createParser(String json) throws IOException {
        JsonParser p = factory.createParser(new StringReader(json));
        assertTrue("Expected ReaderBasedJsonParser instance", p instanceof ReaderBasedJsonParser);
        return (ReaderBasedJsonParser) p;
    }

    // -------------------- basic codec tests --------------------

    @Test
    public void testGetCodec_default_returnsNull() throws IOException {
        ReaderBasedJsonParser p = createParser("123");
        assertNull(p.getCodec());
        p.close();
    }

    @Test
    public void testSetCodec_nullValue_getCodecReturnsNull() throws IOException {
        ReaderBasedJsonParser p = createParser("123");
        p.setCodec(null);
        assertNull(p.getCodec());
        p.close();
    }

    @Test
    public void testGetInputSource_returnsUnderlyingReader() throws IOException {
        StringReader reader = new StringReader("123");
        JsonParser jp = factory.createParser(reader);
        ReaderBasedJsonParser p = (ReaderBasedJsonParser) jp;
        assertSame(reader, p.getInputSource());
        p.close();
    }

    // -------------------- simple token traversal --------------------

    @Test
    public void testNextToken_emptyInput_returnsNullAndCloses() throws IOException {
        ReaderBasedJsonParser p = createParser("");
        JsonToken t = p.nextToken();
        assertNull(t);
        assertTrue(p.isClosed());
    }

    @Test
    public void testNextToken_whitespaceOnly_returnsNull() throws IOException {
        ReaderBasedJsonParser p = createParser("   \t\n  ");
        JsonToken t = p.nextToken();
        assertNull(t);
    }

    @Test
    public void testNextToken_simpleObject_parsesFieldAndValue() throws IOException {
        ReaderBasedJsonParser p = createParser("{\"a\":1}");
        assertEquals(JsonToken.START_OBJECT, p.nextToken());
        assertEquals(JsonToken.FIELD_NAME, p.nextToken());
        assertEquals("a", p.getCurrentName());
        assertEquals(JsonToken.VALUE_NUMBER_INT, p.nextToken());
        assertEquals(1, p.getIntValue());
        assertEquals(JsonToken.END_OBJECT, p.nextToken());
        assertNull(p.nextToken());
        p.close();
    }

    @Test
    public void testNextToken_simpleArray_parsesElements() throws IOException {
        ReaderBasedJsonParser p = createParser("[1,2,3]");
        assertEquals(JsonToken.START_ARRAY, p.nextToken());
        assertEquals(JsonToken.VALUE_NUMBER_INT, p.nextToken());
        assertEquals(1, p.getIntValue());
        assertEquals(JsonToken.VALUE_NUMBER_INT, p.nextToken());
        assertEquals(2, p.getIntValue());
        assertEquals(JsonToken.VALUE_NUMBER_INT, p.nextToken());
        assertEquals(3, p.getIntValue());
        assertEquals(JsonToken.END_ARRAY, p.nextToken());
        p.close();
    }

    @Test
    public void testNextToken_nestedObjectArray_parsesCorrectly() throws IOException {
        ReaderBasedJsonParser p = createParser("{\"list\":[true,false,null]}");
        assertEquals(JsonToken.START_OBJECT, p.nextToken());
        assertEquals(JsonToken.FIELD_NAME, p.nextToken());
        assertEquals(JsonToken.START_ARRAY, p.nextToken());
        assertEquals(JsonToken.VALUE_TRUE, p.nextToken());
        assertEquals(JsonToken.VALUE_FALSE, p.nextToken());
        assertEquals(JsonToken.VALUE_NULL, p.nextToken());
        assertEquals(JsonToken.END_ARRAY, p.nextToken());
        assertEquals(JsonToken.END_OBJECT, p.nextToken());
        p.close();
    }

    @Test(expected = JsonParseException.class)
    public void testNextToken_mismatchedEndMarker_throwsException() throws IOException {
        ReaderBasedJsonParser p = createParser("[1,2}");
        p.nextToken(); // START_ARRAY
        p.nextToken(); // 1
        p.nextToken(); // 2
        p.nextToken(); // should throw on '}'
    }

    @Test(expected = JsonParseException.class)
    public void testNextToken_unexpectedCloseBracket_throwsException() throws IOException {
        ReaderBasedJsonParser p = createParser("{]");
        p.nextToken(); // START_OBJECT
        p.nextToken(); // should throw
    }

    // -------------------- skip string (tokenIncomplete) --------------------

    @Test
    public void testNextToken_skipsIncompleteString_movesToNext() throws IOException {
        ReaderBasedJsonParser p = createParser("[\"abc\",\"def\"]");
        p.nextToken(); // START_ARRAY
        p.nextToken(); // VALUE_STRING "abc" but not read
        p.nextToken(); // should skip "abc" and give VALUE_STRING "def"
        assertEquals("def", p.getText());
        p.close();
    }

        assertTrue(Double.isNaN(p.getDoubleValue()));
