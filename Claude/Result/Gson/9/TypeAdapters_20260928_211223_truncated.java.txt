  private JsonReader reader(String json) {
    JsonReader r = new JsonReader(new StringReader(json));
    r.setLenient(true);
    return r;
  }

  private JsonWriter writer(StringWriter sw) {
    JsonWriter w = new JsonWriter(sw);
    w.setLenient(true);
    return w;
  }

  // ---------- private constructor ----------
  @Test
  public void testConstructor_throwsUnsupportedOperationException() throws Exception {
    Constructor<TypeAdapters> constructor = TypeAdapters.class.getDeclaredConstructor();
    constructor.setAccessible(true);
    try {
      constructor.newInstance();
      fail("Expected exception");
    } catch (InvocationTargetException e) {
      assertTrue(e.getCause() instanceof UnsupportedOperationException);
    }
  }
