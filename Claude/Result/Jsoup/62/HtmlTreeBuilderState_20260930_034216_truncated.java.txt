package org.jsoup.parser;

import org.jsoup.Jsoup;
import org.jsoup.nodes.Document;
import org.jsoup.nodes.Element;
import org.jsoup.select.Elements;
import org.junit.Test;

import static org.junit.Assert.*;

/**
 * Tests for HtmlTreeBuilderState.
 * Since HtmlTreeBuilderState is package-private and is only exercised internally
 * by the HtmlTreeBuilder during parsing, these tests drive it indirectly through
 * the public org.jsoup.Jsoup API, using HTML fragments crafted to exercise the
 * various states and branches of the state machine.
 */
public class HtmlTreeBuilderStateTest {

    // ---------- Initial state ----------

    @Test
    public void testInitial_withValidDoctype_createsDoctypeAndTransitions() {
        String html = "<!DOCTYPE html><html><head><title>T</title></head><body>Hi</body></html>";
        Document doc = Jsoup.parse(html);
        assertNotNull(doc);
        assertEquals("T", doc.title());
    }

    @Test
    public void testInitial_withWhitespaceBeforeDoctype_ignoredWhitespace() {
        String html = "   \n\t <!DOCTYPE html><html><body>x</body></html>";
        Document doc = Jsoup.parse(html);
        assertNotNull(doc);
        assertEquals("x", doc.body().text());
    }

    @Test
    public void testInitial_withCommentBeforeDoctype_insertsComment() {
        String html = "<!-- c1 --><!DOCTYPE html><html><body></body></html>";
        Document doc = Jsoup.parse(html);
        assertNotNull(doc);
        assertTrue(doc.childNode(0).toString().contains("c1") || doc.outerHtml().contains("c1"));
    }

    @Test
    public void testInitial_withoutDoctype_transitionsToBeforeHtml() {
        String html = "<html><head><title>NoDoctype</title></head><body></body></html>";
        Document doc = Jsoup.parse(html);
        assertEquals("NoDoctype", doc.title());
    }

    @Test
    public void testInitial_emptyString_producesEmptyDocument() {
        Document doc = Jsoup.parse("");
        assertNotNull(doc);
        assertNotNull(doc.body());
    }

    // ---------- BeforeHtml state ----------

    @Test
    public void testBeforeHtml_secondDoctypeIgnored() {
        String html = "<!DOCTYPE html><!DOCTYPE html><html><body>ok</body></html>";
        Document doc = Jsoup.parse(html);
        assertEquals("ok", doc.body().text());
    }

    @Test
    public void testBeforeHtml_commentAfterDoctype_inserted() {
        String html = "<!DOCTYPE html><!--hi--><html><body></body></html>";
        Document doc = Jsoup.parse(html);
        assertTrue(doc.outerHtml().contains("hi"));
    }

    @Test
    public void testBeforeHtml_endTagInSetTriggersAnythingElse() {
        String html = "<!DOCTYPE html></head><html><body>t</body></html>";
        Document doc = Jsoup.parse(html);
        assertNotNull(doc);
        assertEquals(1, doc.select("html").size());
    }

    @Test
    public void testBeforeHtml_otherEndTagIgnored() {
        String html = "<!DOCTYPE html></div><html><body>t2</body></html>";
        Document doc = Jsoup.parse(html);
        assertNotNull(doc);
    }

    @Test
    public void testBeforeHtml_otherTokenTriggersAnythingElse() {
        String html = "<!DOCTYPE html>Hello<html><body></body></html>";
        Document doc = Jsoup.parse(html);
        assertNotNull(doc);
        assertTrue(doc.text().contains("Hello"));
    }

        String html = "<html><head><title>H</title></head><body></body></html>";

        assertEquals("H", doc.title());
    }

    @Test

        String html = "<html></body><body>x</body></html>";

    }

    @Test
    public void testBeforeHead_otherStartTagInsertsHead() {
        String html = "<html><p>para</p></html>";

        assertEquals(1, doc.select("p").size());
    }

    // ---------- InHead state ----------

    @Test
    public void testInHead_whitespaceInserted() {

    public void testInHead_commentInserted() {

        Document doc = Jsoup.parse(html);

    @Test
    public void testInHead_baseTagWithHref_setsBaseUri() {
        String html = "<html><head><base href='http://example.com/'></head><body><a href='x'>a</a></body></html>";
        Document doc = Jsoup.parse(html, "http://original.com/");
        Element a = doc.select("a").first();
        assertNotNull(a);
        assertTrue(a.attr("abs:href").contains("example.com") || a.attr("abs:href").contains("x"));
    }

    @Test
    public void testInHead_metaTagInserted() {
        String html = "<html><head><meta charset='utf-8'></head><body></body></html>";

    @Test
    public void testInHead_styleTag_handledAsRawText() {

    // ---------- InBody state ----------

    @Test
    public void testInBody_nullCharacter_errorHandled() {

    @Test

    }

    @Test
    public void testInBody_anchorTag_pushesActiveFormatting() {
        String html = "<html><body><a href='x'>link</a></body></html>";

    }

    @Test
    public void testInBody_duplicateAnchorTag_closesPrevious() {
        String html = "<html><body><a href='1'>one<a href='2'>two</a></a></body></html>";

        assertTrue(doc.select("a").size() >= 1);
    }
