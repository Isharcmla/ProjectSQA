    @Test
    public void testGetNamespaceURIPrefix_customPrefixFound_returnsURIAndCaches() throws Exception {
        Document doc = newDocument();
        Element root = doc.createElement("root");
        root.setAttribute("xmlns:ns", "http://example.com/ns");
        doc.appendChild(root);
        Element child = doc.createElement("child");
        root.appendChild(child);
        DOMNodePointer p = new DOMNodePointer(child, Locale.ENGLISH);
        assertEquals("http://example.com/ns", p.getNamespaceURI("ns"));
        // second call hits the cache branch
        assertEquals("http://example.com/ns", p.getNamespaceURI("ns"));
    }
