    @Test

        Element root = new Element("root");
        Element child = new Element("child");
        root.addContent(child);

        JDOMNodePointer rootPointer = new JDOMNodePointer(root, locale);
        JDOMNodePointer childPointer1 = new JDOMNodePointer(rootPointer, child);

        assertEquals(0, result);
    }

    @Test
    public void testCompareChildNodePointers_elementChildren_returnsOrder() {
        Element root = new Element("root");
        Element child1 = new Element("child1");
        Element child2 = new Element("child2");
        root.addContent(child1);
        root.addContent(child2);

        JDOMNodePointer rootPointer = new JDOMNodePointer(root, locale);
        JDOMNodePointer child1Pointer = new JDOMNodePointer(rootPointer, child1);
        JDOMNodePointer child2Pointer = new JDOMNodePointer(rootPointer, child2);

        int result = rootPointer.compareChildNodePointers(child1Pointer, child2Pointer);
        assertEquals(-1, result);

        int resultReverse = rootPointer.compareChildNodePointers(child2Pointer, child1Pointer);
        assertEquals(1, resultReverse);
    }

    @Test
    public void testCompareChildNodePointers_attributeVsNonAttribute_returnsMinusOne() {
        Element root = new Element("root");
        root.setAttribute("attr1", "v1");
        Attribute attr1 = root.getAttribute("attr1");
        Element child = new Element("child");
        root.addContent(child);

        JDOMNodePointer rootPointer = new JDOMNodePointer(root, locale);
        JDOMNodePointer attrPointer = new JDOMNodePointer(rootPointer, attr1);
        JDOMNodePointer childPointer = new JDOMNodePointer(rootPointer, child);

        int result = rootPointer.compareChildNodePointers(attrPointer, childPointer);
        assertEquals(-1, result);

        int resultReverse = rootPointer.compareChildNodePointers(childPointer, attrPointer);
        assertEquals(1, resultReverse);
    }
