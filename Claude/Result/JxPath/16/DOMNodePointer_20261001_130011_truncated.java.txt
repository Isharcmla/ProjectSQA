public class DOMNodePointerTest {

    private Document doc;
    private Document nsDoc;
    private final Locale locale = Locale.US;

    @Before
    public void setUp() throws Exception {
        DocumentBuilder builder = DocumentBuilderFactory.newInstance().newDocumentBuilder();
        doc = builder.newDocument();

        DocumentBuilderFactory nsFactory = DocumentBuilderFactory.newInstance();
        nsFactory.setNamespaceAware(true);
        DocumentBuilder nsBuilder = nsFactory.newDocumentBuilder();
        nsDoc = nsBuilder.newDocument();
    }

    // ---------- Constructors ----------

    @Test
    public void testConstructor_NodeLocale_setsNode() {
        Element el = doc.createElement("root");
        DOMNodePointer ptr = new DOMNodePointer(el, locale);
        assertSame(el, ptr.getImmediateNode());
    }

    @Test
    public void testConstructor_NodeLocaleId_setsId() {
        Element el = doc.createElement("root");
        DOMNodePointer ptr = new DOMNodePointer(el, locale, "myid");
        assertEquals("id('myid')", ptr.asPath());
    }

    @Test
    public void testConstructor_ParentNode_setsParentAndNode() {
        Element parentEl = doc.createElement("parent");
        Element childEl = doc.createElement("child");
        parentEl.appendChild(childEl);
        DOMNodePointer parentPtr = new DOMNodePointer(parentEl, locale);
        DOMNodePointer childPtr = new DOMNodePointer(parentPtr, childEl);
        assertSame(childEl, childPtr.getImmediateNode());
    }
