    @Before
    public void setUp() throws Exception {
        DocumentBuilderFactory f1 = DocumentBuilderFactory.newInstance();
        f1.setNamespaceAware(false);
        nonNsBuilder = f1.newDocumentBuilder();

        DocumentBuilderFactory f2 = DocumentBuilderFactory.newInstance();
        f2.setNamespaceAware(true);
        nsBuilder = f2.newDocumentBuilder();
    }

    private Document newDocument() {
        return nonNsBuilder.newDocument();

    // ---------- Constructors ----------

    @Test
    public void testConstructor_NodeLocale_createsPointer() {
        Document doc = newDocument();
        Element e = doc.createElement("root");
        DOMNodePointer ptr = new DOMNodePointer(e, Locale.getDefault());
        assertSame(e, ptr.getBaseValue());
    }

    @Test
    public void testConstructor_NodeLocaleId_createsPointerWithId() {
        Document doc = newDocument();
        Element e = doc.createElement("root");
        DOMNodePointer ptr = new DOMNodePointer(e, Locale.getDefault(), "myid");
        assertEquals("id('myid')", ptr.asPath());
    }

    @Test
    public void testConstructor_ParentNode_createsChildPointer() {
        Document doc = newDocument();
        Element root = doc.createElement("root");
        doc.appendChild(root);
        DOMNodePointer parentPtr = new DOMNodePointer(root, Locale.getDefault());
        Element child = doc.createElement("child");
        root.appendChild(child);
        DOMNodePointer childPtr = new DOMNodePointer(parentPtr, child);
        assertSame(child, childPtr.getBaseValue());
    }
