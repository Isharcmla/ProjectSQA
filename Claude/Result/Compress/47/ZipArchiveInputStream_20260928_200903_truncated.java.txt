    // ---------- helper methods to build valid zip byte arrays using JDK classes ----------

    private byte[] createSingleEntryZip(String name, byte[] content, int method) throws IOException {
        ByteArrayOutputStream baos = new ByteArrayOutputStream();
        ZipOutputStream zos = new ZipOutputStream(baos);
        ZipEntry entry = new ZipEntry(name);
        if (method == ZipEntry.STORED) {
            entry.setMethod(ZipEntry.STORED);
            entry.setSize(content.length);
            CRC32 crc = new CRC32();
            crc.update(content);
            entry.setCrc(crc.getValue());
        } else {
            entry.setMethod(ZipEntry.DEFLATED);
        }
        zos.putNextEntry(entry);
        zos.write(content);
        zos.closeEntry();
        zos.finish();
        zos.close();
        return baos.toByteArray();
    }

    private byte[] createTwoEntryZip() throws IOException {
        ByteArrayOutputStream baos = new ByteArrayOutputStream();
        ZipOutputStream zos = new ZipOutputStream(baos);

        byte[] content1 = "Hello World".getBytes("UTF-8");
        ZipEntry e1 = new ZipEntry("first.txt");
        e1.setMethod(ZipEntry.DEFLATED);
        zos.putNextEntry(e1);
        zos.write(content1);
        zos.closeEntry();

        byte[] content2 = "ABCDEFGHIJ".getBytes("UTF-8");
        ZipEntry e2 = new ZipEntry("second.txt");
        e2.setMethod(ZipEntry.STORED);
        e2.setSize(content2.length);
        CRC32 crc = new CRC32();
        crc.update(content2);
        e2.setCrc(crc.getValue());
        zos.putNextEntry(e2);
        zos.write(content2);
        zos.closeEntry();

        zos.finish();
        zos.close();
        return baos.toByteArray();
    }

    private byte[] readAllFromZipArchiveInputStream(ZipArchiveInputStream zis) throws IOException {
        ByteArrayOutputStream out = new ByteArrayOutputStream();
        byte[] buf = new byte[1024];
        int r;
        while ((r = zis.read(buf, 0, buf.length)) > 0) {
            out.write(buf, 0, r);
        }
        return out.toByteArray();
    }

    @Test
    public void testGetNextZipEntry_deflatedEntry_returnsCorrectEntryAndData() throws IOException {
        byte[] content = "Hello World Compressed Content".getBytes("UTF-8");
        byte[] zipBytes = createSingleEntryZip("deflated.txt", content, ZipEntry.DEFLATED);

        ZipArchiveInputStream zis = new ZipArchiveInputStream(new ByteArrayInputStream(zipBytes));
        ZipArchiveEntry entry = zis.getNextZipEntry();

        assertNotNull(entry);
        assertEquals("deflated.txt", entry.getName());
        assertEquals(ZipEntry.DEFLATED, entry.getMethod());

        byte[] readContent = readAllFromZipArchiveInputStream(zis);
        assertArrayEquals(content, readContent);

        zis.close();
    }

    @Test
    public void testGetNextZipEntry_storedEntry_returnsCorrectEntryAndData() throws IOException {
        byte[] content = "STORED_CONTENT_1234567890".getBytes("UTF-8");
        byte[] zipBytes = createSingleEntryZip("stored.txt", content, ZipEntry.STORED);

        ZipArchiveInputStream zis = new ZipArchiveInputStream(new ByteArrayInputStream(zipBytes));
        ZipArchiveEntry entry = zis.getNextZipEntry();

        assertNotNull(entry);
        assertEquals("stored.txt", entry.getName());
        assertEquals(ZipEntry.STORED, entry.getMethod());

        byte[] readContent = readAllFromZipArchiveInputStream(zis);
        assertArrayEquals(content, readContent);

        zis.close();
    }

    @Test
    public void testGetNextZipEntry_multipleEntries_readsAllAndReturnsNullAtEnd() throws IOException {
        byte[] zipBytes = createTwoEntryZip();

        ZipArchiveInputStream zis = new ZipArchiveInputStream(new ByteArrayInputStream(zipBytes));
        int count = 0;
        ZipArchiveEntry e;
        while ((e = zis.getNextZipEntry()) != null) {
            count++;
            readAllFromZipArchiveInputStream(zis);
        }
        assertEquals(2, count);
        zis.close();
    }

    @Test
    public void testSkip_withPositiveValueOnStoredEntry_skipsCorrectBytes() throws IOException {
        byte[] content = "ABCDEFGHIJ".getBytes("UTF-8");
        byte[] zipBytes = createSingleEntryZip("skip_test.txt", content, ZipEntry.STORED);

        ZipArchiveInputStream zis = new ZipArchiveInputStream(new ByteArrayInputStream(zipBytes));
        ZipArchiveEntry entry = zis.getNextZipEntry();
        assertNotNull(entry);

        long skipped = zis.skip(3);
        assertEquals(3, skipped);

        byte[] buffer = new byte[10];
        int total = 0;
        int r;
        ByteArrayOutputStream out = new ByteArrayOutputStream();
        while ((r = zis.read(buffer, 0, buffer.length)) > 0) {
            out.write(buffer, 0, r);
            total += r;
        }
        assertEquals(7, total);
        assertArrayEquals("DEFGHIJ".getBytes("UTF-8"), out.toByteArray());
        zis.close();
    }
