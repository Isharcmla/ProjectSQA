package org.apache.commons.compress.archivers.tar;

import static org.junit.Assert.*;

import java.io.ByteArrayOutputStream;
import java.io.File;
import java.io.IOException;

import org.junit.Test;
import org.apache.commons.compress.archivers.ArchiveEntry;

public class TarArchiveOutputStreamTest {

    @Test
    public void testConstructor_defaultOutputStream_noException() throws IOException {
        ByteArrayOutputStream baos = new ByteArrayOutputStream();
        TarArchiveOutputStream tos = new TarArchiveOutputStream(baos);
        assertNotNull(tos);
        tos.close();
    }

    @Test
    public void testConstructor_withEncoding_noException() throws IOException {
        ByteArrayOutputStream baos = new ByteArrayOutputStream();
        TarArchiveOutputStream tos = new TarArchiveOutputStream(baos, "UTF-8");
        assertNotNull(tos);
        tos.close();
    }

    @Test
    public void testConstructor_withBlockSize_noException() throws IOException {
        ByteArrayOutputStream baos = new ByteArrayOutputStream();
        TarArchiveOutputStream tos = new TarArchiveOutputStream(baos, 10240);
        assertNotNull(tos);
        tos.close();
    }

    @Test
    public void testConstructor_withBlockSizeAndEncoding_noException() throws IOException {
        ByteArrayOutputStream baos = new ByteArrayOutputStream();
        TarArchiveOutputStream tos = new TarArchiveOutputStream(baos, 10240, "UTF-8");
        assertNotNull(tos);
        tos.close();
    }

    @Test
    public void testConstructor_withBlockSizeAndRecordSize_noException() throws IOException {
        ByteArrayOutputStream baos = new ByteArrayOutputStream();
        TarArchiveOutputStream tos = new TarArchiveOutputStream(baos, 10240, 512);
        assertNotNull(tos);
        tos.close();
    }

    @Test
    public void testConstructor_withBlockSizeRecordSizeAndEncoding_noException() throws IOException {
        ByteArrayOutputStream baos = new ByteArrayOutputStream();
        TarArchiveOutputStream tos = new TarArchiveOutputStream(baos, 10240, 512, "UTF-8");
        assertNotNull(tos);
        tos.close();
    }

    @Test
    public void testSetLongFileMode_setsWithoutException() {
        ByteArrayOutputStream baos = new ByteArrayOutputStream();
        TarArchiveOutputStream tos = new TarArchiveOutputStream(baos);
        tos.setLongFileMode(TarArchiveOutputStream.LONGFILE_GNU);
        assertTrue(true);
    }

    @Test
    public void testSetBigNumberMode_setsWithoutException() {
        ByteArrayOutputStream baos = new ByteArrayOutputStream();
        TarArchiveOutputStream tos = new TarArchiveOutputStream(baos);
        tos.setBigNumberMode(TarArchiveOutputStream.BIGNUMBER_POSIX);
        assertTrue(true);
    }

    @Test
    public void testSetAddPaxHeadersForNonAsciiNames_setsWithoutException() {
        ByteArrayOutputStream baos = new ByteArrayOutputStream();
        TarArchiveOutputStream tos = new TarArchiveOutputStream(baos);
        tos.setAddPaxHeadersForNonAsciiNames(true);
        assertTrue(true);
    }

    @Test
    public void testGetRecordSize_defaultValue() {
        ByteArrayOutputStream baos = new ByteArrayOutputStream();
        TarArchiveOutputStream tos = new TarArchiveOutputStream(baos);
        assertEquals(512, tos.getRecordSize());
    }

    @Test
    public void testGetBytesWritten_afterPutAndCloseEntry_returnsPositiveValue() throws IOException {
        ByteArrayOutputStream baos = new ByteArrayOutputStream();
        TarArchiveOutputStream tos = new TarArchiveOutputStream(baos);
        TarArchiveEntry entry = new TarArchiveEntry("test.txt");
        entry.setSize(4);
        tos.putArchiveEntry(entry);
        tos.write("test".getBytes());
        tos.closeArchiveEntry();
        tos.finish();
        assertTrue(tos.getBytesWritten() > 0);
        tos.close();
    }

    @Test
    public void testGetCount_deprecated_matchesBytesWritten() throws IOException {
        ByteArrayOutputStream baos = new ByteArrayOutputStream();
        TarArchiveOutputStream tos = new TarArchiveOutputStream(baos);
        TarArchiveEntry entry = new TarArchiveEntry("test.txt");
        entry.setSize(4);
        tos.putArchiveEntry(entry);
        tos.write("test".getBytes());
        tos.closeArchiveEntry();
        tos.finish();
        assertEquals((int) tos.getBytesWritten(), tos.getCount());
        tos.close();
    }

    @Test
    public void testPutArchiveEntry_normalFile_noException() throws IOException {
        ByteArrayOutputStream baos = new ByteArrayOutputStream();
        TarArchiveOutputStream tos = new TarArchiveOutputStream(baos);
        TarArchiveEntry entry = new TarArchiveEntry("normal.txt");
        entry.setSize(5);
        tos.putArchiveEntry(entry);
        tos.write("hello".getBytes());
        tos.closeArchiveEntry();
        tos.finish();
        tos.close();
        assertTrue(baos.size() > 0);
    }

    @Test
    public void testPutArchiveEntry_directoryEntry_currSizeZero() throws IOException {
        ByteArrayOutputStream baos = new ByteArrayOutputStream();
        TarArchiveOutputStream tos = new TarArchiveOutputStream(baos);
        TarArchiveEntry entry = new TarArchiveEntry("dir/");
        tos.putArchiveEntry(entry);
        tos.closeArchiveEntry();
        tos.finish();
        tos.close();
        assertTrue(baos.size() > 0);
    }
