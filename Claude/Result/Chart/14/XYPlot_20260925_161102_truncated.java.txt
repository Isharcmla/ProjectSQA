public class XYPlotTest {

    private XYPlot plot;
    private NumberAxis domainAxis;
    private NumberAxis rangeAxis;
    private XYSeriesCollection dataset;
    private XYLineAndShapeRenderer renderer;

    /** Simple listener to track change events for notify-flag tests. */
    private static class TestPlotChangeListener implements PlotChangeListener {
        int count = 0;
        public void plotChanged(PlotChangeEvent event) {
            count++;
        }
    }

    @Before
    public void setUp() {
        domainAxis = new NumberAxis("X");
        rangeAxis = new NumberAxis("Y");
        XYSeries series = new XYSeries("Series1");
        series.add(1.0, 10.0);
        series.add(2.0, 20.0);
        series.add(3.0, 5.0);
        dataset = new XYSeriesCollection();
        dataset.addSeries(series);
        renderer = new XYLineAndShapeRenderer();
        plot = new XYPlot(dataset, domainAxis, rangeAxis, renderer);
    }

    // ---------- Constructors ----------

    @Test
    public void testDefaultConstructor_createsEmptyPlot() {
        XYPlot p = new XYPlot();
        assertNull(p.getDataset());
        assertNull(p.getRenderer());
    }

    @Test
    public void testFullConstructor_setsFieldsCorrectly() {
        assertSame(dataset, plot.getDataset());
        assertSame(domainAxis, plot.getDomainAxis());
        assertSame(rangeAxis, plot.getRangeAxis());
        assertSame(renderer, plot.getRenderer());
    }

    @Test
    public void testSetDomainAxis_notifyFalse_doesNotFireEvent() {
        TestPlotChangeListener listener = new TestPlotChangeListener();
        plot.addChangeListener(listener);
        NumberAxis newAxis = new NumberAxis("Silent");
        plot.setDomainAxis(0, newAxis, false);
        assertEquals(0, listener.count);
    }

    @Test
    public void testSetDomainAxis_notifyTrue_firesEvent() {
        TestPlotChangeListener listener = new TestPlotChangeListener();
        plot.addChangeListener(listener);
        NumberAxis newAxis = new NumberAxis("Loud");
        plot.setDomainAxis(0, newAxis, true);
        assertTrue(listener.count > 0);
    }
