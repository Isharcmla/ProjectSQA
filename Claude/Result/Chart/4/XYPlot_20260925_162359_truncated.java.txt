    @Before
    public void setUp() {
        domainAxis = new NumberAxis("X");
        rangeAxis = new NumberAxis("Y");
        dataset = new XYSeriesCollection();
        XYSeries series = new XYSeries("S1");
        series.add(1.0, 2.0);
        series.add(2.0, 3.0);
        series.add(3.0, 1.0);
        dataset.addSeries(series);
        renderer = new XYLineAndShapeRenderer();
        plot = new XYPlot(dataset, domainAxis, rangeAxis, renderer);
    }

    public void testAddDomainMarker_addsMarkerSuccessfully() {
        ValueMarker marker = new ValueMarker(1.5);
        plot.addDomainMarker(marker);
        assertTrue(plot.getDomainMarkers(Layer.FOREGROUND).contains(marker));
    }

    @Test(expected = IllegalArgumentException.class)
    public void testAddDomainMarker_null_throwsException() {
        plot.addDomainMarker(null);
    }

    @Test
    public void testAddDomainMarkerBackgroundLayer_addsMarker() {
        ValueMarker marker = new ValueMarker(2.5);
        plot.addDomainMarker(marker, Layer.BACKGROUND);
        assertTrue(plot.getDomainMarkers(Layer.BACKGROUND).contains(marker));
    }

    @Test
    public void testClearDomainMarkers_removesAllMarkers() {
        ValueMarker marker = new ValueMarker(1.0);
        plot.addDomainMarker(marker);
        plot.clearDomainMarkers();
        assertTrue(plot.getDomainMarkers(Layer.FOREGROUND) == null
                || plot.getDomainMarkers(Layer.FOREGROUND).isEmpty());
    }

    @Test
    public void testClearDomainMarkersByIndex_removesMarkersForIndex() {
        ValueMarker marker = new ValueMarker(1.0);
        plot.addDomainMarker(0, marker, Layer.FOREGROUND);
        plot.clearDomainMarkers(0);
        Object result = plot.getDomainMarkers(0, Layer.FOREGROUND);
        assertTrue(result == null || ((java.util.Collection) result).isEmpty());
    }

    @Test
    public void testAddDomainMarkerByIndexNoNotify_doesNotThrow() {
        ValueMarker marker = new ValueMarker(1.0);
        plot.addDomainMarker(0, marker, Layer.FOREGROUND, false);
        assertTrue(plot.getDomainMarkers(0, Layer.FOREGROUND).contains(marker));
    }
