package org.mockito.internal.creation.bytebuddy;

import static org.junit.Assert.assertFalse;
import static org.junit.Assert.assertNotNull;
import static org.junit.Assert.assertNull;
import static org.junit.Assert.assertTrue;

import java.util.HashSet;
import java.util.Set;

import org.junit.Before;
import org.junit.Test;
import org.mockito.exceptions.base.MockitoException;
import org.mockito.internal.creation.settings.CreationSettings;
import org.mockito.internal.handler.MockHandlerImpl;
import org.mockito.invocation.Invocation;
import org.mockito.invocation.MockHandler;
import org.mockito.mock.MockCreationSettings;
import org.mockito.mock.SerializableMode;

/**
 * Note: CreationSettings (org.mockito.internal.creation.settings.CreationSettings)
 * and MockHandlerImpl (org.mockito.internal.handler.MockHandlerImpl) are real,
 * existing internal Mockito classes used here only as concrete implementations
 * of MockCreationSettings / InternalMockHandler (both of which are interfaces
 * that cannot otherwise be instantiated without a mocking framework). This is
 * not a mocking framework usage - these are genuine production classes from
 * the Mockito codebase being tested.
 */
public class ByteBuddyMockMakerTest {

    private ByteBuddyMockMaker mockMaker;

    public static class SampleClass {
        public String foo() {
            return "foo";
        }
    }

    public interface MarkerInterface {
    }

    @Before
    public void setUp() {
        mockMaker = new ByteBuddyMockMaker();
    }

    private <T> CreationSettings<T> createSettings(Class<T> typeToMock) {
        CreationSettings<T> settings = new CreationSettings<T>();
        settings.setTypeToMock(typeToMock);
        settings.setExtraInterfaces(new HashSet<Class<?>>());
        settings.setSerializableMode(SerializableMode.NONE);
        return settings;
    }

    @Test
    public void testCreateMock_normalClass_returnsNonNullMockInstance() {
        CreationSettings<SampleClass> settings = createSettings(SampleClass.class);
        MockHandler<SampleClass> handler = new MockHandlerImpl<SampleClass>(settings);

        SampleClass mock = mockMaker.createMock(settings, handler);

        assertNotNull(mock);
        assertTrue(mock instanceof SampleClass);
    }

    @Test
    public void testCreateMock_withInterfaceType_returnsMockOfInterface() {
        CreationSettings<Runnable> settings = createSettings(Runnable.class);
        MockHandler<Runnable> handler = new MockHandlerImpl<Runnable>(settings);

        Runnable mock = mockMaker.createMock(settings, handler);

        assertNotNull(mock);
        assertTrue(mock instanceof Runnable);
    }

    @Test
    public void testCreateMock_withExtraInterfaces_returnsMockImplementingInterface() {
        CreationSettings<SampleClass> settings = createSettings(SampleClass.class);
        Set<Class<?>> extraInterfaces = new HashSet<Class<?>>();
        extraInterfaces.add(MarkerInterface.class);
        settings.setExtraInterfaces(extraInterfaces);
        MockHandler<SampleClass> handler = new MockHandlerImpl<SampleClass>(settings);

        SampleClass mock = mockMaker.createMock(settings, handler);

        assertNotNull(mock);
        assertTrue(mock instanceof MarkerInterface);
    }

    @Test(expected = MockitoException.class)
    public void testCreateMock_withAcrossClassloadersSerializableMode_throwsMockitoException() {
        CreationSettings<SampleClass> settings = createSettings(SampleClass.class);
        settings.setSerializableMode(SerializableMode.ACROSS_CLASSLOADERS);
        MockHandler<SampleClass> handler = new MockHandlerImpl<SampleClass>(settings);

        mockMaker.createMock(settings, handler);
    }

    @Test(expected = MockitoException.class)
    public void testCreateMock_withNonInternalMockHandler_throwsMockitoException() {
        final CreationSettings<SampleClass> settings = createSettings(SampleClass.class);
        MockHandler<SampleClass> handler = new MockHandler<SampleClass>() {
            public Object handle(Invocation invocation) throws Throwable {
                return null;
            }

            public MockCreationSettings<SampleClass> getMockSettings() {
                return settings;
            }
        };

        mockMaker.createMock(settings, handler);
    }
