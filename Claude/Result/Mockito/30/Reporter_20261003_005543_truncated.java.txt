 * PrintableInvocation (used as parameter type) and Discrepancy are both

 * their required contract (methods used inside Reporter's source: 
 * toString(), getLocation(), getPluralizedWantedCount(),
 * getPluralizedActualCount(), constructor Discrepancy(int, int)) is directly

public class ReporterTest {

    private Reporter reporter = new Reporter();

    private static class TestInvocation implements PrintableInvocation {
        private final String description;
        private final Location location;

        TestInvocation(String description) {
            this.description = description;
            this.location = new Location();
        }

        @Override
        public String toString() {
            return description;
        }

        @Override
        public Location getLocation() {
            return location;
        }
    }

    @Test(expected = MockitoException.class)
    public void testCheckedExceptionInvalid_normalInput_throwsMockitoException() {
        reporter.checkedExceptionInvalid(new Throwable("custom error"));
    }

    @Test
    public void testCheckedExceptionInvalid_nullThrowable_messageContainsNull() {
        try {
            reporter.checkedExceptionInvalid(null);
            fail("Expected MockitoException");
        } catch (MockitoException e) {
            assertTrue(e.getMessage().contains("Invalid: null"));
        }
    }

    @Test(expected = MockitoException.class)

        try {
            reporter.unfinishedStubbing(null);
            fail("Expected UnfinishedStubbingException");
        } catch (UnfinishedStubbingException e) {
            assertNotNull(e.getMessage());
        }
    }
