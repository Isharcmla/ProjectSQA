  @Test
  public void testGetNearestFunctionName_variousNodes_returnsExpectedName() {
    Assert.assertNull(NodeUtil.getNearestFunctionName(IR.name("notFn")));

    // function foo() {}
    Node fnNamed = IR.function(IR.name("foo"), IR.paramList(), IR.block());
    Assert.assertEquals("foo", NodeUtil.getNearestFunctionName(fnNamed));

    // { 'prop': function() {} }
    Node fnInKey = IR.function(IR.name(""), IR.paramList(), IR.block());
    Node strKey = IR.stringKey("prop", fnInKey);
    Assert.assertEquals("prop", NodeUtil.getNearestFunctionName(fnInKey));

    // { get getter() {} }
    Node fnInGetter = IR.function(IR.name(""), IR.paramList(), IR.block());
    Node getter = IR.getterDef("getter", fnInGetter);
    Assert.assertEquals("getter", NodeUtil.getNearestFunctionName(fnInGetter));

    // { set setter(v) {} }
    Node fnInSetter = IR.function(IR.name(""), IR.paramList(IR.name("v")), IR.block());
    Node setter = IR.setterDef("setter", fnInSetter);
    Assert.assertEquals("setter", NodeUtil.getNearestFunctionName(fnInSetter));

    // { 42: function() {} }
    Node fnInNum = IR.function(IR.name(""), IR.paramList(), IR.block());
    Node num = IR.number(42);
    num.addChildToBack(fnInNum);
    Assert.assertEquals("42", NodeUtil.getNearestFunctionName(fnInNum));

    // Parent is an unsupported token
    Node fnInBlock = IR.function(IR.name(""), IR.paramList(), IR.block());
    IR.block(fnInBlock);
    Assert.assertNull(NodeUtil.getNearestFunctionName(fnInBlock));
  }

  @Test
  public void testIsLValue_variousScenarios_returnsCorrectBoolean() {
    Node name = IR.name("x");
    Assert.assertFalse(NodeUtil.isLValue(name));

    // Var declaration: var x
    IR.var(name);
    Assert.assertTrue(NodeUtil.isLValue(name));

    // Assignment: x = 1
    Node assignName = IR.name("a");
    IR.assign(assignName, IR.number(1));
    Assert.assertTrue(NodeUtil.isLValue(assignName));

    // For-in loop target: for (x in y)
    Node forInName = IR.name("b");
    IR.forIn(forInName, IR.name("obj"), IR.block());
    Assert.assertTrue(NodeUtil.isLValue(forInName));

    // Function name: function f() {}
    Node fnName = IR.name("f");
    IR.function(fnName, IR.paramList(), IR.block());
    Assert.assertTrue(NodeUtil.isLValue(fnName));

    // Increment/decrement: ++x, --x
    Node incName = IR.name("c");
    IR.inc(incName, 0);
    Assert.assertTrue(NodeUtil.isLValue(incName));

    Node decName = IR.name("d");
    IR.dec(decName, 0);
    Assert.assertTrue(NodeUtil.isLValue(decName));

    // Parameter list: function(p) {}
    Node paramName = IR.name("p");
    IR.paramList(paramName);
    Assert.assertTrue(NodeUtil.isLValue(paramName));

    // Catch block: catch (e) {}
    Node catchName = IR.name("e");
    IR.catchNode(catchName, IR.block());
    Assert.assertTrue(NodeUtil.isLValue(catchName));

    // GETPROP and GETELEM as L-values
    Node getProp = IR.getprop(IR.name("obj"), IR.string("prop"));
    IR.assign(getProp, IR.number(2));
    Assert.assertTrue(NodeUtil.isLValue(getProp));

    Node getElem = IR.getelem(IR.name("arr"), IR.number(0));
    IR.assign(getElem, IR.number(3));
    Assert.assertTrue(NodeUtil.isLValue(getElem));
  }

  @Test
  public void testGetFunctionJSDocInfo_directAndAssignedFunctions_returnsJSDoc() {
    JSDocInfo jsDoc = new JSDocInfoBuilder(true).build(null);

    // Direct function with JSDoc
    Node fn = IR.function(IR.name("foo"), IR.paramList(), IR.block());
    fn.setJSDocInfo(jsDoc);
    Assert.assertSame(jsDoc, NodeUtil.getFunctionJSDocInfo(fn));

    // Function expression assigned to a var: var f = function() {};
    Node fnExpr = IR.function(IR.name(""), IR.paramList(), IR.block());
    Node nameNode = IR.name("f");
    nameNode.addChildToBack(fnExpr);
    Node varNode = IR.var(nameNode);
    varNode.setJSDocInfo(jsDoc);
    Assert.assertSame(jsDoc, NodeUtil.getFunctionJSDocInfo(fnExpr));

    // Function expression in assign: f = function() {};
    Node fnExpr2 = IR.function(IR.name(""), IR.paramList(), IR.block());
    Node assignNode = IR.assign(IR.name("f"), fnExpr2);
    assignNode.setJSDocInfo(jsDoc);
    Assert.assertSame(jsDoc, NodeUtil.getFunctionJSDocInfo(fnExpr2));
  }

  @Test
  public void testMayEffectMutableState_andMayHaveSideEffects_evaluatesCorrectly() {
    Node num = IR.number(1);
    Assert.assertFalse(NodeUtil.mayEffectMutableState(num));
    Assert.assertFalse(NodeUtil.mayHaveSideEffects(num));

    Node throwNode = IR.throwNode(IR.string("err"));
    Assert.assertTrue(NodeUtil.mayEffectMutableState(throwNode));
    Assert.assertTrue(NodeUtil.mayHaveSideEffects(throwNode));

    Node objLit = IR.objectlit();
    Assert.assertTrue(NodeUtil.mayEffectMutableState(objLit));
    Assert.assertFalse(NodeUtil.mayHaveSideEffects(objLit));

    Node arrLit = IR.arraylit();
