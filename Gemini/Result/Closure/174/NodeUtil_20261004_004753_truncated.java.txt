  @Test
  public void testGetImpureBooleanValue() {
    assertEquals(TernaryValue.TRUE, NodeUtil.getImpureBooleanValue(IR.string("hello")));
    assertEquals(TernaryValue.FALSE, NodeUtil.getImpureBooleanValue(IR.string("")));
    assertEquals(TernaryValue.TRUE, NodeUtil.getImpureBooleanValue(IR.number(1.0)));
    assertEquals(TernaryValue.FALSE, NodeUtil.getImpureBooleanValue(IR.number(0.0)));
    assertEquals(TernaryValue.TRUE, NodeUtil.getImpureBooleanValue(IR.arraylit()));
    assertEquals(TernaryValue.TRUE, NodeUtil.getImpureBooleanValue(IR.objectlit()));
    assertEquals(TernaryValue.FALSE, NodeUtil.getImpureBooleanValue(IR.voidNode(IR.number(0))));

    Node assign = IR.assign(IR.name("a"), IR.number(1));
    assertEquals(TernaryValue.TRUE, NodeUtil.getImpureBooleanValue(assign));

    Node comma = IR.comma(IR.name("a"), IR.number(0));
    assertEquals(TernaryValue.FALSE, NodeUtil.getImpureBooleanValue(comma));

    Node notTrue = IR.not(IR.trueNode());
    assertEquals(TernaryValue.FALSE, NodeUtil.getImpureBooleanValue(notTrue));

    Node andNode = IR.and(IR.trueNode(), IR.trueNode());
    assertEquals(TernaryValue.TRUE, NodeUtil.getImpureBooleanValue(andNode));

    Node orNode = IR.or(IR.falseNode(), IR.trueNode());
    assertEquals(TernaryValue.TRUE, NodeUtil.getImpureBooleanValue(orNode));

    Node hookSame = IR.hook(IR.name("c"), IR.trueNode(), IR.trueNode());
    assertEquals(TernaryValue.TRUE, NodeUtil.getImpureBooleanValue(hookSame));

    Node hookDiff = IR.hook(IR.name("c"), IR.trueNode(), IR.falseNode());
    assertEquals(TernaryValue.UNKNOWN, NodeUtil.getImpureBooleanValue(hookDiff));
  }

  @Test
  public void testGetFunctionNameAndNearest() {
    Node fnNamed = IR.function(IR.name("foo"), IR.paramList(), IR.block());
    assertEquals("foo", NodeUtil.getFunctionName(fnNamed));
    assertEquals("foo", NodeUtil.getNearestFunctionName(fnNamed));

    Node varAssign = IR.var(IR.name("bar"), IR.function(IR.name(""), IR.paramList(), IR.block()));
    Node fnInVar = varAssign.getFirstChild().getFirstChild();
    assertEquals("bar", NodeUtil.getFunctionName(fnInVar));
    assertEquals("bar", NodeUtil.getNearestFunctionName(fnInVar));

    Node assignExpr = IR.assign(IR.name("baz"), IR.function(IR.name(""), IR.paramList(), IR.block()));
    Node fnInAssign = assignExpr.getLastChild();
    assertEquals("baz", NodeUtil.getFunctionName(fnInAssign));
    assertEquals("baz", NodeUtil.getNearestFunctionName(fnInAssign));

    assertNull(NodeUtil.getNearestFunctionName(IR.number(1)));

    Node objLit = IR.objectlit();
    Node key = IR.stringKey("prop", IR.function(IR.name(""), IR.paramList(), IR.block()));
    objLit.addChildToBack(key);
    assertEquals("prop", NodeUtil.getNearestFunctionName(key.getFirstChild()));

    Node getterKey = IR.getterDef("getter", IR.function(IR.name(""), IR.paramList(), IR.block()));
    assertEquals("getter", NodeUtil.getNearestFunctionName(getterKey.getFirstChild()));

    Node setterKey = IR.setterDef("setter", IR.function(IR.name(""), IR.paramList(), IR.block()));
    assertEquals("setter", NodeUtil.getNearestFunctionName(setterKey.getFirstChild()));

    Node numKey = IR.number(42);
    Node numFn = IR.function(IR.name(""), IR.paramList(), IR.block());
    numKey.addChildToBack(numFn);
    assertEquals("42", NodeUtil.getNearestFunctionName(numFn));
  }

  @Test
  public void testConstructorAndFunctionCallSideEffects() {
    Node newArray = IR.newNode(IR.name("Array"));
    assertFalse(NodeUtil.constructorCallHasSideEffects(newArray));

    Node newCustom = IR.newNode(IR.name("CustomClass"));
    assertTrue(NodeUtil.constructorCallHasSideEffects(newCustom));

    Node newNoSideEffects = IR.newNode(IR.name("CustomClass"));

    Node callMathFloor = IR.call(IR.getprop(IR.name("Math"), IR.string("floor")), IR.number(1.5));
    assertFalse(NodeUtil.functionCallHasSideEffects(callMathFloor));

    Node callCustom = IR.call(IR.name("customFn"));
    assertTrue(NodeUtil.functionCallHasSideEffects(callCustom));

    Node callToString = IR.call(IR.getprop(IR.name("obj"), IR.string("toString")));
    assertFalse(NodeUtil.functionCallHasSideEffects(callToString));

    try {
      NodeUtil.constructorCallHasSideEffects(IR.call(IR.name("x")));
      fail("Expected exception");
    } catch (IllegalStateException expected) {}

    try {
      NodeUtil.functionCallHasSideEffects(IR.newNode(IR.name("x")));
      fail("Expected exception");
    } catch (IllegalStateException expected) {}
  }

  @Test
  public void testNodeTypeMayHaveSideEffectsAndCanBeSideEffected() {
    assertTrue(NodeUtil.nodeTypeMayHaveSideEffects(IR.delprop(IR.name("a"), IR.string("b"))));
    assertTrue(NodeUtil.nodeTypeMayHaveSideEffects(IR.dec(IR.name("a"), false)));
    assertTrue(NodeUtil.nodeTypeMayHaveSideEffects(IR.inc(IR.name("a"), false)));
    assertTrue(NodeUtil.nodeTypeMayHaveSideEffects(IR.throwNode(IR.string("e"))));
    assertFalse(NodeUtil.nodeTypeMayHaveSideEffects(IR.number(1)));

    Node varName = IR.name("x");
    varName.addChildToBack(IR.number(1));
    assertTrue(NodeUtil.nodeTypeMayHaveSideEffects(varName));

    assertTrue(NodeUtil.canBeSideEffected(IR.call(IR.name("f"))));
    assertTrue(NodeUtil.canBeSideEffected(IR.newNode(IR.name("f"))));
    assertTrue(NodeUtil.canBeSideEffected(IR.getprop(IR.name("a"), IR.string("b"))));
    assertTrue(NodeUtil.canBeSideEffected(IR.getelem(IR.name("a"), IR.string("b"))));

    Node nonConstName = IR.name("v");
    assertTrue(NodeUtil.canBeSideEffected(nonConstName));
    assertFalse(NodeUtil.canBeSideEffected(nonConstName, Collections.singleton("v")));

    Node constName = IR.name("CONST");
    constName.putBooleanProp(Node.IS_CONSTANT_NAME, true);
    assertFalse(NodeUtil.canBeSideEffected(constName));
  }
