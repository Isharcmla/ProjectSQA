    @Test
    public void testIsEquals() {
        assertTrue(ArrayUtils.isEquals(null, null));
        assertFalse(ArrayUtils.isEquals(null, new int[]{1}));
        assertFalse(ArrayUtils.isEquals(new int[]{1}, null));
        assertTrue(ArrayUtils.isEquals(new int[]{1, 2}, new int[]{1, 2}));

        assertTrue(ArrayUtils.isEquals(new String[][]{{"a"}}, new String[][]{{"a"}}));
    }

    @Test
    public void testToMap_validInputs() {
        assertNull(ArrayUtils.toMap(null));

        Map<Object, Object> emptyMap = ArrayUtils.toMap(new Object[0]);
        assertTrue(emptyMap.isEmpty());

        Map.Entry<String, String> entry = new AbstractMap.SimpleEntry<String, String>("k1", "v1");
        Object[] arrayWithEntry = new Object[]{entry};
        Map<Object, Object> map1 = ArrayUtils.toMap(arrayWithEntry);
        assertEquals(1, map1.size());
        assertEquals("v1", map1.get("k1"));

        Object[] arrayWithSubArray = new Object[]{
                new String[]{"k2", "v2"},
                new String[]{"k3", "v3", "extra"}
        };
        Map<Object, Object> map2 = ArrayUtils.toMap(arrayWithSubArray);
        assertEquals(2, map2.size());
        assertEquals("v2", map2.get("k2"));
        assertEquals("v3", map2.get("k3"));
    }

    @Test
    public void testClone_objectsAndPrimitives() {
        assertNull(ArrayUtils.clone((Object[]) null));
        String[] sArr = new String[]{"a", "b"};
        assertArrayEquals(sArr, ArrayUtils.clone(sArr));
        assertNotSame(sArr, ArrayUtils.clone(sArr));

        assertNull(ArrayUtils.clone((long[]) null));
        long[] lArr = new long[]{1L, 2L};
        assertArrayEquals(lArr, ArrayUtils.clone(lArr));

        assertNull(ArrayUtils.clone((int[]) null));
        int[] iArr = new int[]{1, 2};
        assertArrayEquals(iArr, ArrayUtils.clone(iArr));

        assertNull(ArrayUtils.clone((short[]) null));
        short[] shArr = new short[]{1, 2};
        assertArrayEquals(shArr, ArrayUtils.clone(shArr));

        assertNull(ArrayUtils.clone((char[]) null));
        char[] cArr = new char[]{'a', 'b'};
        assertArrayEquals(cArr, ArrayUtils.clone(cArr));

        assertNull(ArrayUtils.clone((byte[]) null));
        byte[] bArr = new byte[]{1, 2};
        assertArrayEquals(bArr, ArrayUtils.clone(bArr));

        assertNull(ArrayUtils.clone((double[]) null));
        double[] dArr = new double[]{1.0, 2.0};
        assertArrayEquals(dArr, ArrayUtils.clone(dArr), 0.0);

        assertNull(ArrayUtils.clone((float[]) null));
        float[] fArr = new float[]{1.0f, 2.0f};
        assertArrayEquals(fArr, ArrayUtils.clone(fArr), 0.0f);

        assertNull(ArrayUtils.clone((boolean[]) null));
        boolean[] boolArr = new boolean[]{true, false};
        assertEquals(boolArr.length, ArrayUtils.clone(boolArr).length);
        assertTrue(ArrayUtils.clone(boolArr)[0]);
        assertFalse(ArrayUtils.clone(boolArr)[1]);
    }

    @Test

        assertArrayEquals(new long[]{2L, 3L}, ArrayUtils.subarray(new long[]{1L, 2L, 3L, 4L}, 1, 3));
        assertArrayEquals(new long[]{1L}, ArrayUtils.subarray(new long[]{1L, 2L}, -1, 1));
        assertArrayEquals(new long[]{2L}, ArrayUtils.subarray(new long[]{1L, 2L}, 1, 10));
        assertArrayEquals(new long[0], ArrayUtils.subarray(new long[]{1L, 2L}, 2, 1));

        assertNull(ArrayUtils.subarray((int[]) null, 0, 1));
        assertArrayEquals(new int[]{2, 3}, ArrayUtils.subarray(new int[]{1, 2, 3, 4}, 1, 3));

        assertArrayEquals(new short[]{1}, ArrayUtils.subarray(new short[]{1, 2}, -1, 1));
        assertArrayEquals(new short[]{2}, ArrayUtils.subarray(new short[]{1, 2}, 1, 10));
        assertArrayEquals(new short[0], ArrayUtils.subarray(new short[]{1, 2}, 2, 1));

        assertNull(ArrayUtils.subarray((char[]) null, 0, 1));
        assertArrayEquals(new char[]{'b', 'c'}, ArrayUtils.subarray(new char[]{'a', 'b', 'c', 'd'}, 1, 3));
        assertArrayEquals(new char[]{'a'}, ArrayUtils.subarray(new char[]{'a', 'b'}, -1, 1));
        assertArrayEquals(new char[]{'b'}, ArrayUtils.subarray(new char[]{'a', 'b'}, 1, 10));
        assertArrayEquals(new char[0], ArrayUtils.subarray(new char[]{'a', 'b'}, 2, 1));

        assertNull(ArrayUtils.subarray((byte[]) null, 0, 1));
        assertArrayEquals(new byte[]{2, 3}, ArrayUtils.subarray(new byte[]{1, 2, 3, 4}, 1, 3));
        assertArrayEquals(new byte[]{1}, ArrayUtils.subarray(new byte[]{1, 2}, -1, 1));
        assertArrayEquals(new byte[]{2}, ArrayUtils.subarray(new byte[]{1, 2}, 1, 10));
        assertArrayEquals(new byte[0], ArrayUtils.subarray(new byte[]{1, 2}, 2, 1));

        assertNull(ArrayUtils.subarray((double[]) null, 0, 1));
        assertArrayEquals(new double[]{2.0, 3.0}, ArrayUtils.subarray(new double[]{1.0, 2.0, 3.0, 4.0}, 1, 3), 0.0);
        assertArrayEquals(new double[]{1.0}, ArrayUtils.subarray(new double[]{1.0, 2.0}, -1, 1), 0.0);
        assertArrayEquals(new double[]{2.0}, ArrayUtils.subarray(new double[]{1.0, 2.0}, 1, 10), 0.0);
        assertArrayEquals(new double[0], ArrayUtils.subarray(new double[]{1.0, 2.0}, 2, 1), 0.0);

        assertNull(ArrayUtils.subarray((float[]) null, 0, 1));

        assertArrayEquals(new float[]{1.0f}, ArrayUtils.subarray(new float[]{1.0f, 2.0f}, -1, 1), 0.0f);
        assertArrayEquals(new float[]{2.0f}, ArrayUtils.subarray(new float[]{1.0f, 2.0f}, 1, 10), 0.0f);
        assertArrayEquals(new float[0], ArrayUtils.subarray(new float[]{1.0f, 2.0f}, 2, 1), 0.0f);

        assertNull(ArrayUtils.subarray((boolean[]) null, 0, 1));

        assertEquals(2, bSub.length);
        assertFalse(bSub[0]);
        assertTrue(bSub[1]);
        assertEquals(1, ArrayUtils.subarray(new boolean[]{true, false}, -1, 1).length);
        assertEquals(1, ArrayUtils.subarray(new boolean[]{true, false}, 1, 10).length);
        assertEquals(0, ArrayUtils.subarray(new boolean[]{true, false}, 2, 1).length);

    @Test

    @Test
    public void testReverse() {
        ArrayUtils.reverse((Object[]) null);
        String[] sArr = new String[]{"a", "b", "c"};
        ArrayUtils.reverse(sArr);
        assertArrayEquals(new String[]{"c", "b", "a"}, sArr);

        ArrayUtils.reverse((long[]) null);
        long[] lArr = new long[]{1L, 2L, 3L};
        ArrayUtils.reverse(lArr);
        assertArrayEquals(new long[]{3L, 2L, 1L}, lArr);

        ArrayUtils.reverse((int[]) null);
        int[] iArr = new int[]{1, 2, 3};
        ArrayUtils.reverse(iArr);
        assertArrayEquals(new int[]{3, 2, 1}, iArr);

        ArrayUtils.reverse((short[]) null);
        short[] shArr = new short[]{1, 2, 3};
        ArrayUtils.reverse(shArr);
        assertArrayEquals(new short[]{3, 2, 1}, shArr);

        ArrayUtils.reverse((char[]) null);
        char[] cArr = new char[]{'a', 'b', 'c'};
        ArrayUtils.reverse(cArr);
        assertArrayEquals(new char[]{'c', 'b', 'a'}, cArr);

        ArrayUtils.reverse((byte[]) null);
        byte[] bArr = new byte[]{1, 2, 3};
        ArrayUtils.reverse(bArr);
        assertArrayEquals(new byte[]{3, 2, 1}, bArr);

        ArrayUtils.reverse((double[]) null);
        double[] dArr = new double[]{1.0, 2.0, 3.0};
        ArrayUtils.reverse(dArr);
        assertArrayEquals(new double[]{3.0, 2.0, 1.0}, dArr, 0.0);

        ArrayUtils.reverse((float[]) null);
        float[] fArr = new float[]{1.0f, 2.0f, 3.0f};
        ArrayUtils.reverse(fArr);
        assertArrayEquals(new float[]{3.0f, 2.0f, 1.0f}, fArr, 0.0f);

        ArrayUtils.reverse((boolean[]) null);
        boolean[] boolArr = new boolean[]{true, false, true};
        ArrayUtils.reverse(boolArr);
        assertTrue(boolArr[0]);
        assertFalse(boolArr[1]);
        assertTrue(boolArr[2]);
    }

    @Test
    public void testIndexOfAndContains_Object() {
        assertEquals(-1, ArrayUtils.indexOf((Object[]) null, "a"));
        assertEquals(-1, ArrayUtils.indexOf((Object[]) null, "a", 0));
        assertEquals(-1, ArrayUtils.lastIndexOf((Object[]) null, "a"));
        assertEquals(-1, ArrayUtils.lastIndexOf((Object[]) null, "a", 0));
        assertFalse(ArrayUtils.contains((Object[]) null, "a"));

        String[] array = new String[]{"a", "b", null, "b", "c"};
        assertEquals(0, ArrayUtils.indexOf(array, "a"));
        assertEquals(1, ArrayUtils.indexOf(array, "b"));
        assertEquals(2, ArrayUtils.indexOf(array, null));
        assertEquals(3, ArrayUtils.indexOf(array, "b", 2));
        assertEquals(1, ArrayUtils.indexOf(array, "b", -1));
        assertEquals(-1, ArrayUtils.indexOf(array, "b", 10));
        assertEquals(-1, ArrayUtils.indexOf(array, 123)); // incompatible type branch

        assertEquals(3, ArrayUtils.lastIndexOf(array, "b"));
        assertEquals(2, ArrayUtils.lastIndexOf(array, null));
        assertEquals(1, ArrayUtils.lastIndexOf(array, "b", 2));
        assertEquals(-1, ArrayUtils.lastIndexOf(array, "b", -1));
        assertEquals(3, ArrayUtils.lastIndexOf(array, "b", 10));
        assertEquals(-1, ArrayUtils.lastIndexOf(array, 123)); // incompatible type branch

        assertTrue(ArrayUtils.contains(array, "c"));
        assertTrue(ArrayUtils.contains(array, null));
        assertFalse(ArrayUtils.contains(array, "z"));
    }
