package org.jsoup.parser;

import org.junit.Assert;
import org.junit.Test;

public class TokeniserStateTest {

    private Tokeniser createTokeniser(String input) {
        CharacterReader reader = new CharacterReader(input);
        ParseErrorList errors = ParseErrorList.tracking(100);
        return new Tokeniser(reader, errors);
    }

    @Test
    public void testEnumValuesAndValueOf() {
        TokeniserState[] states = TokeniserState.values();
        Assert.assertNotNull(states);
        Assert.assertTrue(states.length > 0);

        TokeniserState state = TokeniserState.valueOf("Data");
        Assert.assertEquals(TokeniserState.Data, state);
    }

    @Test
    public void testData_transitionsAndEmits() {
        // '&'
        Tokeniser t1 = createTokeniser("&");
        TokeniserState.Data.read(t1, t1.reader);
        Assert.assertEquals(TokeniserState.CharacterReferenceInData, t1.getState());

        // '<'
        Tokeniser t2 = createTokeniser("<");
        TokeniserState.Data.read(t2, t2.reader);
        Assert.assertEquals(TokeniserState.TagOpen, t2.getState());

        // null char '\u0000'
        Tokeniser t3 = createTokeniser("\u0000");
        TokeniserState.Data.read(t3, t3.reader);
        Assert.assertFalse(t3.errors.isEmpty());

        // EOF
        Tokeniser t4 = createTokeniser("");
        TokeniserState.Data.read(t4, t4.reader);

        // default text
        Tokeniser t5 = createTokeniser("abc<");
        TokeniserState.Data.read(t5, t5.reader);
    }

    @Test
    public void testTagName() {
        char[] ws = new char[]{'\t', '\n', '\f', ' '};
        for (char c : ws) {
            Tokeniser t = createTokeniser("div" + c);
            t.createTagPending(true);
            TokeniserState.TagName.read(t, t.reader);
            Assert.assertEquals(TokeniserState.BeforeAttributeName, t.getState());
        }

        Tokeniser t1 = createTokeniser("div/");
        t1.createTagPending(true);
        TokeniserState.TagName.read(t1, t1.reader);
        Assert.assertEquals(TokeniserState.SelfClosingStartTag, t1.getState());

        Tokeniser t2 = createTokeniser("div>");
        t2.createTagPending(true);
        TokeniserState.TagName.read(t2, t2.reader);
        Assert.assertEquals(TokeniserState.Data, t2.getState());

        Tokeniser t3 = createTokeniser("div\u0000");
        t3.createTagPending(true);
        TokeniserState.TagName.read(t3, t3.reader);

        Tokeniser t4 = createTokeniser("div");
        t4.createTagPending(true);
        TokeniserState.TagName.read(t4, t4.reader);
        Assert.assertEquals(TokeniserState.Data, t4.getState());
        Assert.assertFalse(t4.errors.isEmpty());
    }

    @Test
    public void testRCDATAEndTagName() {
        Tokeniser t1 = createTokeniser("itle ");
        t1.createTagPending(false);
        TokeniserState.RCDATAEndTagName.read(t1, t1.reader);

        Tokeniser t2 = createTokeniser(" ");
        t2.createTagPending(false);
        TokeniserState.RCDATAEndTagName.read(t2, t2.reader);
        Assert.assertEquals(TokeniserState.Rcdata, t2.getState());

        Tokeniser t3 = createTokeniser("/");
        t3.createTagPending(false);
        TokeniserState.RCDATAEndTagName.read(t3, t3.reader);
        Assert.assertEquals(TokeniserState.Rcdata, t3.getState());

        Tokeniser t4 = createTokeniser(">");
        t4.createTagPending(false);
        TokeniserState.RCDATAEndTagName.read(t4, t4.reader);
        Assert.assertEquals(TokeniserState.Rcdata, t4.getState());

        Tokeniser t5 = createTokeniser("x");
        t5.createTagPending(false);
        TokeniserState.RCDATAEndTagName.read(t5, t5.reader);
        Assert.assertEquals(TokeniserState.Rcdata, t5.getState());

        // Appropriate end tag token tests
        char[] spaces = {'\t', '\n', '\f', ' '};
        for (char sp : spaces) {
            Tokeniser t = createTokeniser(String.valueOf(sp));
            t.createTagPending(true);
            t.tagPending.name("title");
            t.emitTagPending();
            t.createTagPending(false);
            t.tagPending.name("title");
            TokeniserState.RCDATAEndTagName.read(t, t.reader);
            Assert.assertEquals(TokeniserState.BeforeAttributeName, t.getState());
        }

        Tokeniser tSlash = createTokeniser("/");
        tSlash.createTagPending(true);
        tSlash.tagPending.name("title");
        tSlash.emitTagPending();
        tSlash.createTagPending(false);
        tSlash.tagPending.name("title");
        TokeniserState.RCDATAEndTagName.read(tSlash, tSlash.reader);
        Assert.assertEquals(TokeniserState.SelfClosingStartTag, tSlash.getState());

        Tokeniser tGt = createTokeniser(">");
        tGt.createTagPending(true);
        tGt.tagPending.name("title");
        tGt.emitTagPending();
        tGt.createTagPending(false);
        tGt.tagPending.name("title");
        TokeniserState.RCDATAEndTagName.read(tGt, tGt.reader);
        Assert.assertEquals(TokeniserState.Data, tGt.getState());
    }

        TokeniserState.ScriptDataDoubleEscapeStart.read(t2, t2.reader);
        Assert.assertEquals(TokeniserState.ScriptDataDoubleEscaped, t2.getState());

        Tokeniser t3 = createTokeniser(" ");
        t3.dataBuffer.append("other");
        TokeniserState.ScriptDataDoubleEscapeStart.read(t3, t3.reader);
