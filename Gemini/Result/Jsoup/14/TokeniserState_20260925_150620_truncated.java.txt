package org.jsoup.parser;

import org.junit.Assert;
import org.junit.Before;
import org.junit.Test;

public class TokeniserStateTest {

    private Tokeniser tokeniser;
    private ParseErrorList errorList;

    @Before
    public void setUp() {
        errorList = ParseErrorList.tracking(100);
        tokeniser = new Tokeniser(new CharacterReader(""), errorList);
    }

    private CharacterReader createReader(String input) {
        CharacterReader reader = new CharacterReader(input);
        tokeniser = new Tokeniser(reader, errorList);
        return reader;
    }

    @Test
    public void testEnumConstants_valuesAndValueOf_valid() {
        TokeniserState[] states = TokeniserState.values();
        Assert.assertTrue(states.length > 0);
        Assert.assertEquals(TokeniserState.Data, TokeniserState.valueOf("Data"));
        Assert.assertEquals(TokeniserState.CdataSection, TokeniserState.valueOf("CdataSection"));
    }

    @Test
    public void testData_transitionsAndEmits_allBranches() {
        CharacterReader r1 = createReader("&");
        TokeniserState.Data.read(tokeniser, r1);
        Assert.assertEquals(TokeniserState.CharacterReferenceInData, tokeniser.getState());

        CharacterReader r2 = createReader("<");
        TokeniserState.Data.read(tokeniser, r2);
        Assert.assertEquals(TokeniserState.TagOpen, tokeniser.getState());

        CharacterReader r3 = createReader("\u0000");
        TokeniserState.Data.read(tokeniser, r3);
        Assert.assertFalse(errorList.isEmpty());

        CharacterReader r4 = createReader("");
        TokeniserState.Data.read(tokeniser, r4);

        CharacterReader r5 = createReader("abc&");
        TokeniserState.Data.read(tokeniser, r5);
    }

    @Test

    @Test
    public void testRCDATAEndTagName_branches() {
        CharacterReader r1 = createReader("itle ");
        tokeniser.createTagPending(false);
        tokeniser.tagPending.appendTagName("t");
        tokeniser.dataBuffer.append("t");
        TokeniserState.RCDATAEndTagName.read(tokeniser, r1);

        CharacterReader r2 = createReader(" ");
        tokeniser.createTagPending(false);
        tokeniser.tagPending.appendTagName("title");
        tokeniser.dataBuffer.append("title");
        tokeniser.appropriateEndTagName = "title";
        TokeniserState.RCDATAEndTagName.read(tokeniser, r2);
        Assert.assertEquals(TokeniserState.BeforeAttributeName, tokeniser.getState());

        CharacterReader r3 = createReader(" ");
        tokeniser.createTagPending(false);
        tokeniser.tagPending.appendTagName("other");
        tokeniser.dataBuffer.append("other");
        tokeniser.appropriateEndTagName = "title";
        TokeniserState.RCDATAEndTagName.read(tokeniser, r3);
        Assert.assertEquals(TokeniserState.Rcdata, tokeniser.getState());

        CharacterReader r4 = createReader("/");
        tokeniser.createTagPending(false);
        tokeniser.tagPending.appendTagName("title");
        tokeniser.dataBuffer.append("title");
        tokeniser.appropriateEndTagName = "title";
        TokeniserState.RCDATAEndTagName.read(tokeniser, r4);
        Assert.assertEquals(TokeniserState.SelfClosingStartTag, tokeniser.getState());

        CharacterReader r5 = createReader("/");
        tokeniser.createTagPending(false);
        tokeniser.tagPending.appendTagName("other");
        tokeniser.dataBuffer.append("other");
        tokeniser.appropriateEndTagName = "title";
        TokeniserState.RCDATAEndTagName.read(tokeniser, r5);
        Assert.assertEquals(TokeniserState.Rcdata, tokeniser.getState());

        CharacterReader r6 = createReader(">");
        tokeniser.createTagPending(false);
        tokeniser.tagPending.appendTagName("title");
        tokeniser.dataBuffer.append("title");
        tokeniser.appropriateEndTagName = "title";
        TokeniserState.RCDATAEndTagName.read(tokeniser, r6);
        Assert.assertEquals(TokeniserState.Data, tokeniser.getState());

        CharacterReader r7 = createReader(">");
        tokeniser.createTagPending(false);
        tokeniser.tagPending.appendTagName("other");
        tokeniser.dataBuffer.append("other");
        tokeniser.appropriateEndTagName = "title";
        TokeniserState.RCDATAEndTagName.read(tokeniser, r7);
        Assert.assertEquals(TokeniserState.Rcdata, tokeniser.getState());

        CharacterReader r8 = createReader("x");
        tokeniser.createTagPending(false);
        tokeniser.tagPending.appendTagName("title");
        tokeniser.dataBuffer.append("title");
        TokeniserState.RCDATAEndTagName.read(tokeniser, r8);
        Assert.assertEquals(TokeniserState.Rcdata, tokeniser.getState());
