    private Tokeniser createTokeniser(String input) {
        CharacterReader reader = new CharacterReader(input);
        ParseErrorList errors = ParseErrorList.tracking(100);
        return new Tokeniser(reader, errors);
    }

    private List<Token> tokenize(TokeniserState state, String input) {
        CharacterReader reader = new CharacterReader(input);
        ParseErrorList errors = ParseErrorList.tracking(100);
        Tokeniser tokeniser = new Tokeniser(reader, errors);
        tokeniser.transition(state);

        List<Token> tokens = new ArrayList<Token>();
        while (true) {
            Token token = tokeniser.read();
            if (token == null)
                break;
            tokens.add(token);
            if (token.type == Token.TokenType.EOF)
                break;
        }
        return tokens;
    }

    @Test
    public void testTagName_allBranches() {
        CharacterReader rSpace = new CharacterReader("div ");
        Tokeniser tSpace = new Tokeniser(rSpace, ParseErrorList.tracking(10));
        tSpace.createTagPending(true);
        TokeniserState.TagName.read(tSpace, rSpace);
        Assert.assertEquals(TokeniserState.BeforeAttributeName, tSpace.getState());

        CharacterReader rSlash = new CharacterReader("div/");
        Tokeniser tSlash = new Tokeniser(rSlash, ParseErrorList.tracking(10));
        tSlash.createTagPending(true);
        TokeniserState.TagName.read(tSlash, rSlash);
        Assert.assertEquals(TokeniserState.SelfClosingStartTag, tSlash.getState());

        CharacterReader rGt = new CharacterReader("div>");
        Tokeniser tGt = new Tokeniser(rGt, ParseErrorList.tracking(10));
        tGt.createTagPending(true);
        TokeniserState.TagName.read(tGt, rGt);
        Assert.assertEquals(TokeniserState.Data, tGt.getState());
        Token tagToken = tGt.read();
        Assert.assertEquals("div", ((Token.StartTag) tagToken).name());

        CharacterReader rNull = new CharacterReader("di\u0000v");
        Tokeniser tNull = new Tokeniser(rNull, ParseErrorList.tracking(10));
        tNull.createTagPending(true);
        TokeniserState.TagName.read(tNull, rNull);
        Assert.assertTrue(tNull.tagPending.tagName.contains(String.valueOf(Tokeniser.replacementChar)));

        CharacterReader rEof = new CharacterReader("div");
        Tokeniser tEof = new Tokeniser(rEof, ParseErrorList.tracking(10));
        tEof.createTagPending(true);
        TokeniserState.TagName.read(tEof, rEof);
        Assert.assertEquals(TokeniserState.Data, tEof.getState());
    }
