public class HtmlTreeBuilderStateTest {

    private HtmlTreeBuilder createBuilder(String baseUri) {
        HtmlTreeBuilder tb = new HtmlTreeBuilder();
        tb.initialiseParse(new StringReader(""), baseUri, new Parser(tb));
        return tb;
    }

    private HtmlTreeBuilder createFragmentBuilder(Element context) {
        HtmlTreeBuilder tb = new HtmlTreeBuilder();
        tb.initialiseParseFragment(context, "");
        return tb;
    }

        HtmlTreeBuilderState[] states = HtmlTreeBuilderState.values();
        assertTrue(states.length > 0);
        assertEquals(HtmlTreeBuilderState.Initial, HtmlTreeBuilderState.valueOf("Initial"));
        assertEquals(HtmlTreeBuilderState.ForeignContent, HtmlTreeBuilderState.valueOf("ForeignContent"));
    }

    @Test
    public void testInitial_tokens() {
        HtmlTreeBuilder tb = createBuilder("http://example.com");

        // Whitespace token
        Token.Character ws = new Token.Character().data("   \t\r\n");
        assertTrue(HtmlTreeBuilderState.Initial.process(ws, tb));

        // Comment token
        Token.Comment comment = new Token.Comment().setData("initial comment");
        assertTrue(HtmlTreeBuilderState.Initial.process(comment, tb));

        // Doctype token with force quirks
        Token.Doctype doctype = new Token.Doctype();
        doctype.name.append("html");
        doctype.forceQuirks = true;
        assertTrue(HtmlTreeBuilderState.Initial.process(doctype, tb));
        assertEquals(Document.QuirksMode.quirks, tb.getDocument().quirksMode());
        assertEquals(HtmlTreeBuilderState.BeforeHtml, tb.state());

        // Doctype with pub/sys
        HtmlTreeBuilder tb2 = createBuilder("http://example.com");
        Token.Doctype doctype2 = new Token.Doctype();
        doctype2.name.append("html");
        doctype2.publicIdentifier.append("public-id");
        doctype2.systemIdentifier.append("system-id");
        assertTrue(HtmlTreeBuilderState.Initial.process(doctype2, tb2));
        DocumentType dt = (DocumentType) tb2.getDocument().childNode(0);
        assertEquals("html", dt.name());
        assertEquals("public-id", dt.publicId());
        assertEquals("system-id", dt.systemId());

        // Anything else -> transitions to BeforeHtml and re-processes
        HtmlTreeBuilder tb3 = createBuilder("");
        Token.StartTag st = new Token.StartTag();
        st.name("div");
        assertTrue(HtmlTreeBuilderState.Initial.process(st, tb3));
    }

    @Test
    public void testBeforeHtml_tokens() {
        HtmlTreeBuilder tb = createBuilder("");

        // Doctype
        Token.Doctype dt = new Token.Doctype();
        assertFalse(HtmlTreeBuilderState.BeforeHtml.process(dt, tb));

        // Comment
        Token.Comment comment = new Token.Comment().setData("before html comment");
        assertTrue(HtmlTreeBuilderState.BeforeHtml.process(comment, tb));

        // Whitespace
        Token.Character ws = new Token.Character().data(" ");
        assertTrue(HtmlTreeBuilderState.BeforeHtml.process(ws, tb));

        // StartTag html
        Token.StartTag htmlTag = new Token.StartTag();
        htmlTag.name("html");
        assertTrue(HtmlTreeBuilderState.BeforeHtml.process(htmlTag, tb));
        assertEquals(HtmlTreeBuilderState.BeforeHead, tb.state());

        // EndTag head, body, html, br
        for (String tag : new String[]{"head", "body", "html", "br"}) {
            HtmlTreeBuilder tbLoop = createBuilder("");
            Token.EndTag et = new Token.EndTag();
            et.name(tag);
            assertTrue(HtmlTreeBuilderState.BeforeHtml.process(et, tbLoop));
        }

        // Other EndTag -> error and false
        HtmlTreeBuilder tbErr = createBuilder("");
        Token.EndTag etDiv = new Token.EndTag();
        etDiv.name("div");
        assertFalse(HtmlTreeBuilderState.BeforeHtml.process(etDiv, tbErr));
    }

    @Test
    public void testBeforeHead_tokens() {
        HtmlTreeBuilder tb = createBuilder("");
        tb.transition(HtmlTreeBuilderState.BeforeHead);

        // Whitespace
        assertTrue(HtmlTreeBuilderState.BeforeHead.process(new Token.Character().data("\n "), tb));

        // Comment
        assertTrue(HtmlTreeBuilderState.BeforeHead.process(new Token.Comment().setData("comment"), tb));

        // Doctype
        assertFalse(HtmlTreeBuilderState.BeforeHead.process(new Token.Doctype(), tb));

        // StartTag html
        Token.StartTag htmlTag = new Token.StartTag();
        htmlTag.name("html");
        assertTrue(HtmlTreeBuilderState.BeforeHead.process(htmlTag, tb));

        // StartTag head
        Token.StartTag headTag = new Token.StartTag();
        headTag.name("head");
        assertTrue(HtmlTreeBuilderState.BeforeHead.process(headTag, tb));
        assertEquals(HtmlTreeBuilderState.InHead, tb.state());

        // EndTag head/body/html/br
        HtmlTreeBuilder tbEnd = createBuilder("");
        Token.EndTag endTag = new Token.EndTag();
        endTag.name("head");
        assertTrue(HtmlTreeBuilderState.BeforeHead.process(endTag, tbEnd));

        // Other EndTag
        HtmlTreeBuilder tbEndErr = createBuilder("");
        Token.EndTag endTagDiv = new Token.EndTag();
        endTagDiv.name("span");
        assertFalse(HtmlTreeBuilderState.BeforeHead.process(endTagDiv, tbEndErr));

        // Anything else
        HtmlTreeBuilder tbOther = createBuilder("");
        Token.StartTag div = new Token.StartTag();
        div.name("div");
        assertTrue(HtmlTreeBuilderState.BeforeHead.process(div, tbOther));
    }

    @Test
    public void testInHead_tokens() {
        HtmlTreeBuilder tb = createBuilder("http://example.com");
        tb.processStartTag("html");
        tb.processStartTag("head");
        assertEquals(HtmlTreeBuilderState.InHead, tb.state());

        // Whitespace
        assertTrue(HtmlTreeBuilderState.InHead.process(new Token.Character().data(" "), tb));

        // Comment
        assertTrue(HtmlTreeBuilderState.InHead.process(new Token.Comment().setData("in head"), tb));

        // Doctype
        assertFalse(HtmlTreeBuilderState.InHead.process(new Token.Doctype(), tb));

        // StartTag html
        Token.StartTag htmlTag = new Token.StartTag();
        htmlTag.name("html");
        assertTrue(HtmlTreeBuilderState.InHead.process(htmlTag, tb));

        // Base with and without href
        Token.StartTag baseTag = new Token.StartTag();
        baseTag.name("base");
        baseTag.attributes.put("href", "http://example.com/base/");
        assertTrue(HtmlTreeBuilderState.InHead.process(baseTag, tb));
        assertEquals("http://example.com/base/", tb.getBaseUri());

        Token.StartTag baseNoHref = new Token.StartTag();
        baseNoHref.name("base");
        assertTrue(HtmlTreeBuilderState.InHead.process(baseNoHref, tb));

        for (String emptyTag : new String[]{"basefont", "bgsound", "command", "link", "meta"}) {
            Token.StartTag st = new Token.StartTag();
            st.name(emptyTag);
            assertTrue(HtmlTreeBuilderState.InHead.process(st, tb));
        }

        // Title, Style, Noframes, Noscript, Script
        for (String rcDataTag : new String[]{"title", "noframes", "style", "noscript", "script"}) {
            HtmlTreeBuilder tbHead = createBuilder("");
            tbHead.processStartTag("html");
            tbHead.processStartTag("head");
            Token.StartTag st = new Token.StartTag();
            st.name(rcDataTag);
            assertTrue(HtmlTreeBuilderState.InHead.process(st, tbHead));
        }

        // StartTag head in head -> error
        Token.StartTag nestedHead = new Token.StartTag();
        nestedHead.name("head");
        assertFalse(HtmlTreeBuilderState.InHead.process(nestedHead, tb));

        // EndTag head
        Token.EndTag endHead = new Token.EndTag();
        endHead.name("head");
        assertTrue(HtmlTreeBuilderState.InHead.process(endHead, tb));
        assertEquals(HtmlTreeBuilderState.AfterHead, tb.state());

        // EndTag body/html/br
        HtmlTreeBuilder tbEnd = createBuilder("");
        tbEnd.processStartTag("html");
        tbEnd.processStartTag("head");
        Token.EndTag endBody = new Token.EndTag();
        endBody.name("body");
        assertTrue(HtmlTreeBuilderState.InHead.process(endBody, tbEnd));

        // EndTag other
        HtmlTreeBuilder tbEndErr = createBuilder("");
        tbEndErr.processStartTag("html");
        tbEndErr.processStartTag("head");
        Token.EndTag endDiv = new Token.EndTag();
        endDiv.name("div");
        assertFalse(HtmlTreeBuilderState.InHead.process(endDiv, tbEndErr));

        // Default branch
        HtmlTreeBuilder tbDef = createBuilder("");
        tbDef.processStartTag("html");
        tbDef.processStartTag("head");
        Token.EOF eof = new Token.EOF();
        assertTrue(HtmlTreeBuilderState.InHead.process(eof, tbDef));
    }

    @Test
    public void testInHeadNoscript_tokens() {
        HtmlTreeBuilder tb = createBuilder("");
        tb.processStartTag("html");
        tb.processStartTag("head");
        tb.processStartTag("noscript");
        assertEquals(HtmlTreeBuilderState.InHeadNoscript, tb.state());

        // Doctype
        assertTrue(HtmlTreeBuilderState.InHeadNoscript.process(new Token.Doctype(), tb));

        // StartTag html
        Token.StartTag htmlTag = new Token.StartTag();
        htmlTag.name("html");
        assertTrue(HtmlTreeBuilderState.InHeadNoscript.process(htmlTag, tb));

        // Whitespace, Comment, StartTags
        assertTrue(HtmlTreeBuilderState.InHeadNoscript.process(new Token.Character().data(" "), tb));
        assertTrue(HtmlTreeBuilderState.InHeadNoscript.process(new Token.Comment().setData("c"), tb));
        for (String tag : new String[]{"basefont", "bgsound", "link", "meta", "noframes", "style"}) {
            Token.StartTag st = new Token.StartTag();
            st.name(tag);
            assertTrue(HtmlTreeBuilderState.InHeadNoscript.process(st, tb));
        }

        // EndTag br
        Token.EndTag endBr = new Token.EndTag();
        endBr.name("br");
        assertTrue(HtmlTreeBuilderState.InHeadNoscript.process(endBr, tb));

        // StartTag head/noscript and other end tag -> error, false
        Token.StartTag startHead = new Token.StartTag();
        startHead.name("head");
        assertFalse(HtmlTreeBuilderState.InHeadNoscript.process(startHead, tb));

        Token.EndTag endDiv = new Token.EndTag();
        endDiv.name("div");
        assertFalse(HtmlTreeBuilderState.InHeadNoscript.process(endDiv, tb));

        // Anything else -> error, insert character
        Token.StartTag custom = new Token.StartTag();
        custom.name("custom");
        assertTrue(HtmlTreeBuilderState.InHeadNoscript.process(custom, tb));

        // EndTag noscript -> transition InHead
        Token.EndTag endNoscript = new Token.EndTag();
        endNoscript.name("noscript");
        assertTrue(HtmlTreeBuilderState.InHeadNoscript.process(endNoscript, tb));
        assertEquals(HtmlTreeBuilderState.InHead, tb.state());
    }

    @Test
    public void testAfterHead_tokens() {
        HtmlTreeBuilder tb = createBuilder("");
        tb.processStartTag("html");
        tb.processStartTag("head");
        tb.processEndTag("head");
        assertEquals(HtmlTreeBuilderState.AfterHead, tb.state());

        // Whitespace, comment, doctype
        assertTrue(HtmlTreeBuilderState.AfterHead.process(new Token.Character().data(" "), tb));
        assertTrue(HtmlTreeBuilderState.AfterHead.process(new Token.Comment().setData("c"), tb));
        assertTrue(HtmlTreeBuilderState.AfterHead.process(new Token.Doctype(), tb));

        // StartTag html
        Token.StartTag html = new Token.StartTag();
        html.name("html");
        assertTrue(HtmlTreeBuilderState.AfterHead.process(html, tb));

        // StartTag head -> error & false
        Token.StartTag head = new Token.StartTag();
        head.name("head");
        assertFalse(HtmlTreeBuilderState.AfterHead.process(head, tb));

        // StartTag base/meta/etc. (head tags after head closed)
        for (String hTag : new String[]{"base", "basefont", "bgsound", "link", "meta", "noframes", "script", "style", "title"}) {
            HtmlTreeBuilder tbHeadTags = createBuilder("");
            tbHeadTags.processStartTag("html");
            tbHeadTags.processStartTag("head");
            tbHeadTags.processEndTag("head");
            Token.StartTag st = new Token.StartTag();
            st.name(hTag);
            assertTrue(HtmlTreeBuilderState.AfterHead.process(st, tbHeadTags));
        }

        // StartTag frameset
        HtmlTreeBuilder tbFrame = createBuilder("");
        tbFrame.processStartTag("html");
        tbFrame.processStartTag("head");
        tbFrame.processEndTag("head");
        Token.StartTag frameset = new Token.StartTag();
        frameset.name("frameset");
        assertTrue(HtmlTreeBuilderState.AfterHead.process(frameset, tbFrame));
        assertEquals(HtmlTreeBuilderState.InFrameset, tbFrame.state());

        // StartTag body
        HtmlTreeBuilder tbBody = createBuilder("");
        tbBody.processStartTag("html");
        tbBody.processStartTag("head");
        tbBody.processEndTag("head");
        Token.StartTag body = new Token.StartTag();
        body.name("body");
        assertTrue(HtmlTreeBuilderState.AfterHead.process(body, tbBody));
        assertEquals(HtmlTreeBuilderState.InBody, tbBody.state());

        // EndTag body/html vs other EndTag
        HtmlTreeBuilder tbEnd = createBuilder("");
        tbEnd.processStartTag("html");
        tbEnd.processStartTag("head");
        tbEnd.processEndTag("head");
        Token.EndTag endBody = new Token.EndTag();
        endBody.name("body");
        assertTrue(HtmlTreeBuilderState.AfterHead.process(endBody, tbEnd));

        HtmlTreeBuilder tbEndErr = createBuilder("");
        tbEndErr.processStartTag("html");
        tbEndErr.processStartTag("head");
        tbEndErr.processEndTag("head");
        Token.EndTag endP = new Token.EndTag();
        endP.name("p");
        assertFalse(HtmlTreeBuilderState.AfterHead.process(endP, tbEndErr));

        // Anything else (e.g. div)
        HtmlTreeBuilder tbElse = createBuilder("");
        tbElse.processStartTag("html");
        tbElse.processStartTag("head");
        tbElse.processEndTag("head");
        Token.StartTag div = new Token.StartTag();
        div.name("div");
        assertTrue(HtmlTreeBuilderState.AfterHead.process(div, tbElse));
    }

    @Test
    public void testInBody_variousTags() {
        // Character nullString vs normal character vs whitespace
        HtmlTreeBuilder tb = createBuilder("");
        tb.transition(HtmlTreeBuilderState.InBody);
        Token.Character nullChar = new Token.Character().data("\u0000");
        assertFalse(HtmlTreeBuilderState.InBody.process(nullChar, tb));

        Token.Character wsChar = new Token.Character().data("  ");
        assertTrue(HtmlTreeBuilderState.InBody.process(wsChar, tb));

        Token.Character normalChar = new Token.Character().data("abc");
        assertTrue(HtmlTreeBuilderState.InBody.process(normalChar, tb));

        // Comment, Doctype
        assertTrue(HtmlTreeBuilderState.InBody.process(new Token.Comment().setData("body comm"), tb));
        assertFalse(HtmlTreeBuilderState.InBody.process(new Token.Doctype(), tb));

        // StartTag a (duplicate active formatting element)
        HtmlTreeBuilder tbA = createBuilder("");
        tbA.processStartTag("html");
        tbA.processStartTag("body");
        Token.StartTag aTag = new Token.StartTag();
        aTag.name("a");
        assertTrue(HtmlTreeBuilderState.InBody.process(aTag, tbA));
        assertTrue(HtmlTreeBuilderState.InBody.process(aTag, tbA)); // trigger duplicate formatting element

        // StartTag EmptyFormatters
        for (String tag : new String[]{"area", "br", "embed", "img", "keygen", "wbr"}) {
            Token.StartTag st = new Token.StartTag();
            st.name(tag);
            assertTrue(HtmlTreeBuilderState.InBody.process(st, tbA));
        }

        // StartTag span
        Token.StartTag span = new Token.StartTag();
        span.name("span");
        assertTrue(HtmlTreeBuilderState.InBody.process(span, tbA));

        // StartTag li
        Token.StartTag li = new Token.StartTag();
        li.name("li");
        assertTrue(HtmlTreeBuilderState.InBody.process(li, tbA));
        assertTrue(HtmlTreeBuilderState.InBody.process(li, tbA));

        // StartTag html with attributes
        Token.StartTag html = new Token.StartTag();
        html.name("html");
        html.attributes.put("class", "my-class");
        assertTrue(HtmlTreeBuilderState.InBody.process(html, tbA));

        // StartTag body (attributes merge)
        Token.StartTag body = new Token.StartTag();
        body.name("body");
        body.attributes.put("id", "main-body");
        assertTrue(HtmlTreeBuilderState.InBody.process(body, tbA));

        // StartTag frameset when framesetOk false / true
        Token.StartTag frameset = new Token.StartTag();
        frameset.name("frameset");
        assertFalse(HtmlTreeBuilderState.InBody.process(frameset, tbA));

        HtmlTreeBuilder tbFrameset = createBuilder("");
        tbFrameset.processStartTag("html");
        tbFrameset.processStartTag("body");
        tbFrameset.framesetOk(true);
        assertTrue(HtmlTreeBuilderState.InBody.process(frameset, tbFrameset));

        // Headings h1..h6
        for (String h : new String[]{"h1", "h2", "h3", "h4", "h5", "h6"}) {
            Token.StartTag st = new Token.StartTag();
            st.name(h);
            assertTrue(HtmlTreeBuilderState.InBody.process(st, tbA));
        }

        // Pre, Listing
        Token.StartTag pre = new Token.StartTag();
        pre.name("pre");
        assertTrue(HtmlTreeBuilderState.InBody.process(pre, tbA));

        // Form (single and duplicate)
        Token.StartTag form = new Token.StartTag();
        form.name("form");
        assertTrue(HtmlTreeBuilderState.InBody.process(form, tbA));
        assertFalse(HtmlTreeBuilderState.InBody.process(form, tbA)); // second form rejected

        // Dd, Dt
        for (String ddDt : new String[]{"dd", "dt"}) {
            Token.StartTag st = new Token.StartTag();
            st.name(ddDt);
            assertTrue(HtmlTreeBuilderState.InBody.process(st, tbA));
        }

        // Plaintext
        Token.StartTag pt = new Token.StartTag();
        pt.name("plaintext");
        assertTrue(HtmlTreeBuilderState.InBody.process(pt, tbA));

        // Button
        Token.StartTag btn = new Token.StartTag();
        btn.name("button");
        assertTrue(HtmlTreeBuilderState.InBody.process(btn, tbA));
        assertTrue(HtmlTreeBuilderState.InBody.process(btn, tbA)); // button inside button

        // Formatters
        for (String f : new String[]{"b", "i", "s", "small", "strong", "u"}) {
            Token.StartTag st = new Token.StartTag();
            st.name(f);
            assertTrue(HtmlTreeBuilderState.InBody.process(st, tbA));
        }

        // Nobr
        Token.StartTag nobr = new Token.StartTag();
        nobr.name("nobr");
        assertTrue(HtmlTreeBuilderState.InBody.process(nobr, tbA));
        assertTrue(HtmlTreeBuilderState.InBody.process(nobr, tbA)); // nested nobr

        // Applets
        for (String applet : new String[]{"applet", "marquee", "object"}) {
            Token.StartTag st = new Token.StartTag();
            st.name(applet);
            assertTrue(HtmlTreeBuilderState.InBody.process(st, tbA));
        }

        // Table
        Token.StartTag table = new Token.StartTag();
        table.name("table");
        assertTrue(HtmlTreeBuilderState.InBody.process(table, tbA));
    }

    @Test
    public void testInBody_moreStartTags() {
        HtmlTreeBuilder tb = createBuilder("");
        tb.processStartTag("html");
        tb.processStartTag("body");

        // input (hidden vs text)
        Token.StartTag inputHidden = new Token.StartTag();
        inputHidden.name("input");
        inputHidden.attributes.put("type", "hidden");
        assertTrue(HtmlTreeBuilderState.InBody.process(inputHidden, tb));

        Token.StartTag inputText = new Token.StartTag();
        inputText.name("input");
        inputText.attributes.put("type", "text");
        assertTrue(HtmlTreeBuilderState.InBody.process(inputText, tb));

        // Media
        for (String m : new String[]{"param", "source", "track"}) {
            Token.StartTag st = new Token.StartTag();
            st.name(m);
            assertTrue(HtmlTreeBuilderState.InBody.process(st, tb));
        }

        // hr
        Token.StartTag hr = new Token.StartTag();
        hr.name("hr");
        assertTrue(HtmlTreeBuilderState.InBody.process(hr, tb));

        // image outside vs inside svg
        Token.StartTag image = new Token.StartTag();
        image.name("image");
        assertTrue(HtmlTreeBuilderState.InBody.process(image, tb));

        // isindex
        Token.StartTag isindex = new Token.StartTag();
        isindex.name("isindex");
        isindex.attributes.put("action", "/search");
        isindex.attributes.put("prompt", "Search here: ");
        assertTrue(HtmlTreeBuilderState.InBody.process(isindex, tb));

        // textarea, xmp, iframe, noembed, select
        for (String raw : new String[]{"textarea", "xmp", "iframe", "noembed", "select"}) {
            HtmlTreeBuilder tbRaw = createBuilder("");
            tbRaw.processStartTag("html");
            tbRaw.processStartTag("body");
            Token.StartTag st = new Token.StartTag();
            st.name(raw);
            assertTrue(HtmlTreeBuilderState.InBody.process(st, tbRaw));
        }

        // options, ruby, math, svg, drop tags
        Token.StartTag opt = new Token.StartTag();
        opt.name("option");
        assertTrue(HtmlTreeBuilderState.InBody.process(opt, tb));

        Token.StartTag ruby = new Token.StartTag();
        ruby.name("ruby");
        assertTrue(HtmlTreeBuilderState.InBody.process(ruby, tb));
        Token.StartTag rt = new Token.StartTag();
        rt.name("rt");
        assertTrue(HtmlTreeBuilderState.InBody.process(rt, tb));

        Token.StartTag math = new Token.StartTag();
        math.name("math");
        assertTrue(HtmlTreeBuilderState.InBody.process(math, tb));

        Token.StartTag svg = new Token.StartTag();
        svg.name("svg");
        assertTrue(HtmlTreeBuilderState.InBody.process(svg, tb));

        Token.StartTag imgInSvg = new Token.StartTag();
        imgInSvg.name("image");
        assertTrue(HtmlTreeBuilderState.InBody.process(imgInSvg, tb));

        // Drop tags
        for (String drop : new String[]{"caption", "col", "colgroup", "frame", "head", "tbody", "td", "tfoot", "th", "thead", "tr"}) {
            Token.StartTag st = new Token.StartTag();
            st.name(drop);
            assertFalse(HtmlTreeBuilderState.InBody.process(st, tb));
        }

        // Generic StartTag
        Token.StartTag custom = new Token.StartTag();
        custom.name("custom-tag");
        assertTrue(HtmlTreeBuilderState.InBody.process(custom, tb));
    }

    @Test
    public void testInBody_endTags() {
        HtmlTreeBuilder tb = createBuilder("");
        tb.processStartTag("html");
        tb.processStartTag("body");

        // Adoption formatters: <a>, <b>, <i>, etc.
        Token.EndTag endA = new Token.EndTag();
        endA.name("a");
        assertTrue(HtmlTreeBuilderState.InBody.process(endA, tb));

        // End closers: <div>, <p>, etc.
        tb.processStartTag("div");
        Token.EndTag endDiv = new Token.EndTag();
        endDiv.name("div");
        assertTrue(HtmlTreeBuilderState.InBody.process(endDiv, tb));

        // EndTag li
        tb.processStartTag("li");
        Token.EndTag endLi = new Token.EndTag();
        endLi.name("li");
        assertTrue(HtmlTreeBuilderState.InBody.process(endLi, tb));

        // EndTag span
        tb.processStartTag("span");
        Token.EndTag endSpan = new Token.EndTag();
        endSpan.name("span");
        assertTrue(HtmlTreeBuilderState.InBody.process(endSpan, tb));

        // EndTag form
        tb.processStartTag("form");
        Token.EndTag endForm = new Token.EndTag();
        endForm.name("form");
        assertTrue(HtmlTreeBuilderState.InBody.process(endForm, tb));

        // EndTag p
        Token.EndTag endP = new Token.EndTag();
        endP.name("p");
        assertTrue(HtmlTreeBuilderState.InBody.process(endP, tb));

        // EndTag dd / dt
        tb.processStartTag("dd");
        Token.EndTag endDd = new Token.EndTag();
        endDd.name("dd");
        assertTrue(HtmlTreeBuilderState.InBody.process(endDd, tb));

        // EndTag headings
        tb.processStartTag("h1");
        Token.EndTag endH1 = new Token.EndTag();
        endH1.name("h1");
        assertTrue(HtmlTreeBuilderState.InBody.process(endH1, tb));

        // EndTag sarcasm
        Token.EndTag sarcasm = new Token.EndTag();
        sarcasm.name("sarcasm");
        assertTrue(HtmlTreeBuilderState.InBody.process(sarcasm, tb));

        // EndTag br
        Token.EndTag endBr = new Token.EndTag();
        endBr.name("br");
        assertFalse(HtmlTreeBuilderState.InBody.process(endBr, tb));

        // EndTag applet
        tb.processStartTag("applet");
        Token.EndTag endApplet = new Token.EndTag();
        endApplet.name("applet");
        assertTrue(HtmlTreeBuilderState.InBody.process(endApplet, tb));

        // EndTag body & html
        Token.EndTag endBody = new Token.EndTag();
        endBody.name("body");
        assertTrue(HtmlTreeBuilderState.InBody.process(endBody, tb));
        assertEquals(HtmlTreeBuilderState.AfterBody, tb.state());

        Token.EndTag endHtml = new Token.EndTag();
        endHtml.name("html");
        assertTrue(HtmlTreeBuilderState.InBody.process(endHtml, tb));

        // EOF in InBody
        assertTrue(HtmlTreeBuilderState.InBody.process(new Token.EOF(), tb));
    }
