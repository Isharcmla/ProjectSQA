package org.jsoup.parser;

import org.jsoup.nodes.DocumentType;
import org.junit.Assert;

public class TokeniserStateTest {

    private Tokeniser createTokeniser(String input) {
        CharacterReader reader = new CharacterReader(input);
        ParseErrorList errors = ParseErrorList.tracking(100);
        return new Tokeniser(reader, errors);
    }

    private boolean isSorted(char[] array) {
        char[] copy = Arrays.copyOf(array, array.length);
        Arrays.sort(copy);
        return Arrays.equals(array, copy);
    }

    @Test
    public void testEnumValuesAndValueOf() {
        TokeniserState[] states = TokeniserState.values();
        Assert.assertNotNull(states);
        Assert.assertTrue(states.length > 0);
        Assert.assertEquals(TokeniserState.Data, TokeniserState.valueOf("Data"));
        Assert.assertEquals(TokeniserState.TagOpen, TokeniserState.valueOf("TagOpen"));
    }

    @Test
    public void testArraysAreSorted() {
        Assert.assertTrue(isSorted(TokeniserState.attributeSingleValueCharsSorted));
        Assert.assertTrue(isSorted(TokeniserState.attributeDoubleValueCharsSorted));
        Assert.assertTrue(isSorted(TokeniserState.attributeNameCharsSorted));
        Assert.assertTrue(isSorted(TokeniserState.attributeValueUnquoted));
    }

    @Test
    public void testTagName_allBranches() {
        for (char ws : new char[]{'\t', '\n', '\r', '\f', ' '}) {
            Tokeniser t = createTokeniser("tag" + ws);
            t.createTagPending(true);
            TokeniserState.TagName.read(t, t.reader);
            Assert.assertEquals(TokeniserState.BeforeAttributeName, t.getState());
        }

        Tokeniser t1 = createTokeniser("tag/");
        t1.createTagPending(true);
        TokeniserState.TagName.read(t1, t1.reader);
        Assert.assertEquals(TokeniserState.SelfClosingStartTag, t1.getState());

        Tokeniser t2 = createTokeniser("tag>");
        t2.createTagPending(true);
        TokeniserState.TagName.read(t2, t2.reader);
        Assert.assertEquals(TokeniserState.Data, t2.getState());

        Tokeniser t3 = createTokeniser("tag\u0000");
        t3.createTagPending(true);
        TokeniserState.TagName.read(t3, t3.reader);
        Assert.assertTrue(t3.tagPending.name().contains(String.valueOf(Tokeniser.replacementChar)));

        Tokeniser t4 = createTokeniser("tag");
        t4.createTagPending(true);
        TokeniserState.TagName.read(t4, t4.reader);
        Assert.assertEquals(TokeniserState.Data, t4.getState());
        Assert.assertEquals(1, t4.getErrors().size());

        Tokeniser t5 = createTokeniser("tag#");
        t5.createTagPending(true);
        TokeniserState.TagName.read(t5, t5.reader);
        Assert.assertTrue(t5.tagPending.name().endsWith("#"));
    }

    @Test
    public void testRCDATAEndTagName_allBranches() {
        Tokeniser tLetter = createTokeniser("itle ");
        tLetter.createTagPending(false);
        tLetter.createTempBuffer();
        TokeniserState.RCDATAEndTagName.read(tLetter, tLetter.reader);
        Assert.assertEquals("itle", tLetter.dataBuffer.toString());

        for (char ws : new char[]{'\t', '\n', '\r', '\f', ' '}) {
            Tokeniser t = createTokeniser(String.valueOf(ws));
            t.createTagPending(false);
            t.tagPending.name("title");
            t.dataBuffer = new StringBuilder("title");
            TokeniserState.RCDATAEndTagName.read(t, t.reader);
            Assert.assertEquals(TokeniserState.Rcdata, t.getState());
        }

        Tokeniser tSlash = createTokeniser("/>");
        tSlash.createTagPending(false);
        tSlash.tagPending.name("title");
        tSlash.dataBuffer = new StringBuilder("title");
        TokeniserState.RCDATAEndTagName.read(tSlash, tSlash.reader);
        Assert.assertEquals(TokeniserState.Rcdata, tSlash.getState());

        Tokeniser tGt = createTokeniser(">");
        tGt.createTagPending(false);
        tGt.tagPending.name("title");
        tGt.dataBuffer = new StringBuilder("title");
        TokeniserState.RCDATAEndTagName.read(tGt, tGt.reader);
        Assert.assertEquals(TokeniserState.Rcdata, tGt.getState());

        Tokeniser tDefault = createTokeniser("1");
        tDefault.createTagPending(false);
        tDefault.createTempBuffer();
        TokeniserState.RCDATAEndTagName.read(tDefault, tDefault.reader);
        Assert.assertEquals(TokeniserState.Rcdata, tDefault.getState());
    }
