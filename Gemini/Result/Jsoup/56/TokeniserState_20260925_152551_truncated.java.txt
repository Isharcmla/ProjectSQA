public class TokeniserStateTest {

    private Tokeniser createTokeniser(String input) {
        CharacterReader reader = new CharacterReader(input);
        ParseErrorList errors = ParseErrorList.tracking(100);
        return new Tokeniser(reader, errors);
    }

    private List<Token> tokenize(String input) {
        Tokeniser tokeniser = createTokeniser(input);
        while (tokeniser.read() != null) {
            // Read until complete
        }
        return null;
    }

    @Test
    public void testEnumValuesAndValueOf() {
        TokeniserState[] states = TokeniserState.values();
        Assert.assertTrue(states.length > 0);
        TokeniserState state = TokeniserState.valueOf("Data");
        Assert.assertEquals(TokeniserState.Data, state);
    }

    @Test
    public void testDataState() {
        Tokeniser t = createTokeniser("&lt;<tag\u0000");
        TokeniserState.Data.read(t, t.reader); // '&'
        Assert.assertEquals(TokeniserState.CharacterReferenceInData, t.getState());

        t.transition(TokeniserState.Data);
        t.reader.consume(); // skip '&'
        t.reader.consumeTo('<');
        TokeniserState.Data.read(t, t.reader); // '<'
        Assert.assertEquals(TokeniserState.TagOpen, t.getState());

        t.transition(TokeniserState.Data);
        t.reader.consume(); // skip '<'
        t.reader.consumeTo('\u0000');
        TokeniserState.Data.read(t, t.reader); // nullChar
        Assert.assertFalse(t.getErrors().isEmpty());

        Tokeniser tEof = createTokeniser("");
        TokeniserState.Data.read(tEof, tEof.reader);
        Tokeniser tText = createTokeniser("hello world");
        TokeniserState.Data.read(tText, tText.reader);
    }

    @Test
    public void testCharacterReferenceInData() {

        Assert.assertEquals(TokeniserState.Data, t.getState());

        Tokeniser t2 = createTokeniser("zzzz;");
        TokeniserState.CharacterReferenceInData.read(t2, t2.reader);

    }

    @Test
    public void testRcdataState() {
        Tokeniser t = createTokeniser("&< \u0000");
        TokeniserState.Rcdata.read(t, t.reader); // '&'
        Assert.assertEquals(TokeniserState.CharacterReferenceInRcdata, t.getState());

        t.transition(TokeniserState.Rcdata);
        t.reader.consume();
        TokeniserState.Rcdata.read(t, t.reader); // '<'
        Assert.assertEquals(TokeniserState.RcdataLessthanSign, t.getState());

        t.transition(TokeniserState.Rcdata);
        t.reader.consume();
        TokeniserState.Rcdata.read(t, t.reader); // ' '
        Assert.assertEquals(TokeniserState.Rcdata, t.getState());

        TokeniserState.Rcdata.read(t, t.reader); // nullChar
        Assert.assertFalse(t.getErrors().isEmpty());

        Tokeniser tEof = createTokeniser("");
        TokeniserState.Rcdata.read(tEof, tEof.reader);
    }

    @Test
    public void testScriptDataEscapedTransitions() {
        Tokeniser t1 = createTokeniser("-");
        TokeniserState.ScriptDataEscapeStart.read(t1, t1.reader);
        Assert.assertEquals(TokeniserState.ScriptDataEscapeStartDash, t1.getState());

        Tokeniser t2 = createTokeniser("x");
        TokeniserState.ScriptDataEscapeStart.read(t2, t2.reader);
        Assert.assertEquals(TokeniserState.ScriptData, t2.getState());

        Tokeniser t3 = createTokeniser("-");
        TokeniserState.ScriptDataEscapeStartDash.read(t3, t3.reader);
        Assert.assertEquals(TokeniserState.ScriptDataEscapedDashDash, t3.getState());

        Tokeniser t4 = createTokeniser("x");
        TokeniserState.ScriptDataEscapeStartDash.read(t4, t4.reader);
        Assert.assertEquals(TokeniserState.ScriptData, t4.getState());

        // ScriptDataEscaped
        Tokeniser tEscEmpty = createTokeniser("");
        TokeniserState.ScriptDataEscaped.read(tEscEmpty, tEscEmpty.reader);
        Assert.assertEquals(TokeniserState.Data, tEscEmpty.getState());

        Tokeniser tEsc = createTokeniser("-<\u0000text");
        TokeniserState.ScriptDataEscaped.read(tEsc, tEsc.reader); // '-'
        Assert.assertEquals(TokeniserState.ScriptDataEscapedDash, tEsc.getState());

        tEsc.transition(TokeniserState.ScriptDataEscaped);
        TokeniserState.ScriptDataEscaped.read(tEsc, tEsc.reader); // '<'
        Assert.assertEquals(TokeniserState.ScriptDataEscapedLessthanSign, tEsc.getState());

        tEsc.transition(TokeniserState.ScriptDataEscaped);
        tEsc.reader.consume(); // skip '<'
        TokeniserState.ScriptDataEscaped.read(tEsc, tEsc.reader); // nullChar
        Assert.assertFalse(tEsc.getErrors().isEmpty());

        TokeniserState.ScriptDataEscaped.read(tEsc, tEsc.reader); // text

        // ScriptDataEscapedDash
        Tokeniser tDashEmpty = createTokeniser("");
        TokeniserState.ScriptDataEscapedDash.read(tDashEmpty, tDashEmpty.reader);
        Assert.assertEquals(TokeniserState.Data, tDashEmpty.getState());

        Tokeniser tDash1 = createTokeniser("-");
        TokeniserState.ScriptDataEscapedDash.read(tDash1, tDash1.reader);
        Assert.assertEquals(TokeniserState.ScriptDataEscapedDashDash, tDash1.getState());

        Tokeniser tDash2 = createTokeniser("<");
        TokeniserState.ScriptDataEscapedDash.read(tDash2, tDash2.reader);
        Assert.assertEquals(TokeniserState.ScriptDataEscapedLessthanSign, tDash2.getState());

        Tokeniser tDash3 = createTokeniser("\u0000");
        TokeniserState.ScriptDataEscapedDash.read(tDash3, tDash3.reader);
        Assert.assertEquals(TokeniserState.ScriptDataEscaped, tDash3.getState());

        Tokeniser tDash4 = createTokeniser("a");
        TokeniserState.ScriptDataEscapedDash.read(tDash4, tDash4.reader);
        Assert.assertEquals(TokeniserState.ScriptDataEscaped, tDash4.getState());

        // ScriptDataEscapedDashDash
        Tokeniser tDD1 = createTokeniser("");
        TokeniserState.ScriptDataEscapedDashDash.read(tDD1, tDD1.reader);
        Assert.assertEquals(TokeniserState.Data, tDD1.getState());

        Tokeniser tDD2 = createTokeniser("-");
        TokeniserState.ScriptDataEscapedDashDash.read(tDD2, tDD2.reader);

        Tokeniser tDD3 = createTokeniser("<");
        TokeniserState.ScriptDataEscapedDashDash.read(tDD3, tDD3.reader);
        Assert.assertEquals(TokeniserState.ScriptDataEscapedLessthanSign, tDD3.getState());

        Tokeniser tDD4 = createTokeniser(">");
        TokeniserState.ScriptDataEscapedDashDash.read(tDD4, tDD4.reader);
        Assert.assertEquals(TokeniserState.ScriptData, tDD4.getState());

        Tokeniser tDD5 = createTokeniser("\u0000");
        TokeniserState.ScriptDataEscapedDashDash.read(tDD5, tDD5.reader);
        Assert.assertEquals(TokeniserState.ScriptDataEscaped, tDD5.getState());

        Tokeniser tDD6 = createTokeniser("x");
        TokeniserState.ScriptDataEscapedDashDash.read(tDD6, tDD6.reader);
        Assert.assertEquals(TokeniserState.ScriptDataEscaped, tDD6.getState());

        // ScriptDataEscapedLessthanSign
        Tokeniser tEL1 = createTokeniser("s");
        TokeniserState.ScriptDataEscapedLessthanSign.read(tEL1, tEL1.reader);
        Assert.assertEquals(TokeniserState.ScriptDataDoubleEscapeStart, tEL1.getState());

        Tokeniser tEL2 = createTokeniser("/");
        TokeniserState.ScriptDataEscapedLessthanSign.read(tEL2, tEL2.reader);
        Assert.assertEquals(TokeniserState.ScriptDataEscapedEndTagOpen, tEL2.getState());

        Tokeniser tEL3 = createTokeniser("1");
        TokeniserState.ScriptDataEscapedLessthanSign.read(tEL3, tEL3.reader);
        Assert.assertEquals(TokeniserState.ScriptDataEscaped, tEL3.getState());

        // ScriptDataEscapedEndTagOpen & ScriptDataEscapedEndTagName
        Tokeniser tEEO1 = createTokeniser("s");
        TokeniserState.ScriptDataEscapedEndTagOpen.read(tEEO1, tEEO1.reader);
        Assert.assertEquals(TokeniserState.ScriptDataEscapedEndTagName, tEEO1.getState());

        Tokeniser tEEO2 = createTokeniser("1");
        TokeniserState.ScriptDataEscapedEndTagOpen.read(tEEO2, tEEO2.reader);
        Assert.assertEquals(TokeniserState.ScriptDataEscaped, tEEO2.getState());

        Tokeniser tEEN1 = createTokeniser("cript>");
        tEEN1.createTagPending(false);
        TokeniserState.ScriptDataEscapedEndTagName.read(tEEN1, tEEN1.reader);
    }

    @Test

        TokeniserState.ScriptDataDoubleEscapeStart.read(t1Fallback, t1Fallback.reader);

        Tokeniser t1Def = createTokeniser("?");
        TokeniserState.ScriptDataDoubleEscapeStart.read(t1Def, t1Def.reader);

        Tokeniser tSDEDD2 = createTokeniser("<");

    @Test
    public void testAttributeStates() {
        // BeforeAttributeName
        for (char c : new char[]{'\t', '\n', '\r', '\f', ' '}) {
            Tokeniser t = createTokeniser(String.valueOf(c));
            t.createTagPending(true);
            TokeniserState.BeforeAttributeName.read(t, t.reader);
        }

        Tokeniser tBAN1 = createTokeniser("/");
        tBAN1.createTagPending(true);
        TokeniserState.BeforeAttributeName.read(tBAN1, tBAN1.reader);
        Assert.assertEquals(TokeniserState.SelfClosingStartTag, tBAN1.getState());

        Tokeniser tBAN2 = createTokeniser(">");
        tBAN2.createTagPending(true);
        TokeniserState.BeforeAttributeName.read(tBAN2, tBAN2.reader);
        Assert.assertEquals(TokeniserState.Data, tBAN2.getState());

        Tokeniser tBAN3 = createTokeniser("\u0000");
        tBAN3.createTagPending(true);
        TokeniserState.BeforeAttributeName.read(tBAN3, tBAN3.reader);
        Assert.assertEquals(TokeniserState.AttributeName, tBAN3.getState());

        Tokeniser tBAN4 = createTokeniser("");
        tBAN4.createTagPending(true);
        TokeniserState.BeforeAttributeName.read(tBAN4, tBAN4.reader);
        Assert.assertEquals(TokeniserState.Data, tBAN4.getState());

        for (char c : new char[]{'\"', '\'', '<', '='}) {
            Tokeniser t = createTokeniser(String.valueOf(c));
            t.createTagPending(true);
            TokeniserState.BeforeAttributeName.read(t, t.reader);
            Assert.assertEquals(TokeniserState.AttributeName, t.getState());
        }

        Tokeniser tBAN5 = createTokeniser("attr");
        tBAN5.createTagPending(true);
        TokeniserState.BeforeAttributeName.read(tBAN5, tBAN5.reader);
        Assert.assertEquals(TokeniserState.AttributeName, tBAN5.getState());

        // AttributeName
        Tokeniser tAN1 = createTokeniser("attr ");
        tAN1.createTagPending(true);
        TokeniserState.AttributeName.read(tAN1, tAN1.reader);
        Assert.assertEquals(TokeniserState.AfterAttributeName, tAN1.getState());

        Tokeniser tAN2 = createTokeniser("attr/");
        tAN2.createTagPending(true);
        TokeniserState.AttributeName.read(tAN2, tAN2.reader);
        Assert.assertEquals(TokeniserState.SelfClosingStartTag, tAN2.getState());

        Tokeniser tAN3 = createTokeniser("attr=");
        tAN3.createTagPending(true);
        TokeniserState.AttributeName.read(tAN3, tAN3.reader);
        Assert.assertEquals(TokeniserState.BeforeAttributeValue, tAN3.getState());

        Tokeniser tAN4 = createTokeniser("attr>");
        tAN4.createTagPending(true);
        TokeniserState.AttributeName.read(tAN4, tAN4.reader);
        Assert.assertEquals(TokeniserState.Data, tAN4.getState());

        Tokeniser tAN5 = createTokeniser("attr\u0000");
        tAN5.createTagPending(true);
        TokeniserState.AttributeName.read(tAN5, tAN5.reader);

        Tokeniser tAN6 = createTokeniser("attr");
        tAN6.createTagPending(true);
        TokeniserState.AttributeName.read(tAN6, tAN6.reader);
